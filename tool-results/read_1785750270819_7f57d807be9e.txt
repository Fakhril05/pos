    69→    69→const formatRp = (amount: number) =>
    70→    70→  new Intl.NumberFormat('id-ID', {
    71→    71→    style: 'currency',
    72→    72→    currency: 'IDR',
    73→    73→    minimumFractionDigits: 0,

    75→    75→
    76→    76→const formatTimeAgo = (dateStr: string): string => {
    77→    77→  const now = Date.now();
    78→    78→  const then = new Date(dateStr).getTime();
    79→    79→  const diffMs = now - then;
    80→    80→  const seconds = Math.floor(diffMs / 1000);
    81→    81→  const minutes = Math.floor(seconds / 60);
    82→    82→  const hours = Math.floor(minutes / 60);
    83→    83→  const days = Math.floor(hours / 24);
    84→    84→  if (days > 0) return `${days} hari lalu`;
    85→    85→  if (hours > 0) return `${hours} jam lalu`;
    86→    86→  if (minutes > 0) return `${minutes} menit lalu`;
    87→    87→  return 'Baru saja';
    88→    88→};
    89→    89→
    90→    90→const formatCompact = (amount: number) => {
    91→    91→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    92→    92→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    93→    93→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    94→    94→  return String(amount);
    95→    95→};
