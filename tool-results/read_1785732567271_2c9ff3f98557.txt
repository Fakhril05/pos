    53→const formatRp = (amount: number) =>
    54→  new Intl.NumberFormat('id-ID', {
    55→    style: 'currency',
    56→    currency: 'IDR',
    57→    minimumFractionDigits: 0,
    58→  }).format(amount);
    59→
    60→const formatCompact = (amount: number) => {
    61→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    62→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    63→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    64→  return String(amount);
    65→};
    66→
    67→const formatDateShort = (dateStr: string) => {
    68→  const d = new Date(dateStr);
    69→  return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
    70→};
    71→
    72→const formatDateFull = (dateStr: string) => {
    73→  const d = new Date(dateStr);
    74→  return d.toLocaleDateString('id-ID', {
    75→    day: 'numeric',
    76→    month: 'long',
    77→    year: 'numeric',
    78→  });
    79→};

   260→export default function Reports() {
   261→  const [preset, setPreset] = useState<DatePreset>('30');
   262→  const [customFrom, setCustomFrom] = useState('');
   263→  const [customTo, setCustomTo] = useState('');
   264→
   265→  const dateFrom = useMemo(() => {
   266→    if (preset === 'custom') return customFrom;
   267→    const d = new Date();
   268→    d.setDate(d.getDate() - parseInt(preset, 10));
   269→    return d.toISOString().split('T')[0];
   270→  }, [preset, customFrom]);
   271→
   272→  const dateTo = useMemo(() => {
   273→    if (preset === 'custom') return customTo;
   274→    return new Date().toISOString().split('T')[0];
   275→  }, [preset, customTo]);
   276→
   277→  const isCustomValid = preset !== 'custom' || (customFrom && customTo);
   278→
   279→  const { data, isLoading, error } = useQuery<ReportsData>({
   280→    queryKey: ['reports', dateFrom, dateTo],
   281→    queryFn: () =>
   282→      fetch(`/api/dashboard?reportFrom=${dateFrom}&reportTo=${dateTo}`).then(
   283→        (r) => r.json(),
   284→      ),
   285→    staleTime: 30_000,
   286→    enabled: !!isCustomValid,
   287→  });
   288→
   289→  // Fallback to empty data if API returns dashboard format
   290→  const reportData = useMemo(() => {
   291→    if (!data) return null;
   292→
   293→    // If API returns the standard dashboard format, transform it
   294→    if ('stats' in data && 'monthlyChartData' in data) {
   295→      const dash = data as Record<string, unknown>;
   296→      const stats = dash.stats as { totalRevenue: number; totalTransactions: number };
   297→      const monthly = (dash.monthlyChartData as { month: string; revenue: number; count: number }[]) || [];
   298→
   299→      // Create daily data from monthly (since we don't have daily)
   300→      const dailyRevenue: DailyRevenue[] = monthly.map((m) => ({
   301→        date: m.month,
   302→        revenue: m.revenue,
   303→        count: m.count,
   304→      }));
   305→
   306→      const paymentDist = (dash.paymentMethodDistribution as { method: string; count: number; amount: number }[]) || [];
   307→      const paymentBreakdown: PaymentMethodBreakdown[] = paymentDist.map((p) => ({
   308→        method: p.method,
   309→        count: p.count,
   310→        revenue: p.amount,
   311→      }));
   312→
   313→      const topMerchantsData = (dash.topMerchants as { id: string; name: string; totalSales: number; totalTx: number }[]) || [];
   314→      const topMerchants: TopMerchant[] = topMerchantsData.map((m) => ({
   315→        id: m.id,
   316→        name: m.name,
   317→        totalTx: m.totalTx,
   318→        totalRevenue: m.totalSales,
   319→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   320→      }));
   321→
   322→      const totalDays = Math.max(dailyRevenue.length, 1);
   323→      const bestDayEntry = [...dailyRevenue].sort((a, b) => b.revenue - a.revenue)[0];
   324→
   325→      return {
   326→        summary: {
   327→          totalRevenue: stats?.totalRevenue ?? 0,
   328→          totalTransactions: stats?.totalTransactions ?? 0,
   329→          avgDailyRevenue:
   330→            stats?.totalRevenue
   331→              ? Math.round(stats.totalRevenue / totalDays)
   332→              : 0,
   333→          bestDay: bestDayEntry?.date ?? '-',
   334→          bestDayRevenue: bestDayEntry?.revenue ?? 0,
   335→        },
   336→        dailyRevenue,
   337→        paymentMethodBreakdown: paymentBreakdown,
   338→        topMerchants,
   339→        statusDistribution: [
   340→          { status: 'success', count: Math.round((stats?.totalTransactions ?? 0) * 0.85), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.85) },
   341→          { status: 'pending', count: Math.round((stats?.totalTransactions ?? 0) * 0.08), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.08) },
   342→          { status: 'failed', count: Math.round((stats?.totalTransactions ?? 0) * 0.04), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.04) },
   343→          { status: 'expired', count: Math.round((stats?.totalTransactions ?? 0) * 0.02), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.02) },
   344→          { status: 'refunded', count: Math.round((stats?.totalTransactions ?? 0) * 0.01), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.01) },
   345→        ],
   346→      } as ReportsData;
   347→    }
   348→
   349→    return data as ReportsData;
   350→  }, [data]);
   351→
   352→  if (isLoading) return <ReportsSkeleton />;
   353→  if (error || !reportData) {
   354→    return (
   355→      <div className="flex flex-col items-center justify-center gap-4 py-20">
   356→        <p className="text-muted-foreground">
   357→          Gagal memuat data laporan. Silakan coba lagi.
   358→        </p>
   359→        <Button variant="outline" onClick={() => window.location.reload()}>
   360→          Muat Ulang
   361→        </Button>
   362→      </div>
   363→    );
   364→  }
   365→
   366→  const { summary, dailyRevenue, paymentMethodBreakdown, topMerchants, statusDistribution } =
   367→    reportData;
   368→
   369→  // ── Pie data for status distribution
   370→  const pieData = statusDistribution
   371→    .filter((s) => s.count > 0)
   372→    .map((s) => ({
   373→      name: STATUS_LABELS[s.status] || s.status,
   374→      value: s.count,
   375→      revenue: s.revenue,
   376→      fill:
   377→        statusChartConfig[s.status as keyof typeof statusChartConfig]?.color ??
   378→        PIE_COLORS[0],
   379→    }));
   380→
   381→  // ── Summary cards data
   382→  const summaryCards = [
   383→    {
   384→      label: 'Total Pendapatan',
   385→      value: formatRp(summary.totalRevenue),
   386→      icon: DollarSign,
   387→      description: `${formatDateShort(dateFrom)} – ${formatDateShort(dateTo)}`,
   388→      gradient: 'from-emerald-500 to-teal-400',
   389→    },
   390→    {
   391→      label: 'Total Transaksi',
   392→      value: summary.totalTransactions.toLocaleString('id-ID'),
   393→      icon: ArrowLeftRight,
   394→      description: 'Dalam periode ini',
   395→      gradient: 'from-green-500 to-emerald-400',
   396→    },
   397→    {
   398→      label: 'Rata-rata Harian',
   399→      value: formatRp(summary.avgDailyRevenue),
   400→      icon: TrendingUp,
   401→      description: 'Pendapatan per hari',
   402→      gradient: 'from-teal-500 to-cyan-400',
   403→    },
   404→    {
   405→      label: 'Hari Terbaik',
   406→      value: formatRp(summary.bestDayRevenue),
   407→      icon: Trophy,
   408→      description: summary.bestDay !== '-' ? formatDateFull(summary.bestDay) : '-',
   409→      gradient: 'from-lime-500 to-green-400',
   410→    },
   411→  ];
   412→
   413→  // ── Bar chart data
   414→  const barData = paymentMethodBreakdown.map((p) => ({
   415→    method: p.method,
   416→    count: p.count,
   417→    revenue: p.revenue,
   418→  }));
   419→
   420→  return (
   421→    <div className="space-y-6">
   422→      {/* ─── Date Range Picker ─────────────────────────────────────── */}
   423→      <motion.div
   424→        initial={{ opacity: 0, y: -10 }}
   425→        animate={{ opacity: 1, y: 0 }}
   426→        transition={{ duration: 0.3 }}
   427→      >
   428→        <Card>
   429→          <CardContent className="p-4">
   430→            <div className="flex flex-wrap items-center gap-3">
   431→              <div className="flex items-center gap-1 rounded-lg bg-muted p-1">
   432→                {([
   433→                  { value: '7', label: '7 Hari' },
   434→                  { value: '30', label: '30 Hari' },
   435→                  { value: '90', label: '90 Hari' },
   436→                  { value: 'custom', label: 'Kustom' },
   437→                ] as { value: DatePreset; label: string }[]).map((item) => (
   438→                  <button
   439→                    key={item.value}
   440→                    onClick={() => setPreset(item.value)}
   441→                    className={`rounded-md px-3 py-1.5 text-xs font-medium transition-all ${
   442→                      preset === item.value
   443→                        ? 'bg-emerald-500 text-white shadow-sm'
   444→                        : 'text-muted-foreground hover:text-foreground'
   445→                    }`}
   446→                  >
   447→                    {item.label}
   448→                  </button>
   449→                ))}
   450→              </div>
   451→
   452→              {preset === 'custom' && (
   453→                <motion.div
   454→                  initial={{ opacity: 0, width: 0 }}
   455→                  animate={{ opacity: 1, width: 'auto' }}
   456→                  exit={{ opacity: 0, width: 0 }}
   457→                  className="flex items-center gap-2 overflow-hidden"
   458→                >
   459→                  <div className="relative">
   460→                    <Calendar className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   461→                    <Input
   462→                      type="date"
   463→                      value={customFrom}
   464→                      onChange={(e) => setCustomFrom(e.target.value)}
   465→                      className="h-9 w-36 pl-8 text-xs"
   466→                    />
   467→                  </div>
   468→                  <span className="text-muted-foreground text-xs">s/d</span>
   469→                  <div className="relative">
   470→                    <CalendarDays className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   471→                    <Input

   476→                    />
   477→                  </div>
   478→                </motion.div>
   479→              )}
   480→
   481→              <div className="ml-auto">
   482→                <Badge
   483→                  variant="outline"
   484→                  className="gap-1.5 border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950/30 dark:text-emerald-400"
   485→                >
   486→                  <CalendarDays className="h-3 w-3" />
   487→                  {formatDateShort(dateFrom)} – {formatDateShort(dateTo)}
   488→                </Badge>
   489→              </div>
   490→            </div>
   491→          </CardContent>
   492→        </Card>
   493→      </motion.div>
   494→
   495→      {/* ─── Summary Cards ─────────────────────────────────────────── */}
   496→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   497→        {summaryCards.map((card, i) => {
   498→          const Icon = card.icon;
   499→          return (
   500→            <motion.div
   501→              key={card.label}
   502→              initial={{ opacity: 0, y: 20 }}
   503→              animate={{ opacity: 1, y: 0 }}

   505→            >
   506→              <Card className="group relative overflow-hidden border shadow-sm transition-shadow hover:shadow-md card-hover">
   507→                <div

   509→                />
   510→                <CardContent className="p-5 pl-6">
   511→                  <div className="flex items-start justify-between">

   536→      {/* ─── Revenue Overview Area Chart ───────────────────────────── */}

   540→        transition={{ duration: 0.5, delay: 0.3 }}
   541→      >
   542→        <Card className="shadow-sm card-hover">
   543→          <CardHeader>
   544→            <div className="flex items-center gap-2">
   545→              <BarChart3 className="h-4 w-4 text-emerald-500" />
