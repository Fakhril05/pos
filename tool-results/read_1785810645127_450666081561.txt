   135→function randomBetween(min: number, max: number) {
   136→  return Math.random() * (max - min) + min
   137→}
   138→
   139→function randomItem<T>(arr: T[]): T {
   140→  return arr[Math.floor(Math.random() * arr.length)]
   141→}
   142→
   143→function randomDate(daysBack: number): Date {
   144→  const now = new Date()
   145→  const past = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000)
   146→  return new Date(past.getTime() + Math.random() * (now.getTime() - past.getTime()))
   147→}

   149→function generateSlug(name: string): string {
   150→  return name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')

   484→      const taxAmount = Math.round((subtotalAmount - discountAmount) * 0.11)
   485→      const totalAmount = subtotalAmount - discountAmount + taxAmount
   486→
   487→      const createdAt = randomDate(30)
   488→      const paidAt = status === 'paid' ? new Date(createdAt.getTime() + randomBetween(1000, 300000)) : null
   489→      const voidedAt = status === 'void' ? new Date(createdAt.getTime() + randomBetween(5000, 60000)) : null
   490→
   491→      const invoiceNo = `INV-${createdAt.toISOString().slice(0, 10).replace(/-/g, '')}-${String(i + 1).padStart(5, '0')}`
   492→
   493→      const method = randomItem(paymentMethods)
   494→      let channel = ''
   495→      if (method === 'QRIS') channel = 'QRIS'
   496→      else if (method === 'Bank Transfer') channel = randomItem(['BCA', 'BNI', 'BRI', 'Mandiri'])
   497→      else if (method === 'E-Wallet') channel = randomItem(['GoPay', 'OVO', 'DANA', 'ShopeePay'])
   498→      else if (method === 'Cash') channel = 'Tunai'
   499→      else if (method === 'Virtual Account') channel = randomItem(['BCA VA', 'BNI VA', 'BRI VA'])
   500→      else channel = randomItem(['Visa', 'Mastercard'])
   501→
   502→      const fee = Math.round(totalAmount * randomBetween(0.005, 0.03))
   503→      const paymentStatus = status === 'paid' ? 'fully_paid' : status === 'pending' ? 'unpaid' : 'unpaid'
   504→
   505→      // Create Transaction
   506→      const transaction = await db.transaction.create({
   507→        data: {
   508→          tenantId: tenant.id,
   509→          outletId: outlet.id,
   510→          cashierId: cashier.id,
   511→          invoiceNo,
   512→          customerId: customer?.id || null,
   513→          subtotalAmount,
   514→          discountAmount,
   515→          taxAmount,
   516→          totalAmount,
   517→          status,
   518→          paymentStatus,
   519→          customerName: customer?.name || null,
   520→          customerPhone: customer?.phone || null,
   521→          paidAt,
   522→          voidedAt,
   523→          voidReason: status === 'void' ? 'Barang rusak / salah input' : null,
   524→          createdAt,
   525→          updatedAt: createdAt,
   526→          items: {
   527→            create: items.map((item) => ({
   528→              ...item,
   529→              productId: items.indexOf(item) < tenantProducts.length ? tenantProducts[selectedProducts.values().next().value || 0]?.id || tenantProducts[0].id : tenantProducts[0].id,
   530→            })),
   531→          },
   532→          payments: status === 'paid' ? {
   533→            create: {
   534→              paymentMethod: method,
   535→              paymentChannel: channel,
   536→              amount: totalAmount,
   537→              fee,
   538→              netAmount: totalAmount - fee,
   539→              status: 'success',
   540→              paidAt,
   541→            },
   542→          } : undefined,
   543→        },
   544→      })
   545→
   546→      // Fix product IDs in items (re-query to update)
   547→      // Update transaction stats
   548→      if (status === 'paid') {
   549→        const stats = transactionStats.get(tenant.id) || { totalSales: 0, totalTx: 0 }
   550→        stats.totalSales += totalAmount
   551→        stats.totalTx += 1
   552→        transactionStats.set(tenant.id, stats)
   553→
   554→        if (customer) {
   555→          await db.customer.update({
   556→            where: { id: customer.id },
   557→            data: {
   558→              totalSpent: { increment: totalAmount },
   559→              totalTx: { increment: 1 },
   560→              loyaltyPoints: { increment: Math.floor(totalAmount / 10000) },
   561→            },
   562→          })
   563→        }
   564→      }
   565→    }
   566→
   567→    // Update tenant aggregate stats
   568→    for (const [tenantId, stats] of transactionStats.entries()) {
   569→      await db.tenant.update({
   570→        where: { id: tenantId },
   571→        data: {
   572→          totalSales: stats.totalSales,
   573→          totalTx: stats.totalTx,
   574→        },
   575→      })
   576→    }
   577→
   578→    // ============================================================
   579→    // 9. Create Shifts
   580→    // ============================================================
   581→    const shiftCount = 30
   582→    for (let i = 0; i < shiftCount; i++) {
   583→      const tenant = randomItem(tenants)
   584→      const tenantOutlets = outlets.filter((o) => o.tenantId === tenant.id)
   585→      const tenantCashiers = allUsers.filter((u) => u.tenantId === tenant.id && u.role === 'cashier')
   586→      const outlet = randomItem(tenantOutlets) || outlets[0]
   587→      const cashier = randomItem(tenantCashiers) || allUsers[1]
   588→
   589→      const isClosed = Math.random() > 0.3
   590→      const openedAt = randomDate(7)
   591→      const shiftSales = Math.round(randomBetween(500000, 5000000))
   592→
   593→      await db.shift.create({
   594→        data: {
   595→          tenantId: tenant.id,
   596→          outletId: outlet.id,
   597→          cashierId: cashier.id,
   598→          status: isClosed ? 'closed' : 'open',
   599→          openedAt,
   600→          closedAt: isClosed ? new Date(openedAt.getTime() + randomBetween(8 * 3600000, 12 * 3600000)) : null,
   601→          openingCash: 500000,
   602→          closingCash: isClosed ? 500000 + shiftSales : null,
   603→          expectedCash: isClosed ? 500000 + shiftSales : null,
   604→          cashDifference: isClosed ? Math.round(randomBetween(-5000, 5000)) : null,
   605→          totalSales: shiftSales,
   606→          totalTx: Math.floor(randomBetween(10, 80)),
   607→          totalRefunds: Math.round(randomBetween(0, 200000)),
   608→        },
   609→      })
   610→    }
   611→
   612→    // ============================================================
   613→    // 10. Create Payment Links
   614→    // ============================================================
   615→    const linkTitles = ['Pembayaran Order', 'Donasi', 'Tagihan Bulanan', 'Biaya Pendaftaran', 'Langganan Premium', 'Pembelian Produk']
   616→    for (let i = 0; i < 25; i++) {
   617→      const tenant = randomItem(tenants)
   618→      const amount = Math.round(randomBetween(15000, 2000000))
   619→      const createdAt = randomDate(15)
   620→
   621→      await db.paymentLink.create({
   622→        data: {
   623→          tenantId: tenant.id,
   624→          title: randomItem(linkTitles),
   625→          description: `Link pembayaran untuk ${tenant.name}`,
   626→          amount,
   627→          status: Math.random() > 0.3 ? 'active' : 'expired',
   628→          linkType: Math.random() > 0.5 ? 'single' : 'multi',
   629→          maxQty: Math.floor(randomBetween(1, 100)),
   630→          soldQty: Math.floor(randomBetween(0, 50)),
   631→          expiresAt: new Date(createdAt.getTime() + 7 * 24 * 60 * 60 * 1000),
   632→          createdAt,
   633→          updatedAt: createdAt,
   634→        },
   635→      })
   636→    }
   637→
   638→    // ============================================================
   639→    // 11. Create Payouts
   640→    // ============================================================
   641→    const bankNames = ['BCA', 'BNI', 'BRI', 'Mandiri', 'Permata', 'CIMB']
   642→    for (let i = 0; i < 15; i++) {
   643→      const tenant = randomItem(tenants)
   644→      const amount = Math.round(randomBetween(500000, 25000000))
   645→      const fee = Math.round(amount * 0.005)
   646→      const status = randomItem(['completed', 'pending', 'processing'])
   647→      const requestedAt = randomDate(14)
   648→
   649→      await db.payout.create({
   650→        data: {
   651→          tenantId: tenant.id,
   652→          amount,
   653→          fee,
   654→          status,
   655→          bankName: randomItem(bankNames),
   656→          accountNo: `${Math.floor(1000000000 + Math.random() * 9000000000)}`,
   657→          accountName: tenant.name,
   658→          notes: 'Payout rutin mingguan',
   659→          processedAt: status === 'completed' ? new Date(requestedAt.getTime() + randomBetween(3600000, 172800000)) : null,
   660→          requestedAt,
   661→          createdAt: requestedAt,
   662→          updatedAt: requestedAt,
   663→        },
   664→      })
   665→    }
   666→
   667→    // ============================================================
   668→    // 12. Create Refunds (for some paid transactions)
   669→    // ============================================================
   670→    const paidTransactions = await db.transaction.findMany({
   671→      where: { status: 'paid' },
   672→      take: 10,
   673→      orderBy: { createdAt: 'desc' },
   674→    })
   675→
   676→    for (const tx of paidTransactions) {
   677→      if (Math.random() > 0.5) continue
   678→      await db.refund.create({
   679→        data: {
   680→          transactionId: tx.id,
   681→          amount: Math.round(tx.totalAmount * randomBetween(0.3, 1.0)),
   682→          reason: Math.random() > 0.5 ? 'Barang tidak sesuai' : 'Pembatalan oleh pelanggan',
   683→          status: randomItem(['completed', 'approved', 'pending']),
   684→          processedAt: Math.random() > 0.5 ? new Date() : null,
   685→        },
   686→      })
   687→
   688→      await db.transaction.update({
   689→        where: { id: tx.id },
   690→        data: { status: 'refunded' },
   691→      })
   692→    }
   693→
   694→    // ============================================================
   695→    // 13. Create Notifications for users
   696→    // ============================================================
   697→    const notifTemplates = [
   698→      { title: 'Pembayaran berhasil', message: 'Rp 2.500.000 diterima', type: 'success', category: 'transaction' },
   699→      { title: 'Stok menipis', message: 'Nasi Goreng Spesial sisa < 10 unit', type: 'warning', category: 'inventory' },
   700→      { title: 'Shift dibuka', message: 'Shift baru dimulai', type: 'info', category: 'shift' },
   701→      { title: 'Payout berhasil', message: 'Rp 15.000.000 telah ditransfer ke BCA', type: 'success', category: 'payout' },
   702→      { title: 'Pelanggan baru', message: 'Pelanggan baru terdaftar', type: 'info', category: 'system' },
   703→      { title: 'Transaksi gagal', message: 'Pembayaran expired', type: 'warning', category: 'transaction' },
   704→      { title: 'Refund diproses', message: 'Refund Rp 150.000 berhasil', type: 'info', category: 'refund' },
   705→    ]
   706→
   707→    for (const user of allUsers) {
   708→      if (user.role === 'super_admin' || Math.random() > 0.5) continue
   709→      const notifCount = Math.floor(randomBetween(1, 5))
   710→      for (let i = 0; i < notifCount; i++) {
   711→        const template = randomItem(notifTemplates)
   712→        await db.notification.create({
   713→          data: {
   714→            userId: user.id,
   715→            title: template.title,
   716→            message: template.message,
   717→            type: template.type,
   718→            category: template.category,
   719→            isRead: Math.random() > 0.6,
   720→            readAt: Math.random() > 0.6 ? randomDate(2) : null,
   721→          },
   722→        })
   723→      }
   724→    }
   725→
   726→    // ============================================================
   727→    // 14. Create API Keys for tenants
   728→    // ============================================================
   729→    for (const tenant of tenants) {
   730→      await db.apiKey.create({
   731→        data: {
   732→          tenantId: tenant.id,
   733→          name: 'Production Key',
   734→          key: `pkrl_${randomUUID().replace(/-/g, '')}`,
   735→          prefix: 'pkrl',
   736→          permissions: 'read_write',
   737→          isActive: true,
   738→        },
   739→      })
   740→    }
   741→
   742→    // ============================================================
   743→    // 15. Create Audit Log entries
   744→    // ============================================================
   745→    const auditActions = [
   746→      { action: 'login', entity: 'User', category: 'system' },
   747→      { action: 'create', entity: 'Transaction', category: 'transaction' },
   748→      { action: 'update', entity: 'Product', category: 'inventory' },
   749→      { action: 'void', entity: 'Transaction', category: 'transaction' },
   750→      { action: 'refund', entity: 'Transaction', category: 'refund' },
   751→    ]
   752→
   753→    for (let i = 0; i < 50; i++) {
   754→      const tenant = randomItem(tenants)
   755→      const tenantUsers = allUsers.filter((u) => u.tenantId === tenant.id)
   756→      const user = randomItem(tenantUsers) || superAdmin
   757→
   758→      const audit = randomItem(auditActions)
   759→      await db.auditLog.create({
   760→        data: {
   761→          tenantId: tenant.id,
   762→          userId: user.id,
   763→          action: audit.action,
   764→          entity: audit.entity,
   765→          entityId: randomUUID().replace(/-/g, '').slice(0, 12),
   766→          ipAddress: `192.168.${Math.floor(Math.random() * 255)}.${Math.floor(Math.random() * 255)}`,
   767→          createdAt: randomDate(7),
   768→        },
   769→      })
   770→    }
   771→
   772→    // ============================================================
   773→    // 16. Create Discounts for some tenants
   774→    // ============================================================
   775→    for (const tenant of tenants) {
   776→      if (Math.random() > 0.4) continue
   777→      const discount = await db.discount.create({
   778→        data: {
   779→          tenantId: tenant.id,
   780→          name: 'Diskon Akhir Bulan',
   781→          description: 'Diskon 10% untuk semua transaksi di atas Rp 100.000',
   782→          discountType: 'percentage',
   783→          discountValue: 10,
   784→          minPurchase: 100000,
   785→          maxDiscount: 50000,
   786→          appliesTo: 'all',
   787→          isActive: true,
   788→          startsAt: new Date(),
   789→          expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
   790→          maxUsage: 100,
   791→        },
   792→      })
   793→    }
   794→
   795→    // ============================================================
   796→    // Summary
   797→    // ============================================================
   798→    const summary = {
   799→      success: true,
   800→      message: 'Multi-tenant POS seed completed',
   801→      data: {
   802→        tenants: tenants.length,
   803→        outlets: outlets.length,
   804→        users: allUsers.length,
   805→        products: allProducts.length,
   806→        customers: allCustomers.length,
   807→        transactions: txCount,
   808→        shifts: shiftCount,
   809→        paymentLinks: 25,
   810→        payouts: 15,
   811→      },
   812→    }
   813→
   814→    return NextResponse.json(summary)
   815→  } catch (error) {

   818→  }
