   135→function randomBetween(min: number, max: number) {
   136→  return Math.random() * (max - min) + min
   137→}
   138→
   139→function randomItem<T>(arr: T[]): T {
   140→  return arr[Math.floor(Math.random() * arr.length)]
   141→}
   142→
   143→function randomDate(daysBack: number): Date {
   144→  const now = new Date()
   145→  const past = new Date(now.getTime() - daysBack * 24 * 60 * 60 * 1000)
   146→  return new Date(past.getTime() + Math.random() * (now.getTime() - past.getTime()))
   147→}

   149→function generateSlug(name: string): string {
   150→  return name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '')
   151→}

   492→      const tenant = randomItem(tenants)
   493→      const tenantOutlets = outlets.filter((o) => o.tenantId === tenant.id)
   494→      const tenantCashiers = allUsers.filter((u) => u.tenantId === tenant.id && u.role === 'cashier')
   495→      const tenantProducts = allProducts.filter((p) => p.tenantId === tenant.id)
   496→      const tenantCustomers = allCustomers.filter((c) => c.tenantId === tenant.id)
   497→
   498→      const outlet = randomItem(tenantOutlets) || outlets[0]
   499→      const cashier = randomItem(tenantCashiers) || allUsers[1]
   500→      const customer = Math.random() > 0.3 ? randomItem(tenantCustomers) : null
   501→      const status = randomItem(weightedStatuses)
   502→
   503→      // Generate line items (1-5 items per transaction)
   504→      const itemCount = Math.floor(randomBetween(1, 6))
   505→      const items: any[] = []
   506→      let subtotalAmount = 0
   507→
   508→      const selectedProducts = new Set<string>()
   509→      for (let j = 0; j < itemCount && j < tenantProducts.length; j++) {
   510→        let product
   511→        do {
   512→          product = randomItem(tenantProducts)
   513→        } while (selectedProducts.has(product.id))
   514→        selectedProducts.add(product.id)
   515→
   516→        const qty = Math.floor(randomBetween(1, 4))
   517→        const itemDiscount = Math.random() > 0.8 ? Math.round(randomBetween(0, 5000)) : 0
   518→        const totalPrice = (product.sellPrice * qty) - itemDiscount
   519→        subtotalAmount += totalPrice
   520→
   521→        items.push({
   522→          name: product.name,
   523→          sku: product.sku || null,
   524→          buyPrice: product.buyPrice,
   525→          sellPrice: product.sellPrice,
   526→          quantity: qty,
   527→          unit: product.unit,
   528→          discount: itemDiscount,
   529→          taxRate: 0,
   530→          totalPrice,
   531→        })
   532→      }
   533→
   534→      const discountAmount = Math.random() > 0.7 ? Math.round(subtotalAmount * 0.05) : 0
   535→      const taxAmount = Math.round((subtotalAmount - discountAmount) * 0.11)
   536→      const totalAmount = subtotalAmount - discountAmount + taxAmount
   537→
   538→      const createdAt = randomDate(30)
   539→      const paidAt = status === 'paid' ? new Date(createdAt.getTime() + randomBetween(1000, 300000)) : null
   540→      const voidedAt = status === 'void' ? new Date(createdAt.getTime() + randomBetween(5000, 60000)) : null
   541→
   542→      const invoiceNo = `INV-${createdAt.toISOString().slice(0, 10).replace(/-/g, '')}-${String(i + 1).padStart(5, '0')}`
   543→
   544→      const method = randomItem(paymentMethods)
   545→      let channel = ''
   546→      if (method === 'QRIS') channel = 'QRIS'
   547→      else if (method === 'Bank Transfer') channel = randomItem(['BCA', 'BNI', 'BRI', 'Mandiri'])
   548→      else if (method === 'E-Wallet') channel = randomItem(['GoPay', 'OVO', 'DANA', 'ShopeePay'])
   549→      else if (method === 'Cash') channel = 'Tunai'
   550→      else if (method === 'Virtual Account') channel = randomItem(['BCA VA', 'BNI VA', 'BRI VA'])
   551→      else channel = randomItem(['Visa', 'Mastercard'])
   552→
   553→      const fee = Math.round(totalAmount * randomBetween(0.005, 0.03))
   554→      const paymentStatus = status === 'paid' ? 'fully_paid' : status === 'pending' ? 'unpaid' : 'unpaid'
   555→
   556→      // Create Transaction
   557→      const transaction = await db.transaction.create({
   558→        data: {
   559→          tenantId: tenant.id,
   560→          outletId: outlet.id,
   561→          cashierId: cashier.id,
   562→          invoiceNo,
   563→          customerId: customer?.id || null,
   564→          subtotalAmount,
   565→          discountAmount,
   566→          taxAmount,
   567→          totalAmount,
   568→          status,
   569→          paymentStatus,
   570→          customerName: customer?.name || null,
   571→          customerPhone: customer?.phone || null,
   572→          paidAt,
   573→          voidedAt,
   574→          voidReason: status === 'void' ? 'Barang rusak / salah input' : null,
   575→          createdAt,
   576→          updatedAt: createdAt,
   577→          items: {
   578→            create: items.map((item) => ({
   579→              ...item,
   580→              productId: items.indexOf(item) < tenantProducts.length ? tenantProducts[selectedProducts.values().next().value || 0]?.id || tenantProducts[0].id : tenantProducts[0].id,
   581→            })),
   582→          },
   583→          payments: status === 'paid' ? {
   584→            create: {
   585→              paymentMethod: method,
   586→              paymentChannel: channel,
   587→              amount: totalAmount,
   588→              fee,
   589→              netAmount: totalAmount - fee,
   590→              status: 'success',
   591→              paidAt,
   592→            },
   593→          } : undefined,
   594→        },
   595→      })
   596→
   597→      // Fix product IDs in items (re-query to update)
   598→      // Update transaction stats
   599→      if (status === 'paid') {
   600→        const stats = transactionStats.get(tenant.id) || { totalSales: 0, totalTx: 0 }
   601→        stats.totalSales += totalAmount
   602→        stats.totalTx += 1
   603→        transactionStats.set(tenant.id, stats)
   604→
   605→        if (customer) {
   606→          await db.customer.update({
   607→            where: { id: customer.id },
   608→            data: {
   609→              totalSpent: { increment: totalAmount },
   610→              totalTx: { increment: 1 },
   611→              loyaltyPoints: { increment: Math.floor(totalAmount / 10000) },
   612→            },
   613→          })
   614→        }
   615→      }
   616→    }
   617→
   618→    // Update tenant aggregate stats
   619→    for (const [tenantId, stats] of transactionStats.entries()) {
   620→      await db.tenant.update({
   621→        where: { id: tenantId },
   622→        data: {
   623→          totalSales: stats.totalSales,
   624→          totalTx: stats.totalTx,
   625→        },
   626→      })
   627→    }
   628→
   629→    // ============================================================
   630→    // 9. Create Shifts
   631→    // ============================================================
   632→    const shiftCount = 30
   633→    for (let i = 0; i < shiftCount; i++) {
   634→      const tenant = randomItem(tenants)
   635→      const tenantOutlets = outlets.filter((o) => o.tenantId === tenant.id)
   636→      const tenantCashiers = allUsers.filter((u) => u.tenantId === tenant.id && u.role === 'cashier')
   637→      const outlet = randomItem(tenantOutlets) || outlets[0]
   638→      const cashier = randomItem(tenantCashiers) || allUsers[1]
   639→
   640→      const isClosed = Math.random() > 0.3
   641→      const openedAt = randomDate(7)
   642→      const shiftSales = Math.round(randomBetween(500000, 5000000))
   643→
   644→      await db.shift.create({
   645→        data: {
   646→          tenantId: tenant.id,
   647→          outletId: outlet.id,
   648→          cashierId: cashier.id,
   649→          status: isClosed ? 'closed' : 'open',
   650→          openedAt,
   651→          closedAt: isClosed ? new Date(openedAt.getTime() + randomBetween(8 * 3600000, 12 * 3600000)) : null,
   652→          openingCash: 500000,
   653→          closingCash: isClosed ? 500000 + shiftSales : null,
   654→          expectedCash: isClosed ? 500000 + shiftSales : null,
   655→          cashDifference: isClosed ? Math.round(randomBetween(-5000, 5000)) : null,
   656→          totalSales: shiftSales,
   657→          totalTx: Math.floor(randomBetween(10, 80)),
   658→          totalRefunds: Math.round(randomBetween(0, 200000)),
   659→        },
   660→      })
   661→    }
   662→
   663→    // ============================================================
   664→    // 10. Create Payment Links
   665→    // ============================================================
   666→    const linkTitles = ['Pembayaran Order', 'Donasi', 'Tagihan Bulanan', 'Biaya Pendaftaran', 'Langganan Premium', 'Pembelian Produk']
   667→    for (let i = 0; i < 25; i++) {
   668→      const tenant = randomItem(tenants)
   669→      const amount = Math.round(randomBetween(15000, 2000000))
   670→      const createdAt = randomDate(15)
   671→
   672→      await db.paymentLink.create({
   673→        data: {
   674→          tenantId: tenant.id,
   675→          title: randomItem(linkTitles),
   676→          description: `Link pembayaran untuk ${tenant.name}`,
   677→          amount,
   678→          status: Math.random() > 0.3 ? 'active' : 'expired',
   679→          linkType: Math.random() > 0.5 ? 'single' : 'multi',
   680→          maxQty: Math.floor(randomBetween(1, 100)),
   681→          soldQty: Math.floor(randomBetween(0, 50)),
   682→          expiresAt: new Date(createdAt.getTime() + 7 * 24 * 60 * 60 * 1000),
   683→          createdAt,
   684→          updatedAt: createdAt,
   685→        },
   686→      })
   687→    }
   688→
   689→    // ============================================================
   690→    // 11. Create Payouts
   691→    // ============================================================
   692→    const bankNames = ['BCA', 'BNI', 'BRI', 'Mandiri', 'Permata', 'CIMB']
   693→    for (let i = 0; i < 15; i++) {
   694→      const tenant = randomItem(tenants)
   695→      const amount = Math.round(randomBetween(500000, 25000000))
   696→      const fee = Math.round(amount * 0.005)
   697→      const status = randomItem(['completed', 'pending', 'processing'])
   698→      const requestedAt = randomDate(14)
   699→
   700→      await db.payout.create({
   701→        data: {
   702→          tenantId: tenant.id,
   703→          amount,
   704→          fee,
   705→          status,
   706→          bankName: randomItem(bankNames),
   707→          accountNo: `${Math.floor(1000000000 + Math.random() * 9000000000)}`,
   708→          accountName: tenant.name,
   709→          notes: 'Payout rutin mingguan',
   710→          processedAt: status === 'completed' ? new Date(requestedAt.getTime() + randomBetween(3600000, 172800000)) : null,
   711→          requestedAt,
   712→          createdAt: requestedAt,
   713→          updatedAt: requestedAt,
   714→        },
   715→      })
   716→    }
   717→
   718→    // ============================================================
   719→    // 12. Create Refunds (for some paid transactions)
   720→    // ============================================================
   721→    const paidTransactions = await db.transaction.findMany({
   722→      where: { status: 'paid' },
   723→      take: 10,
   724→      orderBy: { createdAt: 'desc' },
   725→    })
   726→
   727→    for (const tx of paidTransactions) {
   728→      if (Math.random() > 0.5) continue
   729→      await db.refund.create({
   730→        data: {
   731→          transactionId: tx.id,
   732→          amount: Math.round(tx.totalAmount * randomBetween(0.3, 1.0)),
   733→          reason: Math.random() > 0.5 ? 'Barang tidak sesuai' : 'Pembatalan oleh pelanggan',
   734→          status: randomItem(['completed', 'approved', 'pending']),
   735→          processedAt: Math.random() > 0.5 ? new Date() : null,
   736→        },
   737→      })
   738→
   739→      await db.transaction.update({
   740→        where: { id: tx.id },
   741→        data: { status: 'refunded' },
   742→      })
   743→    }
   744→
   745→    // ============================================================
   746→    // 13. Create Notifications for users
   747→    // ============================================================
   748→    const notifTemplates = [
   749→      { title: 'Pembayaran berhasil', message: 'Rp 2.500.000 diterima', type: 'success', category: 'transaction' },
   750→      { title: 'Stok menipis', message: 'Nasi Goreng Spesial sisa < 10 unit', type: 'warning', category: 'inventory' },
   751→      { title: 'Shift dibuka', message: 'Shift baru dimulai', type: 'info', category: 'shift' },
   752→      { title: 'Payout berhasil', message: 'Rp 15.000.000 telah ditransfer ke BCA', type: 'success', category: 'payout' },
   753→      { title: 'Pelanggan baru', message: 'Pelanggan baru terdaftar', type: 'info', category: 'system' },
   754→      { title: 'Transaksi gagal', message: 'Pembayaran expired', type: 'warning', category: 'transaction' },
   755→      { title: 'Refund diproses', message: 'Refund Rp 150.000 berhasil', type: 'info', category: 'refund' },
   756→    ]
   757→
   758→    for (const user of allUsers) {
   759→      if (user.role === 'super_admin' || Math.random() > 0.5) continue
   760→      const notifCount = Math.floor(randomBetween(1, 5))
   761→      for (let i = 0; i < notifCount; i++) {
   762→        const template = randomItem(notifTemplates)
   763→        await db.notification.create({
   764→          data: {
   765→            userId: user.id,
   766→            title: template.title,
   767→            message: template.message,
   768→            type: template.type,
   769→            category: template.category,
   770→            isRead: Math.random() > 0.6,
   771→            readAt: Math.random() > 0.6 ? randomDate(2) : null,
   772→          },
   773→        })
   774→      }
   775→    }
   776→
   777→    // ============================================================
   778→    // 14. Create API Keys for tenants
   779→    // ============================================================
   780→    for (const tenant of tenants) {
   781→      await db.apiKey.create({
   782→        data: {
   783→          tenantId: tenant.id,
   784→          name: 'Production Key',
   785→          key: `pkrl_${randomUUID().replace(/-/g, '')}`,
   786→          prefix: 'pkrl',
   787→          permissions: 'read_write',
   788→          isActive: true,
   789→        },
   790→      })
   791→    }
   792→
   793→    // ============================================================
   794→    // 15. Create Audit Log entries
   795→    // ============================================================
   796→    const auditActions = [
   797→      { action: 'login', entity: 'User', category: 'system' },
   798→      { action: 'create', entity: 'Transaction', category: 'transaction' },
   799→      { action: 'update', entity: 'Product', category: 'inventory' },
   800→      { action: 'void', entity: 'Transaction', category: 'transaction' },
   801→      { action: 'refund', entity: 'Transaction', category: 'refund' },
   802→    ]
   803→
   804→    for (let i = 0; i < 50; i++) {
   805→      const tenant = randomItem(tenants)
   806→      const tenantUsers = allUsers.filter((u) => u.tenantId === tenant.id)
   807→      const user = randomItem(tenantUsers) || superAdmin
   808→
   809→      const audit = randomItem(auditActions)
   810→      await db.auditLog.create({
   811→        data: {
   812→          tenantId: tenant.id,
   813→          userId: user.id,
   814→          action: audit.action,
   815→          entity: audit.entity,
   816→          entityId: randomUUID().replace(/-/g, '').slice(0, 12),
   817→          ipAddress: `192.168.${Math.floor(Math.random() * 255)}.${Math.floor(Math.random() * 255)}`,
   818→          createdAt: randomDate(7),
   819→        },
   820→      })
   821→    }
   822→
   823→    // ============================================================
   824→    // 16. Create Discounts for some tenants
   825→    // ============================================================
   826→    for (const tenant of tenants) {
   827→      if (Math.random() > 0.4) continue
   828→      const discount = await db.discount.create({
   829→        data: {
   830→          tenantId: tenant.id,
   831→          name: 'Diskon Akhir Bulan',
   832→          description: 'Diskon 10% untuk semua transaksi di atas Rp 100.000',
   833→          discountType: 'percentage',
   834→          discountValue: 10,
   835→          minPurchase: 100000,
   836→          maxDiscount: 50000,
   837→          appliesTo: 'all',
   838→          isActive: true,
   839→          startsAt: new Date(),
   840→          expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
   841→          maxUsage: 100,
   842→        },
   843→      })
   844→    }
   845→
   846→    // ============================================================
   847→    // Summary
   848→    // ============================================================
   849→    const summary = {
   850→      success: true,
   851→      message: 'Multi-tenant POS seed completed',
   852→      data: {
   853→        tenants: tenants.length,
   854→        outlets: outlets.length,
   855→        users: allUsers.length,
   856→        products: allProducts.length,
   857→        customers: allCustomers.length,
   858→        transactions: txCount,
   859→        shifts: shiftCount,
   860→        paymentLinks: 25,
   861→        payouts: 15,
   862→      },
   863→    }
   864→
   865→    return NextResponse.json(summary)
   866→  } catch (error) {
   867→    console.error('Seed error:', error)
   868→    return NextResponse.json({ error: 'Seed failed', details: String(error) }, { status: 500 })
   869→  }
   870→}
