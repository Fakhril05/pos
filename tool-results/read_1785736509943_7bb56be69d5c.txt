    61→    61→  new Intl.NumberFormat('id-ID', {

    67→    67→const formatTimeAgo = (dateStr: string): string => {
    68→    68→  const now = Date.now();
    69→    69→  const then = new Date(dateStr).getTime();
    70→    70→  const diffMs = now - then;
    71→    71→  const seconds = Math.floor(diffMs / 1000);
    72→    72→  const minutes = Math.floor(seconds / 60);
    73→    73→  const hours = Math.floor(minutes / 60);
    74→    74→  const days = Math.floor(hours / 24);
    75→    75→  if (days > 0) return `${days} hari lalu`;
    76→    76→  if (hours > 0) return `${hours} jam lalu`;
    77→    77→  if (minutes > 0) return `${minutes} menit lalu`;
    78→    78→  return 'Baru saja';
    79→    79→};
    80→    80→
    81→    81→const formatCompact = (amount: number) => {
    82→    82→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    83→    83→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    84→    84→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    85→    85→  return String(amount);
