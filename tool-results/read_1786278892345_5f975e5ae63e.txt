   150→// ── Component ──────────────────────────────────────────────────────────
   151→
   152→export function Kasir() {
   153→  const user = useAppStore((s) => s.user)
   154→  const queryClient = useQueryClient()
   155→
   156→  const [search, setSearch] = useState('')
   157→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   158→  const [cart, setCart] = useState<CartItem[]>([])
   159→  const [selectedPayment, setSelectedPayment] = useState<(typeof PAYMENT_METHODS)[number] | null>(null)
   160→  const [tendered, setTendered] = useState('')
   161→  const [showSuccess, setShowSuccess] = useState(false)
   162→  const [lastTx, setLastTx] = useState<TxResult | null>(null)
   163→  const [showShiftDialog, setShowShiftDialog] = useState<'open' | 'close' | null>(null)
   164→  const [shiftOpeningCash, setShiftOpeningCash] = useState('')
   165→  const [shiftClosingCash, setShiftClosingCash] = useState('')
   166→  const [shiftNote, setShiftNote] = useState('')
   167→  const [activeTab, setActiveTab] = useState<'pos' | 'history'>('pos')
   168→  const [historyFilter, setHistoryFilter] = useState<'shift' | 'today' | 'all'>('shift')
   169→  const [showTxDetail, setShowTxDetail] = useState<TxHistoryItem | null>(null)
   170→  const [showMobileCart, setShowMobileCart] = useState(false)
   171→  const searchRef = useRef<HTMLInputElement>(null)
   172→
   173→  // ── Queries ──────────────────────────────────────────────────────
   174→
   175→  const { data: productsData, isLoading: productsLoading } = useQuery({
   176→    queryKey: ['cashier-products', user?.tenantId, search, activeCategory],
   177→    queryFn: async () => {
   178→      const params = new URLSearchParams({ tenantId: user!.tenantId! })
   179→      if (search) params.set('search', search)
   180→      if (activeCategory) params.set('categoryId', activeCategory)
   181→      const res = await fetch(`/api/cashier/products?${params}`)
   182→      if (!res.ok) throw new Error('Gagal memuat produk')
   183→      return res.json() as Promise<{ products: Product[]; categories: Category[] }>
   184→    },
   185→    enabled: !!user?.tenantId,
   186→    refetchInterval: 30000,
   187→  })
   188→
   189→  const { data: shiftData, isLoading: shiftLoading } = useQuery({
   190→    queryKey: ['cashier-shift', user?.outletId, user?.id],
   191→    queryFn: async () => {
   192→      const params = new URLSearchParams({
   193→        outletId: user!.outletId!,
   194→        cashierId: user!.id,
   195→      })
   196→      const res = await fetch(`/api/cashier/shifts?${params}`)
   197→      if (!res.ok) throw new Error('Gagal memuat shift')
   198→      return res.json() as Promise<{ shift: ShiftData | null }>
   199→    },
   200→    enabled: !!user?.outletId && !!user?.id,
   201→  })
   202→
   203→  const { data: historyData, isLoading: historyLoading } = useQuery({
   204→    queryKey: ['cashier-history', user?.id, historyFilter],
   205→    queryFn: async () => {
   206→      const params = new URLSearchParams({
   207→        cashierId: user!.id,
   208→        filter: historyFilter,
   209→      })
   210→      const res = await fetch(`/api/cashier/history?${params}`)
   211→      if (!res.ok) throw new Error('Gagal memuat riwayat')
   212→      return res.json() as Promise<{ transactions: TxHistoryItem[] }>
   213→    },
   214→    enabled: activeTab === 'history' && !!user?.id,
   215→  })
   216→
   217→  // ── Mutations ──────────────────────────────────────────────────
   218→
   219→  const createTransaction = useMutation({
   220→    mutationFn: async () => {
   221→      if (!user || !shiftData?.shift || !selectedPayment) return null
   222→
   223→      const subtotal = cart.reduce((sum, c) => sum + c.price * c.quantity, 0)
   224→      const discountAmount = 0
   225→      const taxAmount = Math.round(subtotal * TAX_RATE)
   226→      const serviceCharge = Math.round(subtotal * SERVICE_CHARGE_RATE)
   227→      const totalAmount = subtotal - discountAmount + taxAmount + serviceCharge
   228→
   229→      const res = await fetch('/api/cashier/transaction', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({
   233→          tenantId: user.tenantId,
   234→          outletId: user.outletId,
   235→          cashierId: user.id,
   236→          items: cart.map((c) => ({
   237→            productId: c.productId,
   238→            name: c.name,
   239→            price: c.price,
   240→            buyPrice: c.buyPrice,
   241→            quantity: c.quantity,
   242→            unit: c.unit,
   243→            sku: c.sku,
   244→          })),
   245→          subtotalAmount: subtotal,
   246→          discountAmount,
   247→          taxAmount,
   248→          serviceCharge,
   249→          totalAmount,
   250→          paymentMethod: selectedPayment.method,
   251→          paymentChannel: selectedPayment.channel,
   252→          tendered: selectedPayment.method === 'cash' ? parseFloat(tendered) || 0 : undefined,
   253→        }),
   254→      })
   255→      if (!res.ok) {
   256→        const err = await res.json()
   257→        throw new Error(err.error || 'Gagal membuat transaksi')
   258→      }
   259→      return res.json() as Promise<TxResult>
   260→    },
   261→    onSuccess: (data) => {
   262→      if (data) {
   263→        setLastTx(data)
   264→        setShowSuccess(true)
   265→        setCart([])
   266→        setSelectedPayment(null)
   267→        setTendered('')
   268→        queryClient.invalidateQueries({ queryKey: ['cashier-products'] })
   269→        queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   270→        queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   271→      }
   272→    },
   273→    onError: (err) => {
   274→      toast.error(err.message)
   275→    },
   276→  })
   277→
   278→  const shiftMutation = useMutation({
   279→    mutationFn: async (action: 'open' | 'close') => {
   280→      if (!user) return null
   281→      const body: Record<string, unknown> = {
   282→        action,
   283→        tenantId: user.tenantId,
   284→        outletId: user.outletId,
   285→        cashierId: user.id,
   286→      }
   287→      if (action === 'open') body.openingCash = parseFloat(shiftOpeningCash) || 0
   288→      if (action === 'close') {
   289→        body.closingCash = parseFloat(shiftClosingCash) || 0
   290→        body.note = shiftNote
   291→      }
   292→      const res = await fetch('/api/cashier/shifts', {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify(body),
   296→      })
   297→      if (!res.ok) {
   298→        const err = await res.json()
   299→        throw new Error(err.error || 'Gagal')
   300→      }
   301→      return res.json()
   302→    },
   303→    onSuccess: () => {
   304→      queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   305→      queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   306→      setShowShiftDialog(null)
   307→      setShiftOpeningCash('')
   308→      setShiftClosingCash('')
   309→      setShiftNote('')
   310→      toast.success(
   311→        shiftMutation.variables === 'open' ? 'Shift berhasil dibuka' : 'Shift berhasil ditutup',
   312→      )
   313→    },
   314→    onError: (err) => {
   315→      toast.error(err.message)
   316→    },
   317→  })
   318→
   319→  // ── Cart Logic ──────────────────────────────────────────────────
   320→
   321→  const addToCart = useCallback(
   322→    (product: Product) => {
   323→      setCart((prev) => {
   324→        const existing = prev.find((c) => c.productId === product.id)
   325→        if (existing) {
   326→          if (existing.quantity >= product.stock) {
   327→            toast.warning(`Stok ${product.name} tidak cukup`)
   328→            return prev
   329→          }
   330→          return prev.map((c) =>
   331→            c.productId === product.id
   332→              ? { ...c, quantity: c.quantity + 1 }
   333→              : c,
   334→          )
   335→        }
   336→        if (product.stock <= 0) {
   337→          toast.warning(`${product.name} habis`)
   338→          return prev
   339→        }
   340→        return [
   341→          ...prev,
   342→          {
   343→            productId: product.id,
   344→            name: product.name,
   345→            price: product.sellPrice,
   346→            buyPrice: product.buyPrice,
   347→            quantity: 1,
   348→            unit: product.unit,
   349→            sku: product.sku,
   350→          },
   351→        ]
   352→      })
   353→    },
   354→    [],
   355→  )
   356→
   357→  const updateQty = useCallback((productId: string, delta: number) => {
   358→    setCart((prev) => {
   359→      return prev
   360→        .map((c) => {
   361→          if (c.productId !== productId) return c
   362→          const newQty = c.quantity + delta
   363→          if (newQty <= 0) return null
   364→          return { ...c, quantity: newQty }
   365→        })
   366→        .filter(Boolean) as CartItem[]
   367→    })
   368→  }, [])
   369→
   370→  const removeFromCart = useCallback((productId: string) => {
   371→    setCart((prev) => prev.filter((c) => c.productId !== productId))
   372→  }, [])
   373→
   374→  const clearCart = useCallback(() => {
   375→    setCart([])
   376→    setSelectedPayment(null)
   377→    setTendered('')
   378→  }, [])
   379→
   380→  // ── Computed ──────────────────────────────────────────────────
   381→
   382→  const cartSubtotal = useMemo(() => cart.reduce((s, c) => s + c.price * c.quantity, 0), [cart])
   383→  const cartTax = Math.round(cartSubtotal * TAX_RATE)
   384→  const cartServiceCharge = Math.round(cartSubtotal * SERVICE_CHARGE_RATE)
   385→  const cartTotal = cartSubtotal + cartTax + cartServiceCharge
   386→  const cartItemCount = cart.reduce((s, c) => s + c.quantity, 0)
   387→  const tenderedNum = parseFloat(tendered) || 0
   388→  const change = tenderedNum - cartTotal
   389→
   390→  const filteredProducts = useMemo(() => {
   391→    const prods = productsData?.products || []
   392→    if (!activeCategory) return prods
   393→    return prods.filter((p) => p.category?.id === activeCategory)
   394→  }, [productsData, activeCategory])
   395→
   396→  const categories = productsData?.categories || []
   397→  const activeShift = shiftData?.shift || null
   398→  const isProcessing = createTransaction.isPending || shiftMutation.isPending
   399→
   400→  // ── Keyboard shortcuts ──────────────────────────────────────────
   401→
   402→  useEffect(() => {
   403→    const handleKey = (e: KeyboardEvent) => {
   404→      const tag = (e.target as HTMLElement)?.tagName
   405→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   406→      if (e.key === 'F2') {
   407→        e.preventDefault()
   408→        searchRef.current?.focus()
   409→      }
   410→      if (e.key === 'Escape') {
   411→        setShowSuccess(false)
   412→        setShowShiftDialog(null)
   413→        setShowTxDetail(null)
   414→      }
   415→    }
   416→    document.addEventListener('keydown', handleKey)
   417→    return () => document.removeEventListener('keydown', handleKey)
   418→  }, [])
   419→
   420→  // ── Print Receipt ──────────────────────────────────────────
   421→
   422→  const printReceipt = useCallback(() => {
   423→    if (!lastTx) return
   424→    const txSubtotal = lastTx.items.reduce((s, i) => s + i.price * i.quantity, 0)
   425→    const txTax = Math.round(txSubtotal * TAX_RATE)
   426→    const txServiceCharge = Math.round(txSubtotal * SERVICE_CHARGE_RATE)
   427→
   428→    const html = generateReceiptHtml({
   429→      invoiceNo: lastTx.invoiceNo,
   430→      tenantName: lastTx.tenantName,
   431→      outletName: lastTx.outletName,
   432→      outletAddress: lastTx.outletAddress,
   433→      outletCity: lastTx.outletCity,
   434→      tenantPhone: lastTx.tenantPhone,
   435→      cashierName: lastTx.cashierName,
   436→      items: lastTx.items.map((i) => ({
   437→        name: i.name,
   438→        quantity: i.quantity,
   439→        price: i.price,
   440→        subtotal: i.price * i.quantity,
   441→      })),
   442→      subtotalAmount: txSubtotal,
   443→      discountAmount: 0,
   444→      taxAmount: txTax,
   445→      serviceCharge: txServiceCharge,
   446→      totalAmount: lastTx.transaction.totalAmount,
   447→      paymentMethod: lastTx.payment.paymentMethod,
   448→      paymentChannel: lastTx.payment.paymentChannel,
   449→      tendered: lastTx.payment.paymentMethod === 'cash' ? parseFloat(tendered) || 0 : undefined,
   450→      change: lastTx.payment.paymentMethod === 'cash' ? change : undefined,
   451→      createdAt: lastTx.transaction.createdAt,
   452→    })
   453→
   454→    const w = window.open('', '_blank', 'width=400,height=700')
   455→    if (w) {
   456→      w.document.write(html)
   457→      w.document.close()
   458→      setTimeout(() => w.print(), 300)
   459→    }
   460→  }, [lastTx, tendered, change, cart])
   461→
   462→
   463→  // ── Render ──────────────────────────────────────────────────
   464→
   465→  return (
   466→    <div className="flex h-[calc(100vh-8.5rem)] flex-col gap-4 md:gap-6">
   467→      {/* Tab Switcher */}
   468→      <div className="flex items-center gap-1">
   469→        <button
   470→          onClick={() => setActiveTab('pos')}
   471→          className={cn(
   472→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   473→            activeTab === 'pos'
   474→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   475→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   476→          )}
   477→        >
   478→          <span className="flex items-center gap-2">
   479→            <ShoppingCart className="h-4 w-4" />
   480→            Kasir POS
   481→          </span>
   482→        </button>
   483→        <button
   484→          onClick={() => setActiveTab('history')}
   485→          className={cn(
   486→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   487→            activeTab === 'history'
   488→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   489→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   490→          )}
   491→        >
   492→          <span className="flex items-center gap-2">
   493→            <History className="h-4 w-4" />
   494→            Riwayat
   495→          </span>
   496→        </button>
   497→      </div>
   498→
   499→      {activeTab === 'pos' ? (
   500→        <div className="flex min-h-0 flex-1 flex-col gap-4 lg:flex-row">
   501→          {/* Mobile shift status bar */}
   502→          <div className="lg:hidden">
   503→            {shiftLoading ? (
   504→              <Skeleton className="h-10 w-full rounded-lg" />
   505→            ) : !activeShift ? (
   506→              <button
   507→                onClick={() => setShowShiftDialog('open')}
   508→                className="flex w-full items-center justify-between rounded-lg bg-amber-50 px-4 py-2.5 text-sm font-medium text-amber-800 transition-all hover:bg-amber-100 dark:bg-amber-950/50 dark:text-amber-300 dark:hover:bg-amber-950 btn-press"
   509→              >
   510→                <span className="flex items-center gap-2">
   511→                  <Clock className="h-4 w-4" />
   512→                  Shift belum dibuka
   513→                </span>
   514→                <Badge className="badge-gradient-amber text-white">Buka Shift</Badge>
   515→              </button>
   516→            ) : (
   517→              <div className="flex items-center justify-between rounded-lg bg-emerald-50 px-4 py-2.5 dark:bg-emerald-950/30">
   518→                <div className="flex items-center gap-2">
   519→                  <div className="relative">
   520→                    <div className="h-2.5 w-2.5 rounded-full bg-emerald-500" />
   521→                    <div className="absolute inset-0 animate-ping rounded-full bg-emerald-400 opacity-75" />
   522→                  </div>
   523→                  <span className="text-sm font-semibold text-emerald-700 dark:text-emerald-400">
   524→                    Shift Aktif
   525→                  </span>
   526→                  <span className="text-xs text-muted-foreground">
   527→                    • {formatTime(activeShift.openedAt)}
   528→                  </span>
   529→                </div>
   530→                <button
   531→                  onClick={() => setShowShiftDialog('close')}
   532→                  className="rounded-md bg-emerald-100 px-2 py-1 text-[10px] font-semibold text-emerald-700 transition-colors hover:bg-emerald-200 dark:bg-emerald-900/50 dark:text-emerald-300 dark:hover:bg-emerald-900 btn-press"
   533→                >
   534→                  Tutup
   535→                </button>
   536→              </div>
   537→            )}
   538→          </div>
   539→
   540→          {/* ─── LEFT PANEL: Products ─── */}
   541→          <div className="flex min-h-0 flex-1 flex-col gap-3 lg:flex-[3]">
   542→            {/* Search */}
   543→            <div className="relative">
   544→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   545→              <Input
   546→                ref={searchRef}
   547→                value={search}
   548→                onChange={(e) => setSearch(e.target.value)}
   549→                placeholder="Cari produk... (F2)"
   550→                className="pl-10"
   551→              />
   552→            </div>
   553→
