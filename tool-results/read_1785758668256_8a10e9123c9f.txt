   205→// ─── Component ─────────────────────────────────────────────────────────────
   206→
   207→export default function Merchants() {
   208→  const [page, setPage] = useState(1);
   209→  const [search, setSearch] = useState('');
   210→  const [searchInput, setSearchInput] = useState('');
   211→  const [selectedMerchantId, setSelectedMerchantId] = useState<string | null>(null);
   212→  const [sheetOpen, setSheetOpen] = useState(false);
   213→  const [copiedId, setCopiedId] = useState<string | null>(null);
   214→  const [showApiKey, setShowApiKey] = useState(false);
   215→  const [createOpen, setCreateOpen] = useState(false);
   216→  const [createForm, setCreateForm] = useState({
   217→    name: '',
   218→    email: '',
   219→    phone: '',
   220→    address: '',
   221→  });
   222→
   223→  // ── Fetch paginated merchants ──
   224→  const { data, isLoading, isError } = useQuery({
   225→    queryKey: ['merchants', page, search],
   226→    queryFn: async () => {
   227→      const params = new URLSearchParams({
   228→        page: String(page),
   229→        limit: '9',
   230→      });
   231→      if (search) params.set('search', search);
   232→      const res = await fetch(`/api/merchants?${params}`);
   233→      if (!res.ok) throw new Error('Gagal memuat data merchant');
   234→      const json = await res.json();
   235→      return json as {
   236→        data: Merchant[];
   237→        pagination: { page: number; limit: number; total: number; totalPages: number };
   238→      };
   239→    },
   240→    staleTime: 15_000,
   241→  });
   242→
   243→  // ── Fetch stats (all merchants) ──
   244→  const { data: allMerchants } = useQuery({
   245→    queryKey: ['merchants-stats'],
   246→    queryFn: async () => {
   247→      const res = await fetch('/api/merchants?limit=9999');
   248→      if (!res.ok) throw new Error('Gagal memuat statistik');
   249→      const json = await res.json();
   250→      return json.data as Merchant[];
   251→    },
   252→    staleTime: 30_000,
   253→  });
   254→
   255→  // ── Fetch merchant detail ──
   256→  const { data: detailData, isLoading: detailLoading } = useQuery({
   257→    queryKey: ['merchant-detail', selectedMerchantId],
   258→    queryFn: async () => {
   259→      const res = await fetch(`/api/merchants/${selectedMerchantId}`);
   260→      if (!res.ok) throw new Error('Gagal memuat detail merchant');
   261→      const json = await res.json();
   262→      return json.data as MerchantDetail;
   263→    },
   264→    enabled: !!selectedMerchantId && sheetOpen,
   265→    staleTime: 10_000,
   266→  });
   267→
   268→  const detail = detailData ?? null;
   269→
   270→  const stats = useMemo(() => {
   271→    if (!allMerchants) return { total: 0, active: 0, revenue: 0 };
   272→    const total = allMerchants.length;
   273→    const active = allMerchants.filter((m) => m.status === 'active').length;
   274→    const revenue = allMerchants.reduce((s, m) => s + m.totalSales, 0);
   275→    return { total, active, revenue };
   276→  }, [allMerchants]);
   277→
   278→  const merchants = data?.data ?? [];
   279→  const pagination = data?.pagination;
   280→
   281→  // ── Create Merchant Mutation ──
   282→  const queryClient = useQueryClient();
   283→
   284→  const createMutation = useMutation({
   285→    mutationFn: (body: { name: string; email: string; phone?: string; address?: string }) =>
   286→      fetch('/api/merchants', {
   287→        method: 'POST',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify(body),
   290→      }).then((r) => {
   291→        if (!r.ok) return r.json().then((e) => Promise.reject(e.error));
   292→        return r.json();
   293→      }),
   294→    onSuccess: () => {
   295→      queryClient.invalidateQueries({ queryKey: ['merchants'] });
   296→      queryClient.invalidateQueries({ queryKey: ['merchants-stats'] });
   297→      setCreateOpen(false);
   298→      setCreateForm({ name: '', email: '', phone: '', address: '' });
   299→      toast.success('Merchant berhasil ditambahkan');
   300→    },
   301→    onError: (err: string) => {
   302→      toast.error(err || 'Gagal menambahkan merchant');
   303→    },
   304→  });
   305→
   306→  const handleSubmitCreate = useCallback(() => {
   307→    if (!createForm.name.trim()) {
   308→      toast.error('Nama merchant wajib diisi');
   309→      return;
   310→    }
   311→    if (!createForm.email.trim() || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(createForm.email)) {
   312→      toast.error('Format email tidak valid');
   313→      return;
   314→    }
   315→    createMutation.mutate({
   316→      name: createForm.name.trim(),
   317→      email: createForm.email.trim(),
   318→      phone: createForm.phone.trim() || undefined,
   319→      address: createForm.address.trim() || undefined,
   320→    });
   321→  }, [createForm, createMutation]);
   322→
   323→  // ── Handlers ──
   324→  const handleSearch = (e: React.FormEvent) => {
   325→    e.preventDefault();
   326→    setSearch(searchInput);
   327→    setPage(1);
   328→  };
   329→
   330→  const handleClearSearch = () => {
   331→    setSearchInput('');
   332→    setSearch('');
   333→    setPage(1);
   334→  };
   335→
   336→  const handleCopyKey = (id: string, key: string) => {
   337→    navigator.clipboard.writeText(key);
   338→    setCopiedId(id);
   339→    toast.success('API Key berhasil disalin');
   340→    setTimeout(() => setCopiedId(null), 2000);
   341→  };
   342→
   343→  const handleOpenDetail = useCallback((id: string) => {
   344→    setSelectedMerchantId(id);
   345→    setShowApiKey(false);
   346→    setSheetOpen(true);
   347→  }, []);
   348→
   349→  const handleCloseSheet = useCallback(() => {
   350→    setSheetOpen(false);
   351→    setSelectedMerchantId(null);
   352→    setShowApiKey(false);
   353→  }, []);
   354→
   355→  // ── Page range for pagination ──
   356→  const pageRange = useMemo(() => {
   357→    if (!pagination) return [];
   358→    const { page: p, totalPages: tp } = pagination;
   359→    const pages: (number | 'ellipsis')[] = [];
   360→    if (tp <= 7) {
   361→      for (let i = 1; i <= tp; i++) pages.push(i);
   362→    } else {
   363→      pages.push(1);
   364→      if (p > 3) pages.push('ellipsis');
   365→      const start = Math.max(2, p - 1);
   366→      const end = Math.min(tp - 1, p + 1);
   367→      for (let i = start; i <= end; i++) pages.push(i);
   368→      if (p < tp - 2) pages.push('ellipsis');
   369→      pages.push(tp);
   370→    }
   371→    return pages;
   372→  }, [pagination]);
   373→
   374→  // ─── Render ───────────────────────────────────────────────────────────
   375→
   376→  if (isError) {
   377→    return (
   378→      <div className="flex flex-col items-center justify-center py-20">
   379→        <Store className="h-12 w-12 text-muted-foreground mb-4" />
   380→        <p className="text-muted-foreground">Gagal memuat data merchant</p>
   381→        <Button variant="outline" className="mt-4" onClick={() => window.location.reload()}>
   382→          Coba Lagi
   383→        </Button>
   384→      </div>
   385→    );
   386→  }
   387→
   388→  return (
   389→    <div className="space-y-6 bg-mesh">
   390→      <PageHeader
   391→        title="Merchant"
   392→        description="Kelola merchant dan konfigurasi pembayaran"
   393→        icon={Store}
   394→        iconGradient="amber"
   395→      >
   396→        <Button
   397→          onClick={() => setCreateOpen(true)}
   398→          className="bg-emerald-600 hover:bg-emerald-700 text-white btn-press"
   399→        >
   400→          <Plus className="h-4 w-4 mr-2" />
   401→          <span className="hidden sm:inline">Tambah Merchant</span>
   402→        </Button>
   403→      </PageHeader>
   404→
   405→      {/* ── Stats Row ── */}
   406→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   407→        {[
   408→          { label: 'Total Merchant', value: stats.total.toLocaleString('id-ID'), icon: Store, gradient: 'from-emerald-500 to-teal-500' },
   409→          { label: 'Merchant Aktif', value: stats.active.toLocaleString('id-ID'), icon: ArrowUpRight, gradient: 'from-emerald-600 to-green-500' },
   410→          { label: 'Total Pendapatan Merchant', value: formatMoney(stats.revenue), icon: Banknote, gradient: 'from-teal-500 to-emerald-500' },
   411→        ].map((stat, i) => (
   412→          <motion.div
   413→            key={stat.label}
   414→            initial={{ opacity: 0, y: 20 }}
   415→            animate={{ opacity: 1, y: 0 }}
   416→            transition={{ duration: 0.4, delay: i * 0.08 }}
   417→          >
   418→            <Card className="shadow-sm hover:shadow-md transition-shadow card-hover">
   419→              <CardContent className="p-4 sm:p-6 flex items-center gap-4">
   420→                <div className={`flex-shrink-0 h-11 w-11 rounded-xl bg-gradient-to-br ${stat.gradient} flex items-center justify-center`}>{
   421→                  <stat.icon className="h-5 w-5 text-white" />
   422→                }</div>
   423→                <div className="min-w-0">
   424→                  <p className="text-sm text-muted-foreground truncate">{stat.label}</p>
   425→                  <p className="text-lg font-bold truncate">{stat.value}</p>
   426→                </div>
   427→              </CardContent>
   428→            </Card>
   429→          </motion.div>
   430→        ))}
   431→      </div>
   432→
   433→      {/* ── Search + Add ── */}
   434→      <motion.div
   435→        initial={{ opacity: 0, y: 12 }}
   436→        animate={{ opacity: 1, y: 0 }}
   437→        transition={{ duration: 0.35, delay: 0.2 }}
   438→        className="flex flex-col sm:flex-row gap-3"
   439→      >
   440→        <form onSubmit={handleSearch} className="flex flex-1 gap-3">
   441→          <div className="relative flex-1">
   442→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   443→            <Input
   444→              placeholder="Cari nama atau email merchant..."
   445→              value={searchInput}
   446→              onChange={(e) => setSearchInput(e.target.value)}
   447→              className="pl-9 pr-9"
   448→            />
   449→            {search && (
   450→              <button
   451→                type="button"
   452→                onClick={handleClearSearch}
   453→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   454→              >
   455→                <X className="h-4 w-4" />
   456→              </button>
   457→            )}
   458→          </div>
   459→          <Button type="submit" className="bg-emerald-600 hover:bg-emerald-700 text-white btn-press">
   460→            <Search className="h-4 w-4 mr-2" />
   461→            Cari
   462→          </Button>
   463→        </form>
   464→
   465→      </motion.div>
   466→
   467→      {/* ── Active search indicator ── */}
   468→      <AnimatePresence>
   469→        {search && (
   470→          <motion.div
   471→            initial={{ opacity: 0, height: 0 }}
   472→            animate={{ opacity: 1, height: 'auto' }}
   473→            exit={{ opacity: 0, height: 0 }}
   474→            className="flex items-center gap-2"
   475→          >
   476→            <span className="text-sm text-muted-foreground">Filter:</span>
   477→            <Badge variant="secondary" className="gap-1 pr-1">
   478→              &ldquo;{search}&rdquo;
   479→              <button onClick={handleClearSearch} className="ml-1 rounded-full p-0.5 hover:bg-muted">
   480→                <X className="h-3 w-3" />
   481→              </button>
   482→            </Badge>
   483→          </motion.div>
   484→        )}
   485→      </AnimatePresence>
   486→
   487→      {/* ── Cards Grid ── */}
   488→      {isLoading ? (
   489→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   490→          {Array.from({ length: 6 }).map((_, i) => (
   491→            <Card key={i} className="overflow-hidden">
   492→              <Skeleton className="h-1.5 w-full" />
   493→              <CardContent className="p-5 space-y-3">
   494→                <Skeleton className="h-5 w-3/4" />
   495→                <Skeleton className="h-4 w-1/2" />
   496→                <Skeleton className="h-4 w-2/3" />
   497→                <Skeleton className="h-4 w-full" />
   498→                <div className="flex gap-4 pt-2">
   499→                  <Skeleton className="h-4 w-20" />
   500→                  <Skeleton className="h-4 w-20" />
   501→                </div>
   502→              </CardContent>
   503→            </Card>
   504→          ))}
   505→        </div>
   506→      ) : merchants.length === 0 ? (
   507→        <EmptyState
   508→          title="Tidak ada merchant ditemukan"
   509→          description="Coba ubah kata kunci pencarian"
   510→          icon="merchant"
   511→          actionLabel={(!search) ? 'Tambah' : undefined}
   512→          onAction={(!search) ? () => setCreateOpen(true) : undefined}
   513→        />
   514→      ) : (
   515→        <motion.div
   516→          layout
   517→          className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4"
   518→        >
   519→          <AnimatePresence mode="popLayout">
   520→            {merchants.map((m, i) => (
   521→              <motion.div
   522→                key={m.id}
   523→                layout
   524→                initial={{ opacity: 0, y: 16 }}
   525→                animate={{ opacity: 1, y: 0 }}
   526→                exit={{ opacity: 0, y: -8 }}
   527→                transition={{ duration: 0.3, delay: i * 0.04 }}
   528→              >
   529→                <Card
   530→                  className="overflow-hidden shadow-sm hover:shadow-md hover:-translate-y-1 transition-all cursor-pointer group card-hover"
   531→                  onClick={() => handleOpenDetail(m.id)}
   532→                >
   533→                  {/* Gradient top border */}
   534→                  <div className={`h-1.5 w-full bg-gradient-to-r ${m.status === 'active' ? 'from-emerald-400 via-emerald-500 to-teal-400' : 'from-gray-300 via-gray-400 to-gray-300'}`} />
   535→                  <CardContent className="p-5 space-y-3.5">
   536→                    {/* Name + Status */}
   537→                    <div className="flex items-start justify-between gap-2">
   538→                      <div className="min-w-0">
   539→                        <h3
   540→                          className="font-semibold text-sm truncate text-emerald-600 dark:text-emerald-400 hover:underline cursor-pointer"
   541→                          onClick={(e) => {
   542→                            e.stopPropagation();
   543→                            handleOpenDetail(m.id);
   544→                          }}
   545→                        >
   546→                          {m.name}
   547→                        </h3>
   548→                        <div className="flex items-center gap-1 mt-1 text-xs text-muted-foreground">
   549→                          <Mail className="h-3 w-3 flex-shrink-0" />
   550→                          <span className="truncate">{m.email}</span>
   551→                        </div>
   552→                      </div>
   553→                      <Badge className={`text-[10px] px-2 py-0.5 flex-shrink-0 ${statusColor[m.status] || ''}`}>
   554→                        {statusLabel[m.status] || m.status}
   555→                      </Badge>
   556→                    </div>
   557→
   558→                    {/* Phone */}
   559→                    {m.phone && (
