   302→   302→    },
   303→   303→    onSuccess: () => {

   331→   331→            c.productId === product.id
   332→   332→              ? { ...c, quantity: c.quantity + 1 }
   333→   333→              : c,
   334→   334→          )
   335→   335→        }
   336→   336→        if (product.stock <= 0) {
   337→   337→          toast.warning(`${product.name} habis`)
   338→   338→          return prev
   339→   339→        }

   342→   342→          {

   426→   426→    const txServiceCharge = Math.round(txSubtotal * SERVICE_CHARGE_RATE)
   427→   427→
   428→   428→    const html = generateReceiptHtml({
   429→   429→      invoiceNo: lastTx.invoiceNo,
   430→   430→      tenantName: lastTx.tenantName,
   431→   431→      outletName: lastTx.outletName,

   436→   436→      items: lastTx.items.map((i) => ({
   437→   437→        name: i.name,
   438→   438→        quantity: i.quantity,
   439→   439→        price: i.price,
   440→   440→        subtotal: i.price * i.quantity,

   442→   442→      subtotalAmount: txSubtotal,

   447→   447→      paymentMethod: lastTx.payment.paymentMethod,
   448→   448→      paymentChannel: lastTx.payment.paymentChannel,

   451→   451→      createdAt: lastTx.transaction.createdAt,
   452→   452→    })
   453→   453→
   454→   454→    const w = window.open('', '_blank', 'width=400,height=700')

   456→   456→      w.document.write(html)
   457→   457→      w.document.close()

   459→   459→    }
   460→   460→  }, [lastTx, tendered, change, cart])
   461→   461→
   462→   462→
   463→   463→  // ── Render ──────────────────────────────────────────────────
   464→   464→
   465→   465→  return (
