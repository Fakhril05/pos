   237→// ── Pagination Helper ────────────────────────────────────────────────────────
   238→
   239→function getPaginationRange(current: number, total: number) {
   240→  const delta = 2;
   241→  const range: (number | 'dots')[] = [];
   242→  const left = Math.max(2, current - delta);
   243→  const right = Math.min(total - 1, current + delta);
   244→
   245→  range.push(1);
   246→  if (left > 2) range.push('dots');
   247→  for (let i = left; i <= right; i++) range.push(i);
   248→  if (right < total - 1) range.push('dots');
   249→  if (total > 1) range.push(total);
   250→
   251→  return range;
   252→}
