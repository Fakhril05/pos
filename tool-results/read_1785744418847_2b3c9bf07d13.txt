   153→// ── Page range helper ────────────────────────────────────────────────────────
   154→
   155→function getPageRange(current: number, total: number) {
   156→  const pages: (number | 'ellipsis')[] = [];
   157→  if (total <= 7) {
   158→    for (let i = 1; i <= total; i++) pages.push(i);
   159→  } else {
   160→    pages.push(1);
   161→    if (current > 3) pages.push('ellipsis');
   162→    const start = Math.max(2, current - 1);
   163→    const end = Math.min(total - 1, current + 1);
   164→    for (let i = start; i <= end; i++) pages.push(i);
   165→    if (current < total - 2) pages.push('ellipsis');
   166→    pages.push(total);
   167→  }
   168→  return pages;
   169→}
