    54→    54→const formatRp = (amount: number) =>
    55→    55→  new Intl.NumberFormat('id-ID', {
    56→    56→    style: 'currency',
    57→    57→    currency: 'IDR',
    58→    58→    minimumFractionDigits: 0,
    59→    59→  }).format(amount);
    60→    60→
    61→    61→const formatTimeAgo = (dateStr: string): string => {
    62→    62→  const now = Date.now();
    63→    63→  const then = new Date(dateStr).getTime();
    64→    64→  const diffMs = now - then;
    65→    65→  const seconds = Math.floor(diffMs / 1000);
    66→    66→  const minutes = Math.floor(seconds / 60);
    67→    67→  const hours = Math.floor(minutes / 60);
    68→    68→  const days = Math.floor(hours / 24);
    69→    69→  if (days > 0) return `${days} hari lalu`;
    70→    70→  if (hours > 0) return `${hours} jam lalu`;
    71→    71→  if (minutes > 0) return `${minutes} menit lalu`;
    72→    72→  return 'Baru saja';
    73→    73→};
    74→    74→
    75→    75→const formatCompact = (amount: number) => {
    76→    76→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    77→    77→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    78→    78→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    79→    79→  return String(amount);
    80→    80→};
