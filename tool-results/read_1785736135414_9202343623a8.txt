   152→// ── Page range helper ────────────────────────────────────────────────────────
   153→
   154→function getPageRange(current: number, total: number) {
   155→  const pages: (number | 'ellipsis')[] = [];
   156→  if (total <= 7) {
   157→    for (let i = 1; i <= total; i++) pages.push(i);
   158→  } else {
   159→    pages.push(1);
   160→    if (current > 3) pages.push('ellipsis');
   161→    const start = Math.max(2, current - 1);
   162→    const end = Math.min(total - 1, current + 1);
   163→    for (let i = start; i <= end; i++) pages.push(i);
   164→    if (current < total - 2) pages.push('ellipsis');
   165→    pages.push(total);
   166→  }
   167→  return pages;
   168→}
