   203→function buildTimeline(tx: Transaction): TimelineEvent[] {
   204→  const events: TimelineEvent[] = [
   205→    {
   206→      label: 'Transaksi dibuat',
   207→      date: tx.createdAt,
   208→      icon: Receipt,
   209→      color: 'text-emerald-500',
   210→    },
   211→  ];
   212→  if (tx.paidAt) {
   213→    events.push({
   214→      label: 'Pembayaran diterima',
   215→      date: tx.paidAt,
   216→      icon: CheckCircle2,
   217→      color: 'text-emerald-500',
   218→    });
   219→  }
   220→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   221→    events.push({
   222→      label: 'Transaksi kedaluwarsa',
   223→      date: tx.expiredAt,
   224→      icon: AlertTriangle,
   225→      color: 'text-gray-500',
   226→    });
   227→  }
   228→  if (tx.status === 'failed') {
   229→    events.push({
   230→      label: 'Transaksi gagal',
   231→      date: tx.updatedAt,
   232→      icon: XCircle,
   233→      color: 'text-red-500',
   234→    });
   235→  }
   236→  if (tx.status === 'refunded') {
   237→    events.push({
   238→      label: 'Refund diproses',
   239→      date: tx.updatedAt,
   240→      icon: RotateCcw,
   241→      color: 'text-blue-500',
   242→    });
   243→  }
   244→  return events;
   245→}

   282→
   283→  const handleRefund = async () => {
   284→    if (!tx) return;
   285→    setRefunding(true);
   286→    try {
   287→      const res = await fetch(`/api/transactions/${tx.id}`, {
   288→        method: 'POST',
   289→        headers: { 'Content-Type': 'application/json' },
   290→        body: JSON.stringify({ amount: tx.amount, reason: 'Permintaan refund dari dashboard' }),
   291→      });

   293→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   294→        throw new Error(err.error || 'Gagal memproses refund');
   295→      }
   296→      toast.success('Refund berhasil diajukan');
   297→      queryClient.invalidateQueries({ queryKey: ['transactions'] });

   299→    } catch (e) {
   300→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');

   311→  };
   312→

   332→      formattedCreatedAt: formatDate(tx.createdAt),
   333→      formattedPaidAt: tx.paidAt ? formatDate(tx.paidAt) : null,
   334→      statusLabel: statusInfo.label,
   335→    });

   338→    const a = document.createElement('a');
   339→    a.href = url;
   340→    a.download = `struk-${tx.invoiceNo}.html`;
   341→    document.body.appendChild(a);
   342→    a.click();
   343→    document.body.removeChild(a);
   344→    URL.revokeObjectURL(url);

   587→// ── Pagination Helper ────────────────────────────────────────────────────────
   588→
   589→function getPaginationRange(current: number, total: number) {
   590→  const delta = 2;
   591→  const range: (number | 'dots')[] = [];
   592→  const left = Math.max(2, current - delta);
   593→  const right = Math.min(total - 1, current + delta);
   594→
   595→  range.push(1);
   596→  if (left > 2) range.push('dots');
   597→  for (let i = left; i <= right; i++) range.push(i);
   598→  if (right < total - 1) range.push('dots');
   599→  if (total > 1) range.push(total);
   600→
   601→  return range;
   602→}
