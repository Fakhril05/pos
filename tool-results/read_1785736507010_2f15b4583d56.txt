    58→// ── Helpers ──────────────────────────────────────────────────────────────────
    59→
    60→const formatRp = (amount: number) =>
    61→  new Intl.NumberFormat('id-ID', {
    62→    style: 'currency',
    63→    currency: 'IDR',
    64→    minimumFractionDigits: 0,
    65→  }).format(amount);
    66→
    67→const formatTimeAgo = (dateStr: string): string => {
    68→  const now = Date.now();
    69→  const then = new Date(dateStr).getTime();
    70→  const diffMs = now - then;
    71→  const seconds = Math.floor(diffMs / 1000);
    72→  const minutes = Math.floor(seconds / 60);
    73→  const hours = Math.floor(minutes / 60);
    74→  const days = Math.floor(hours / 24);
    75→  if (days > 0) return `${days} hari lalu`;
    76→  if (hours > 0) return `${hours} jam lalu`;
    77→  if (minutes > 0) return `${minutes} menit lalu`;
    78→  return 'Baru saja';
    79→};
    80→
    81→const formatCompact = (amount: number) => {
    82→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    83→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    84→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    85→  return String(amount);
    86→};

   319→const statusDotColors: Record<string, string> = {
   320→  success: 'bg-emerald-500',
   321→  pending: 'bg-amber-500',
   322→  failed: 'bg-red-500',

   326→
   327→function ActivityFeed() {
   328→  const { setActivePage } = useAppStore();
   329→  const [liveItems, setLiveItems] = useState<SocketTransaction[]>([]);
   330→
   331→  const { data: activityData, isLoading: activityLoading } = useQuery<{
   332→    data: ActivityTransaction[];
   333→  }>({
   334→    queryKey: ['activity-feed'],
   335→    queryFn: () =>
   336→      fetch('/api/transactions?limit=8&status=success').then((r) => r.json()),
   337→    staleTime: 15_000,
   338→    refetchInterval: 30_000,
   339→  });
   340→
   341→  const baseItems = activityData?.data?.slice(0, 6) ?? [];
   342→
   343→  // Socket integration — receive live transactions
   344→  useTransactionSocket({
   345→    onTransaction: (tx) => {
   346→      setLiveItems((prev) => [tx, ...prev].slice(0, 10));
   347→      toast('Pembayaran Baru!', {
   348→        description: `${formatRp(tx.amount)} dari ${tx.customerName} — ${tx.merchantName}`,
   349→        icon: <CreditCard className="h-4 w-4 text-emerald-500" />,
   350→      });
   351→    },
   352→  });
   353→
   354→  // Merge: live items first, then DB items, deduplicated
   355→  const liveIds = new Set(liveItems.map((t) => t.id));
   356→  const merged: (ActivityTransaction | SocketTransaction)[] = [
   357→    ...liveItems,
   358→    ...baseItems.filter((t) => !liveIds.has(t.id)),
   359→  ].slice(0, 6);
   360→
   361→  return (
   362→    <motion.div
   363→      initial={{ opacity: 0, y: 20 }}
   364→      animate={{ opacity: 1, y: 0 }}
   365→      transition={{ duration: 0.5, delay: 0.8 }}
   366→    >
   367→      <Card className="card-hover shadow-sm">
   368→        <CardHeader>
   369→          <div className="flex items-center justify-between">
   370→            <div className="flex items-center gap-2">
   371→              <Activity className="h-4 w-4 text-emerald-500" />
   372→              <CardTitle className="text-base">Aktivitas Terkini</CardTitle>
   373→            </div>
   374→            <span className="relative flex h-2.5 w-2.5">
   375→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   376→              <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-emerald-500" />
   377→            </span>
   378→          </div>
   379→          <p className="text-muted-foreground text-sm">
   380→            Transaksi terbaru yang berhasil diproses
   381→          </p>
   382→        </CardHeader>
   383→        <CardContent className="scrollbar-thin">
   384→          {activityLoading ? (
   385→            <div className="space-y-4">
   386→              {Array.from({ length: 4 }).map((_, i) => (
   387→                <div key={i} className="flex gap-3">
   388→                  <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   389→                  <div className="flex-1 space-y-2">
   390→                    <Skeleton className="h-4 w-3/4" />
   391→                    <Skeleton className="h-3 w-1/2" />
   392→                  </div>
   393→                </div>
   394→              ))}
   395→            </div>
   396→          ) : merged.length === 0 ? (
   397→            <p className="py-8 text-center text-sm text-muted-foreground">
   398→              Belum ada aktivitas terkini.
   399→            </p>
   400→          ) : (
   401→            <div className="relative space-y-0">
   402→              {/* Vertical connecting line */}
   403→              <div className="absolute bottom-0 left-[11px] top-2 w-px bg-border" />
   404→
   405→              {merged.map((tx, idx) => {
   406→                // Handle both ActivityTransaction (nested) and SocketTransaction (flat)
   407→                const merchantName =
   408→                  'merchant' in tx && tx.merchant
   409→                    ? (tx as ActivityTransaction).merchant.name
   410→                    : (tx as SocketTransaction).merchantName;
   411→                const customerName =
   412→                  'customer' in tx && tx.customer
   413→                    ? (tx as ActivityTransaction).customer!.name
   414→                    : (tx as SocketTransaction).customerName;
   415→
   416→                return (
   417→                <motion.div
   418→                  key={tx.id}
   419→                  initial={{ opacity: 0, x: -10 }}
   420→                  animate={{ opacity: 1, x: 0 }}
   421→                  transition={{ duration: 0.3, delay: idx * 0.07 }}
   422→                  className={cn(
   423→                    'group relative flex gap-4 rounded-lg p-2.5 transition-colors hover:bg-muted/50',
   424→                    idx < liveItems.length && 'bg-emerald-50/50 dark:bg-emerald-950/20'
   425→                  )}
   426→                >
   427→                  {/* Timeline dot */}
   428→                  <div className="relative z-10 mt-1 flex shrink-0 items-start justify-center">
   429→                    <div
   430→                      className={cn(
   431→                        'h-3 w-3 rounded-full ring-4 ring-background',
   432→                        statusDotColors[tx.status] ?? 'bg-gray-400'
   433→                      )}
   434→                    />
   435→                  </div>
   436→
   437→                  {/* Content */}
   438→                  <div className="min-w-0 flex-1 pb-3">
   439→                    <div className="flex flex-col gap-0.5 sm:flex-row sm:items-baseline sm:justify-between">
   440→                      <p className="text-sm">
   441→                        <span className="font-semibold">{merchantName}</span>
   442→                        <span className="text-muted-foreground"> — </span>
   443→                        <span className="text-muted-foreground">
   444→                          {customerName.split(' ').slice(0, 2).join(' ')}
   445→                        </span>
   446→                      </p>
   447→                      <span className="text-xs font-semibold text-emerald-600 dark:text-emerald-400">
   448→                        +{formatRp(tx.amount)}

   457→              })}
   458→            </div>
   459→          )}
   460→
   461→          {/* View all link */}
   462→          <div className="mt-2 border-t pt-3">
   463→            <button
   464→              onClick={() => setActivePage('transactions')}
   465→              className="text-sm font-medium text-emerald-600 transition-colors hover:text-emerald-700 dark:text-emerald-400 dark:hover:text-emerald-300"
   466→            >
   467→              Lihat Semua →
   468→            </button>
   469→          </div>
   470→        </CardContent>
   471→      </Card>
   472→    </motion.div>
   473→  );
   474→}
