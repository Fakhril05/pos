   194→   194→// ── Pagination Helper ────────────────────────────────────────────────────────
   195→   195→
   196→   196→function getPaginationRange(current: number, total: number) {
   197→   197→  const delta = 2;
   198→   198→  const range: (number | 'dots')[] = [];
   199→   199→  const left = Math.max(2, current - delta);
   200→   200→  const right = Math.min(total - 1, current + delta);
   201→   201→
   202→   202→  range.push(1);
   203→   203→  if (left > 2) range.push('dots');
   204→   204→  for (let i = left; i <= right; i++) range.push(i);
   205→   205→  if (right < total - 1) range.push('dots');
   206→   206→  if (total > 1) range.push(total);
   207→   207→
   208→   208→  return range;
   209→   209→}
