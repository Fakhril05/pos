   195→// ── Timeline Item ────────────────────────────────────────────────────────────
   196→
   197→interface TimelineEvent {
   198→  label: string;
   199→  date: string;
   200→  icon: React.ElementType;
   201→  color: string;
   202→}
   203→
   204→function buildTimeline(tx: Transaction): TimelineEvent[] {
   205→  const events: TimelineEvent[] = [
   206→    {
   207→      label: 'Transaksi dibuat',
   208→      date: tx.createdAt,
   209→      icon: Receipt,
   210→      color: 'text-emerald-500',
   211→    },
   212→  ];
   213→  if (tx.paidAt) {
   214→    events.push({
   215→      label: 'Pembayaran diterima',
   216→      date: tx.paidAt,
   217→      icon: CheckCircle2,
   218→      color: 'text-emerald-500',
   219→    });
   220→  }
   221→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   222→    events.push({
   223→      label: 'Transaksi kedaluwarsa',
   224→      date: tx.expiredAt,
   225→      icon: AlertTriangle,
   226→      color: 'text-gray-500',
   227→    });
   228→  }
   229→  if (tx.status === 'failed') {
   230→    events.push({
   231→      label: 'Transaksi gagal',
   232→      date: tx.updatedAt,
   233→      icon: XCircle,
   234→      color: 'text-red-500',
   235→    });
   236→  }
   237→  if (tx.status === 'refunded') {
   238→    events.push({
   239→      label: 'Refund diproses',
   240→      date: tx.updatedAt,
   241→      icon: RotateCcw,
   242→      color: 'text-blue-500',
   243→    });
   244→  }
   245→  return events;
   246→}
   247→
   248→function TimelineItem({ event, isLast }: { event: TimelineEvent; isLast: boolean }) {
   249→  const Icon = event.icon;
   250→  return (
   251→    <div className="flex gap-3">
   252→      <div className="flex flex-col items-center">
   253→        <div
   254→          className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-muted ${event.color}`}
   255→        >
   256→          <Icon className="h-4 w-4" />
   257→        </div>
   258→        {!isLast && <div className="w-px flex-1 bg-border" />}
   259→      </div>
   260→      <div className={`pb-6 ${isLast ? 'pb-0' : ''}`}>
   261→        <p className="text-sm font-medium">{event.label}</p>
   262→        <p className="text-muted-foreground text-xs">{formatDate(event.date)}</p>
   263→      </div>
   264→    </div>
   265→  );
   266→}

   283→
   284→  const timeline = useMemo(() => (tx ? buildTimeline(tx) : []), [tx]);
   285→
   286→  const handleRefund = async () => {
   287→    if (!tx) return;
   288→    if (!refundReason.trim()) {
   289→      setRefundReasonError('Alasan refund wajib diisi');
   290→      return;
   291→    }
   292→    setRefunding(true);
   293→    try {
   294→      const res = await fetch(`/api/transactions/${tx.id}`, {
   295→        method: 'POST',
   296→        headers: { 'Content-Type': 'application/json' },
   297→        body: JSON.stringify({ amount: tx.amount, reason: refundReason.trim() }),
   298→      });
   299→      if (!res.ok) {
   300→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   301→        throw new Error(err.error || 'Gagal memproses refund');
   302→      }
   303→      toast.success('Refund berhasil diajukan');
   304→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   305→      setRefundDialogOpen(false);
   306→      setRefundReason('');
   307→      onOpenChange(false);
   308→    } catch (e) {
   309→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');
   310→    } finally {
   311→      setRefunding(false);
   312→    }
   313→  };
   314→
   315→  const handleCopyInvoice = () => {
   316→    if (tx) {
   317→      navigator.clipboard.writeText(tx.invoiceNo);
   318→      toast.success('Nomor invoice disalin');
   319→    }
   320→  };
   321→
   322→  if (!tx) return null;
   323→
   324→  const statusInfo = STATUS_MAP[tx.status] ?? STATUS_MAP.pending;
   325→  const StatusIcon = statusInfo.icon;
   326→
   327→  return (
   328→    <Dialog open={open} onOpenChange={onOpenChange}>
   329→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   330→        <DialogHeader>
   331→          <div className="flex items-center gap-3">
   332→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950/50">
   333→              <Receipt className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   334→            </div>
   335→            <div>
   336→              <DialogTitle className="text-base">Detail Transaksi</DialogTitle>
   337→              <DialogDescription>
   338→                <span className="flex items-center gap-1.5 font-mono text-xs">
   339→                  {tx.invoiceNo}
   340→                  <button
   341→                    onClick={handleCopyInvoice}
   342→                    className="inline-flex hover:text-foreground transition-colors"
   343→                  >
   344→                    <Copy className="h-3 w-3" />
   345→                  </button>
   346→                </span>
   347→              </DialogDescription>
   348→            </div>
   349→          </div>
   350→        </DialogHeader>
   351→
   352→        <div className="space-y-5">
   353→          {/* Status Banner */}
   354→          <div
   355→            className={`flex items-center gap-3 rounded-lg border p-4 ${statusInfo.bgClass}`}
   356→          >
   357→            <StatusIcon className="h-5 w-5" />
   358→            <div>
   359→              <p className="text-sm font-semibold">Status: {statusInfo.label}</p>
   360→              <p className="text-xs opacity-80">
   361→                {tx.status === 'success'
   362→                  ? 'Pembayaran telah berhasil diterima'
   363→                  : tx.status === 'pending'
   364→                    ? 'Menunggu pembayaran dari pelanggan'
   365→                    : tx.status === 'failed'
   366→                      ? 'Pembayaran gagal diproses'
   367→                      : tx.status === 'expired'
   368→                        ? 'Batas waktu pembayaran telah habis'
   369→                        : 'Dana telah dikembalikan ke pelanggan'}
   370→              </p>
   371→            </div>
   372→          </div>
   373→
   374→          {/* Amount Section */}
   375→          <div className="rounded-lg border bg-muted/30 p-4">
   376→            <p className="text-muted-foreground mb-1 text-xs font-medium uppercase tracking-wide">
   377→              Jumlah Pembayaran

   379→            <p className="text-2xl font-bold tracking-tight">{formatRp(tx.amount)}</p>

   522→          <DialogHeader>

   541→                <p className="font-medium text-red-800 dark:text-red-300">Refund sebesar {tx ? formatRp(tx.amount) : ''}</p>

   548→            {/* Refund reason */}

   682→// ── Pagination Helper ────────────────────────────────────────────────────────
   683→
   684→function getPaginationRange(current: number, total: number) {
   685→  const delta = 2;
   686→  const range: (number | 'dots')[] = [];
   687→  const left = Math.max(2, current - delta);
   688→  const right = Math.min(total - 1, current + delta);
   689→
   690→  range.push(1);
   691→  if (left > 2) range.push('dots');
   692→  for (let i = left; i <= right; i++) range.push(i);
   693→  if (right < total - 1) range.push('dots');
   694→  if (total > 1) range.push(total);
   695→
   696→  return range;
   697→}
