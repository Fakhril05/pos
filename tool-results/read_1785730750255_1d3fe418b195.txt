   538→// ── Pagination Helper ────────────────────────────────────────────────────────
   539→
   540→function getPaginationRange(current: number, total: number) {
   541→  const delta = 2;
   542→  const range: (number | 'dots')[] = [];
   543→  const left = Math.max(2, current - delta);
   544→  const right = Math.min(total - 1, current + delta);
   545→
   546→  range.push(1);
   547→  if (left > 2) range.push('dots');
   548→  for (let i = left; i <= right; i++) range.push(i);
   549→  if (right < total - 1) range.push('dots');
   550→  if (total > 1) range.push(total);
   551→
   552→  return range;
   553→}
