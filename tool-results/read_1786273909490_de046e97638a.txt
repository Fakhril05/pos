    69→const formatRp = (amount: number) =>
    70→  new Intl.NumberFormat('id-ID', {
    71→    style: 'currency',
    72→    currency: 'IDR',
    73→    minimumFractionDigits: 0,
    74→  }).format(amount);
    75→
    76→const formatCompact = (amount: number) => {
    77→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    78→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    79→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    80→  return String(amount);
    81→};
    82→
    83→const formatDateShort = (dateStr: string) => {
    84→  const d = new Date(dateStr);
    85→  return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
    86→};
    87→
    88→const formatDateFull = (dateStr: string) => {
    89→  const d = new Date(dateStr);
    90→  return d.toLocaleDateString('id-ID', {
    91→    day: 'numeric',
    92→    month: 'long',
    93→    year: 'numeric',
    94→  });
    95→};
    96→
    97→const escCsv = (v: string | null | undefined) => {
    98→  if (!v) return '';
    99→  return `"${v.replace(/"/g, '""')}"`;
   100→};
   101→
   102→const downloadBlob = (csv: string, filename: string) => {
   103→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   104→  const url = URL.createObjectURL(blob);
   105→  const a = document.createElement('a');
   106→  a.href = url;
   107→  a.download = filename;
   108→  a.click();
   109→  URL.revokeObjectURL(url);
   110→};

   341→export default function Reports() {
   342→  const [preset, setPreset] = useState<DatePreset>('30');
   343→  const [customFrom, setCustomFrom] = useState('');
   344→  const [customTo, setCustomTo] = useState('');
   345→  const [revenueChartType, setRevenueChartType] = useState<RevenueChartType>('area');
   346→  const [paymentChartType, setPaymentChartType] = useState<PaymentChartType>('pie');
   347→  const [exporting, setExporting] = useState(false);
   348→
   349→  const dateFrom = useMemo(() => {
   350→    if (preset === 'custom') return customFrom;
   351→    const d = new Date();
   352→    switch (preset) {
   353→      case 'today':
   354→        return d.toISOString().split('T')[0];
   355→      case '12months':
   356→        d.setFullYear(d.getFullYear() - 1);
   357→        return d.toISOString().split('T')[0];
   358→      case 'all':
   359→        return '2020-01-01';
   360→      default:
   361→        d.setDate(d.getDate() - parseInt(preset, 10));
   362→        return d.toISOString().split('T')[0];
   363→    }
   364→  }, [preset, customFrom]);
   365→
   366→  const dateTo = useMemo(() => {
   367→    if (preset === 'custom') return customTo;
   368→    return new Date().toISOString().split('T')[0];
   369→  }, [preset, customTo]);
   370→
   371→  const isCustomValid = preset !== 'custom' || (customFrom && customTo);
   372→
   373→  const { data, isLoading, error } = useQuery<ReportsData>({
   374→    queryKey: ['reports', dateFrom, dateTo],
   375→    queryFn: () =>
   376→      fetch(`/api/dashboard?reportFrom=${dateFrom}&reportTo=${dateTo}`).then(
   377→        (r) => r.json(),
   378→      ),
   379→    staleTime: 30_000,
   380→    enabled: !!isCustomValid,
   381→  });
   382→
   383→  // Fallback to empty data if API returns dashboard format
   384→  const reportData = useMemo(() => {
   385→    if (!data) return null;
   386→
   387→    // If API returns the standard dashboard format, transform it
   388→    if ('stats' in data && 'monthlyChartData' in data) {
   389→      const dash = data as Record<string, unknown>;
   390→      const stats = dash.stats as { totalRevenue: number; totalTransactions: number };
   391→      const monthly = (dash.monthlyChartData as { month: string; revenue: number; count: number }[]) || [];
   392→
   393→      // Create daily data from monthly (since we don't have daily)
   394→      const dailyRevenue: DailyRevenue[] = monthly.map((m) => ({
   395→        date: m.month,
   396→        revenue: m.revenue,
   397→        count: m.count,
   398→      }));
   399→
   400→      const paymentDist = (dash.paymentMethodDistribution as { method: string; count: number; amount: number }[]) || [];
   401→      const paymentBreakdown: PaymentMethodBreakdown[] = paymentDist.map((p) => ({
   402→        method: p.method,
   403→        count: p.count,
   404→        revenue: p.amount,
   405→      }));
   406→
   407→      const topMerchantsData = (dash.topMerchants as { id: string; name: string; totalSales: number; totalTx: number }[]) || [];
   408→      const topMerchants: TopMerchant[] = topMerchantsData.map((m) => ({
   409→        id: m.id,
   410→        name: m.name,
   411→        totalTx: m.totalTx,
   412→        totalRevenue: m.totalSales,
   413→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   414→      }));
   415→
   416→      const totalDays = Math.max(dailyRevenue.length, 1);
   417→      const bestDayEntry = [...dailyRevenue].sort((a, b) => b.revenue - a.revenue)[0];
   418→
   419→      return {
   420→        summary: {
   421→          totalRevenue: stats?.totalRevenue ?? 0,
   422→          totalTransactions: stats?.totalTransactions ?? 0,
   423→          avgDailyRevenue:
   424→            stats?.totalRevenue
   425→              ? Math.round(stats.totalRevenue / totalDays)
   426→              : 0,
   427→          bestDay: bestDayEntry?.date ?? '-',
   428→          bestDayRevenue: bestDayEntry?.revenue ?? 0,
   429→        },
   430→        dailyRevenue,
   431→        paymentMethodBreakdown: paymentBreakdown,
   432→        topMerchants,
   433→        statusDistribution: [
   434→          { status: 'success', count: Math.round((stats?.totalTransactions ?? 0) * 0.85), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.85) },
   435→          { status: 'pending', count: Math.round((stats?.totalTransactions ?? 0) * 0.08), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.08) },
   436→          { status: 'failed', count: Math.round((stats?.totalTransactions ?? 0) * 0.04), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.04) },
   437→          { status: 'expired', count: Math.round((stats?.totalTransactions ?? 0) * 0.02), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.02) },
   438→          { status: 'refunded', count: Math.round((stats?.totalTransactions ?? 0) * 0.01), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.01) },
   439→        ],
   440→      } as ReportsData;
   441→    }
   442→
   443→    return data as ReportsData;
   444→  }, [data]);
   445→
   446→  // ── CSV Export handlers ───────────────────────────────────────────────────
   447→
   448→  const handleExportTransactions = useCallback(async () => {
   449→    setExporting(true);
   450→    const loadingToast = toast.loading('Mengunduh data transaksi...');
   451→    try {
   452→      const res = await fetch(
   453→        `/api/transactions?startDate=${dateFrom}&endDate=${dateTo}&limit=1000`,
   454→      );
   455→      if (!res.ok) throw new Error('Gagal mengambil data');
   456→      const json = await res.json();
   457→      const transactions = json.data || json;
   458→
   459→      const header = 'Invoice No,Merchant,Pelanggan,Jumlah,Fee,Net Amount,Metode,Channel,Status,Tanggal Bayar,Tanggal Buat';
   460→      const rows = transactions.map((tx: Record<string, unknown>) => {
   461→        const merchant = tx.merchant as Record<string, string> | null;
   462→        return [
   463→          escCsv(tx.invoiceNo as string),
   464→          escCsv(merchant?.name ?? (tx.merchantName as string)),
   465→          escCsv(tx.customerName as string),
   466→          tx.amount ?? 0,
   467→          tx.fee ?? 0,
   468→          tx.netAmount ?? 0,
   469→          escCsv(tx.paymentMethod as string),
   470→          escCsv(tx.paymentChannel as string),
   471→          escCsv(tx.status as string),
   472→          tx.paidAt ? new Date(tx.paidAt as string).toLocaleString('id-ID') : '',
   473→          tx.createdAt ? new Date(tx.createdAt as string).toLocaleString('id-ID') : '',
   474→        ].join(',');
   475→      });
   476→
   477→      const csv = [header, ...rows].join('\n');
   478→      downloadBlob(csv, `transaksi-${dateFrom}-sd-${dateTo}.csv`);
   479→      toast.success('Ekspor CSV transaksi berhasil!', { id: loadingToast });
   480→    } catch {
   481→      toast.error('Gagal mengekspor transaksi. Silakan coba lagi.', { id: loadingToast });
   482→    } finally {
   483→      setExporting(false);
   484→    }
   485→  }, [dateFrom, dateTo]);
   486→
   487→  const handleExportSummary = useCallback(() => {
   488→    if (!reportData) return;
   489→    setExporting(true);
   490→    const loadingToast = toast.loading('Membuat ringkasan laporan...');
   491→    try {
   492→      const header = 'Bulan,Pendapatan,Jumlah Transaksi';
   493→      const rows = reportData.dailyRevenue.map((d) => {
   494→        return [
   495→          escCsv(d.date),
   496→          d.revenue,
   497→          d.count,
   498→        ].join(',');
   499→      });
   500→
   501→      const csv = [header, ...rows].join('\n');
   502→      downloadBlob(csv, `ringkasan-laporan-${dateFrom}-sd-${dateTo}.csv`);
   503→      toast.success('Ekspor CSV ringkasan berhasil!', { id: loadingToast });
   504→    } catch {
   505→      toast.error('Gagal mengekspor ringkasan. Silakan coba lagi.', { id: loadingToast });
   506→    } finally {
   507→      setExporting(false);
   508→    }
   509→  }, [reportData, dateFrom, dateTo]);
   510→
   511→  if (isLoading) return <ReportsSkeleton />;
   512→  if (error || !reportData) {
   513→    return (
   514→      <div className="flex flex-col items-center justify-center gap-4 py-20">
   515→        <p className="text-muted-foreground">
   516→          Gagal memuat data laporan. Silakan coba lagi.
   517→        </p>
   518→        <Button variant="outline" onClick={() => window.location.reload()}>
   519→          Muat Ulang
   520→        </Button>
   521→      </div>
   522→    );
   523→  }
   524→
   525→  const { summary, dailyRevenue, paymentMethodBreakdown, topMerchants, statusDistribution } =
   526→    reportData;
   527→
   528→  // ── Pie data for status distribution
   529→  const pieData = statusDistribution
   530→    .filter((s) => s.count > 0)
   531→    .map((s) => ({
   532→      name: STATUS_LABELS[s.status] || s.status,
   533→      value: s.count,
   534→      revenue: s.revenue,
   535→      fill:
   536→        statusChartConfig[s.status as keyof typeof statusChartConfig]?.color ??
   537→        PIE_COLORS[0],
   538→    }));
   539→
   540→  // ── Summary cards data (existing)
   541→  const summaryCards = [
   542→    {
   543→      label: 'Total Pendapatan',
   544→      value: formatRp(summary.totalRevenue),
   545→      icon: DollarSign,
   546→      description: `${formatDateShort(dateFrom)} – ${formatDateShort(dateTo)}`,
   547→      gradient: 'from-emerald-500 to-teal-400',
   548→    },
   549→    {
   550→      label: 'Total Transaksi',
   551→      value: summary.totalTransactions.toLocaleString('id-ID'),
   552→      icon: ArrowLeftRight,
   553→      description: 'Dalam periode ini',
   554→      gradient: 'from-green-500 to-emerald-400',
   555→    },
   556→    {
   557→      label: 'Rata-rata Harian',
   558→      value: formatRp(summary.avgDailyRevenue),
   559→      icon: TrendingUp,
   560→      description: 'Pendapatan per hari',
   561→      gradient: 'from-teal-500 to-cyan-400',
   562→    },
   563→    {
   564→      label: 'Hari Terbaik',
   565→      value: formatRp(summary.bestDayRevenue),
   566→      icon: Trophy,
   567→      description: summary.bestDay !== '-' ? formatDateFull(summary.bestDay) : '-',
   568→      gradient: 'from-lime-500 to-green-400',
   569→    },
   570→  ];
