   204→function buildTimeline(tx: Transaction): TimelineEvent[] {
   205→  const events: TimelineEvent[] = [
   206→    {
   207→      label: 'Transaksi dibuat',
   208→      date: tx.createdAt,
   209→      icon: Receipt,
   210→      color: 'text-emerald-500',
   211→    },
   212→  ];
   213→  if (tx.paidAt) {
   214→    events.push({
   215→      label: 'Pembayaran diterima',
   216→      date: tx.paidAt,
   217→      icon: CheckCircle2,
   218→      color: 'text-emerald-500',
   219→    });
   220→  }
   221→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   222→    events.push({
   223→      label: 'Transaksi kedaluwarsa',
   224→      date: tx.expiredAt,
   225→      icon: AlertTriangle,
   226→      color: 'text-gray-500',
   227→    });
   228→  }
   229→  if (tx.status === 'failed') {
   230→    events.push({
   231→      label: 'Transaksi gagal',
   232→      date: tx.updatedAt,
   233→      icon: XCircle,
   234→      color: 'text-red-500',
   235→    });
   236→  }
   237→  if (tx.status === 'refunded') {
   238→    events.push({
   239→      label: 'Refund diproses',
   240→      date: tx.updatedAt,
   241→      icon: RotateCcw,
   242→      color: 'text-blue-500',
   243→    });
   244→  }
   245→  return events;
   246→}

   683→// ── Pagination Helper ────────────────────────────────────────────────────────
   684→
   685→function getPaginationRange(current: number, total: number) {
   686→  const delta = 2;
   687→  const range: (number | 'dots')[] = [];
   688→  const left = Math.max(2, current - delta);
   689→  const right = Math.min(total - 1, current + delta);
   690→
   691→  range.push(1);
   692→  if (left > 2) range.push('dots');
   693→  for (let i = left; i <= right; i++) range.push(i);
   694→  if (right < total - 1) range.push('dots');
   695→  if (total > 1) range.push(total);
   696→
   697→  return range;
   698→}
