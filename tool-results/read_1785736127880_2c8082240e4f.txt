   199→function buildTimeline(tx: Transaction): TimelineEvent[] {
   200→  const events: TimelineEvent[] = [
   201→    {
   202→      label: 'Transaksi dibuat',
   203→      date: tx.createdAt,
   204→      icon: Receipt,
   205→      color: 'text-emerald-500',
   206→    },
   207→  ];
   208→  if (tx.paidAt) {
   209→    events.push({
   210→      label: 'Pembayaran diterima',
   211→      date: tx.paidAt,
   212→      icon: CheckCircle2,
   213→      color: 'text-emerald-500',
   214→    });
   215→  }
   216→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   217→    events.push({
   218→      label: 'Transaksi kedaluwarsa',
   219→      date: tx.expiredAt,
   220→      icon: AlertTriangle,
   221→      color: 'text-gray-500',
   222→    });
   223→  }
   224→  if (tx.status === 'failed') {
   225→    events.push({
   226→      label: 'Transaksi gagal',
   227→      date: tx.updatedAt,
   228→      icon: XCircle,
   229→      color: 'text-red-500',
   230→    });
   231→  }
   232→  if (tx.status === 'refunded') {
   233→    events.push({
   234→      label: 'Refund diproses',
   235→      date: tx.updatedAt,
   236→      icon: RotateCcw,
   237→      color: 'text-blue-500',

   577→// ── Pagination Helper ────────────────────────────────────────────────────────
   578→
   579→function getPaginationRange(current: number, total: number) {
   580→  const delta = 2;
   581→  const range: (number | 'dots')[] = [];
   582→  const left = Math.max(2, current - delta);
   583→  const right = Math.min(total - 1, current + delta);
   584→
   585→  range.push(1);
   586→  if (left > 2) range.push('dots');
   587→  for (let i = left; i <= right; i++) range.push(i);
   588→  if (right < total - 1) range.push('dots');
   589→  if (total > 1) range.push(total);
   590→
   591→  return range;
   592→}
