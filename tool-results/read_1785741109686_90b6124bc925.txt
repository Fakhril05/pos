    62→    62→const formatRp = (amount: number) =>
    63→    63→  new Intl.NumberFormat('id-ID', {
    64→    64→    style: 'currency',
    65→    65→    currency: 'IDR',
    66→    66→    minimumFractionDigits: 0,

    68→    68→
    69→    69→const formatTimeAgo = (dateStr: string): string => {
    70→    70→  const now = Date.now();
    71→    71→  const then = new Date(dateStr).getTime();
    72→    72→  const diffMs = now - then;
    73→    73→  const seconds = Math.floor(diffMs / 1000);
    74→    74→  const minutes = Math.floor(seconds / 60);
    75→    75→  const hours = Math.floor(minutes / 60);
    76→    76→  const days = Math.floor(hours / 24);
    77→    77→  if (days > 0) return `${days} hari lalu`;
    78→    78→  if (hours > 0) return `${hours} jam lalu`;
    79→    79→  if (minutes > 0) return `${minutes} menit lalu`;
    80→    80→  return 'Baru saja';
    81→    81→};
    82→    82→
    83→    83→const formatCompact = (amount: number) => {
    84→    84→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    85→    85→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    86→    86→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    87→    87→  return String(amount);
    88→    88→};
