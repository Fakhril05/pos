   194→// ── Timeline Item ────────────────────────────────────────────────────────────
   195→
   196→interface TimelineEvent {
   197→  label: string;
   198→  date: string;
   199→  icon: React.ElementType;
   200→  color: string;
   201→}
   202→
   203→function buildTimeline(tx: Transaction): TimelineEvent[] {
   204→  const events: TimelineEvent[] = [
   205→    {
   206→      label: 'Transaksi dibuat',
   207→      date: tx.createdAt,
   208→      icon: Receipt,
   209→      color: 'text-emerald-500',
   210→    },
   211→  ];
   212→  if (tx.paidAt) {
   213→    events.push({
   214→      label: 'Pembayaran diterima',
   215→      date: tx.paidAt,
   216→      icon: CheckCircle2,
   217→      color: 'text-emerald-500',
   218→    });
   219→  }
   220→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   221→    events.push({
   222→      label: 'Transaksi kedaluwarsa',
   223→      date: tx.expiredAt,
   224→      icon: AlertTriangle,
   225→      color: 'text-gray-500',
   226→    });
   227→  }
   228→  if (tx.status === 'failed') {
   229→    events.push({
   230→      label: 'Transaksi gagal',
   231→      date: tx.updatedAt,
   232→      icon: XCircle,
   233→      color: 'text-red-500',
   234→    });
   235→  }
   236→  if (tx.status === 'refunded') {
   237→    events.push({
   238→      label: 'Refund diproses',
   239→      date: tx.updatedAt,
   240→      icon: RotateCcw,
   241→      color: 'text-blue-500',
   242→    });
   243→  }
   244→  return events;
   245→}
   246→
   247→function TimelineItem({ event, isLast }: { event: TimelineEvent; isLast: boolean }) {
   248→  const Icon = event.icon;
   249→  return (
   250→    <div className="flex gap-3">
   251→      <div className="flex flex-col items-center">
   252→        <div
   253→          className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-muted ${event.color}`}
   254→        >
   255→          <Icon className="h-4 w-4" />
   256→        </div>
   257→        {!isLast && <div className="w-px flex-1 bg-border" />}
   258→      </div>
   259→      <div className={`pb-6 ${isLast ? 'pb-0' : ''}`}>
   260→        <p className="text-sm font-medium">{event.label}</p>
   261→        <p className="text-muted-foreground text-xs">{formatDate(event.date)}</p>
   262→      </div>
   263→    </div>
   264→  );
   265→}

   286→    if (!tx) return;
   287→    if (!refundReason.trim()) {
   288→      setRefundReasonError('Alasan refund wajib diisi');
   289→      return;
   290→    }

   292→    try {
   293→      const res = await fetch(`/api/transactions/${tx.id}`, {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify({ amount: tx.amount, reason: refundReason.trim() }),
   297→      });

   299→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   300→        throw new Error(err.error || 'Gagal memproses refund');
   301→      }
   302→      toast.success('Refund berhasil diajukan');
   303→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   304→      setRefundDialogOpen(false);
   305→      setRefundReason('');
   306→      onOpenChange(false);
   307→    } catch (e) {
   308→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');

   310→      setRefunding(false);
   311→    }
   312→  };

   315→    if (tx) {
   316→      navigator.clipboard.writeText(tx.invoiceNo);

   319→  };
   320→
   321→  if (!tx) return null;

   324→  const StatusIcon = statusInfo.icon;

   334→            <div>
   335→              <DialogTitle className="text-base">Detail Transaksi</DialogTitle>
   336→              <DialogDescription>
   337→                <span className="flex items-center gap-1.5 font-mono text-xs">
   338→                  {tx.invoiceNo}
   339→                  <button
   340→                    onClick={handleCopyInvoice}
   341→                    className="inline-flex hover:text-foreground transition-colors"
   342→                  >
   343→                    <Copy className="h-3 w-3" />
   344→                  </button>
   345→                </span>
   346→              </DialogDescription>
   347→            </div>
   348→          </div>
   349→        </DialogHeader>
   350→
   351→        <div className="space-y-5">
   352→          {/* Status Banner */}
   353→          <div
   354→            className={`flex items-center gap-3 rounded-lg border p-4 ${statusInfo.bgClass}`}
   355→          >
   356→            <StatusIcon className="h-5 w-5" />
   357→            <div>
   358→              <p className="text-sm font-semibold">Status: {statusInfo.label}</p>
   359→              <p className="text-xs opacity-80">
   360→                {tx.status === 'success'
   361→                  ? 'Pembayaran telah berhasil diterima'
   362→                  : tx.status === 'pending'
   363→                    ? 'Menunggu pembayaran dari pelanggan'

   365→                      ? 'Pembayaran gagal diproses'

   373→          {/* Amount Section */}

   681→// ── Pagination Helper ────────────────────────────────────────────────────────
   682→
   683→function getPaginationRange(current: number, total: number) {
   684→  const delta = 2;
   685→  const range: (number | 'dots')[] = [];
   686→  const left = Math.max(2, current - delta);
   687→  const right = Math.min(total - 1, current + delta);
   688→
   689→  range.push(1);
   690→  if (left > 2) range.push('dots');
   691→  for (let i = left; i <= right; i++) range.push(i);
   692→  if (right < total - 1) range.push('dots');
   693→  if (total > 1) range.push(total);
   694→
   695→  return range;
   696→}
