   260→   260→
   261→   261→export default function Reports() {
   262→   262→  const [preset, setPreset] = useState<DatePreset>('30');
   263→   263→  const [customFrom, setCustomFrom] = useState('');
   264→   264→  const [customTo, setCustomTo] = useState('');
   265→   265→
   266→   266→  const dateFrom = useMemo(() => {
   267→   267→    if (preset === 'custom') return customFrom;
   268→   268→    const d = new Date();
   269→   269→    d.setDate(d.getDate() - parseInt(preset, 10));
   270→   270→    return d.toISOString().split('T')[0];
   271→   271→  }, [preset, customFrom]);
   272→   272→
   273→   273→  const dateTo = useMemo(() => {
   274→   274→    if (preset === 'custom') return customTo;
   275→   275→    return new Date().toISOString().split('T')[0];
   276→   276→  }, [preset, customTo]);
   277→   277→
   278→   278→  const isCustomValid = preset !== 'custom' || (customFrom && customTo);
   279→   279→
   280→   280→  const { data, isLoading, error } = useQuery<ReportsData>({
   281→   281→    queryKey: ['reports', dateFrom, dateTo],
   282→   282→    queryFn: () =>
   283→   283→      fetch(`/api/dashboard?reportFrom=${dateFrom}&reportTo=${dateTo}`).then(
   284→   284→        (r) => r.json(),
   285→   285→      ),
   286→   286→    staleTime: 30_000,
   287→   287→    enabled: !!isCustomValid,
   288→   288→  });
   289→   289→
   290→   290→  // Fallback to empty data if API returns dashboard format
   291→   291→  const reportData = useMemo(() => {
   292→   292→    if (!data) return null;
   293→   293→
   294→   294→    // If API returns the standard dashboard format, transform it
   295→   295→    if ('stats' in data && 'monthlyChartData' in data) {
   296→   296→      const dash = data as Record<string, unknown>;
   297→   297→      const stats = dash.stats as { totalRevenue: number; totalTransactions: number };
   298→   298→      const monthly = (dash.monthlyChartData as { month: string; revenue: number; count: number }[]) || [];
   299→   299→
   300→   300→      // Create daily data from monthly (since we don't have daily)
   301→   301→      const dailyRevenue: DailyRevenue[] = monthly.map((m) => ({
   302→   302→        date: m.month,
   303→   303→        revenue: m.revenue,
   304→   304→        count: m.count,
   305→   305→      }));
   306→   306→
   307→   307→      const paymentDist = (dash.paymentMethodDistribution as { method: string; count: number; amount: number }[]) || [];
   308→   308→      const paymentBreakdown: PaymentMethodBreakdown[] = paymentDist.map((p) => ({
   309→   309→        method: p.method,
   310→   310→        count: p.count,
   311→   311→        revenue: p.amount,
   312→   312→      }));
   313→   313→
   314→   314→      const topMerchantsData = (dash.topMerchants as { id: string; name: string; totalSales: number; totalTx: number }[]) || [];
   315→   315→      const topMerchants: TopMerchant[] = topMerchantsData.map((m) => ({
   316→   316→        id: m.id,
   317→   317→        name: m.name,
   318→   318→        totalTx: m.totalTx,
   319→   319→        totalRevenue: m.totalSales,
   320→   320→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   321→   321→      }));
   322→   322→
   323→   323→      const totalDays = Math.max(dailyRevenue.length, 1);
   324→   324→      const bestDayEntry = [...dailyRevenue].sort((a, b) => b.revenue - a.revenue)[0];
   325→   325→
   326→   326→      return {
   327→   327→        summary: {
   328→   328→          totalRevenue: stats?.totalRevenue ?? 0,
   329→   329→          totalTransactions: stats?.totalTransactions ?? 0,
   330→   330→          avgDailyRevenue:
   331→   331→            stats?.totalRevenue
   332→   332→              ? Math.round(stats.totalRevenue / totalDays)
   333→   333→              : 0,
   334→   334→          bestDay: bestDayEntry?.date ?? '-',
   335→   335→          bestDayRevenue: bestDayEntry?.revenue ?? 0,
   336→   336→        },
   337→   337→        dailyRevenue,
   338→   338→        paymentMethodBreakdown: paymentBreakdown,
   339→   339→        topMerchants,
   340→   340→        statusDistribution: [
   341→   341→          { status: 'success', count: Math.round((stats?.totalTransactions ?? 0) * 0.85), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.85) },
   342→   342→          { status: 'pending', count: Math.round((stats?.totalTransactions ?? 0) * 0.08), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.08) },
   343→   343→          { status: 'failed', count: Math.round((stats?.totalTransactions ?? 0) * 0.04), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.04) },
   344→   344→          { status: 'expired', count: Math.round((stats?.totalTransactions ?? 0) * 0.02), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.02) },
   345→   345→          { status: 'refunded', count: Math.round((stats?.totalTransactions ?? 0) * 0.01), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.01) },
   346→   346→        ],
   347→   347→      } as ReportsData;
   348→   348→    }
   349→   349→
   350→   350→    return data as ReportsData;
   351→   351→  }, [data]);

   358→   358→          Gagal memuat data laporan. Silakan coba lagi.
   359→   359→        </p>

   414→   414→  // ── Bar chart data

   421→   421→  return (

   430→   430→      {/* ─── Date Range Picker ─────────────────────────────────────── */}

   435→   435→      >
   436→   436→        <Card className="card-elevated">
   437→   437→          <CardContent className="p-4">
   438→   438→            <div className="flex flex-wrap items-center gap-3">
   439→   439→              <div className="flex items-center gap-1 rounded-lg bg-muted p-1">
   440→   440→                {([

   459→   459→
   460→   460→              {preset === 'custom' && (
