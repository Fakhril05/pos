    52→// ── Helpers ──────────────────────────────────────────────────────────────────
    53→
    54→const formatRp = (amount: number) =>
    55→  new Intl.NumberFormat('id-ID', {
    56→    style: 'currency',
    57→    currency: 'IDR',
    58→    minimumFractionDigits: 0,
    59→  }).format(amount);
    60→
    61→const formatTimeAgo = (dateStr: string): string => {
    62→  const now = Date.now();
    63→  const then = new Date(dateStr).getTime();
    64→  const diffMs = now - then;
    65→  const seconds = Math.floor(diffMs / 1000);
    66→  const minutes = Math.floor(seconds / 60);
    67→  const hours = Math.floor(minutes / 60);
    68→  const days = Math.floor(hours / 24);
    69→  if (days > 0) return `${days} hari lalu`;
    70→  if (hours > 0) return `${hours} jam lalu`;
    71→  if (minutes > 0) return `${minutes} menit lalu`;
    72→  return 'Baru saja';
    73→};
    74→
    75→const formatCompact = (amount: number) => {
    76→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    77→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    78→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    79→  return String(amount);
    80→};
