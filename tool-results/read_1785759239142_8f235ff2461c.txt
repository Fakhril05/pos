    63→// ── Helpers ──────────────────────────────────────────────────────────────────
    64→
    65→const formatRp = (amount: number) =>
    66→  new Intl.NumberFormat('id-ID', {
    67→    style: 'currency',
    68→    currency: 'IDR',
    69→    minimumFractionDigits: 0,
    70→  }).format(amount);
    71→
    72→const formatTimeAgo = (dateStr: string): string => {
    73→  const now = Date.now();
    74→  const then = new Date(dateStr).getTime();
    75→  const diffMs = now - then;
    76→  const seconds = Math.floor(diffMs / 1000);
    77→  const minutes = Math.floor(seconds / 60);
    78→  const hours = Math.floor(minutes / 60);
    79→  const days = Math.floor(hours / 24);
    80→  if (days > 0) return `${days} hari lalu`;
    81→  if (hours > 0) return `${hours} jam lalu`;
    82→  if (minutes > 0) return `${minutes} menit lalu`;
    83→  return 'Baru saja';
    84→};
    85→
    86→const formatCompact = (amount: number) => {
    87→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    88→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    89→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    90→  return String(amount);
    91→};

   486→// ── Live Clock Widget ───────────────────────────────────────────────────────

   488→const HARI = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];
   489→const BULAN = [
   490→  'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
   491→  'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember',
   492→];
   493→
   494→function LiveClock() {
   495→  const [now, setNow] = useState(() => new Date());
   496→
   497→  useEffect(() => {
   498→    const id = setInterval(() => setNow(new Date()), 1000);
   499→    return () => clearInterval(id);
   500→  }, []);
   501→
   502→  const time = now.toLocaleTimeString('id-ID', {
   503→    hour: '2-digit',
   504→    minute: '2-digit',
   505→    second: '2-digit',
   506→    hour12: false,
   507→  });
   508→  const date = `${HARI[now.getDay()]}, ${now.getDate()} ${BULAN[now.getMonth()]} ${now.getFullYear()}`;
   509→
   510→  return (
   511→    <div className="glass-card flex flex-col items-end rounded-xl px-4 py-2.5 text-sm">
   512→      <div className="flex items-center gap-2">
   513→        <span className="relative flex h-2 w-2">

   519→        </span>

   522→        {time}
   523→      </span>

   526→  );
