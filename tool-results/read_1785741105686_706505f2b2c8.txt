    60→// ── Helpers ──────────────────────────────────────────────────────────────────
    61→
    62→const formatRp = (amount: number) =>
    63→  new Intl.NumberFormat('id-ID', {
    64→    style: 'currency',
    65→    currency: 'IDR',
    66→    minimumFractionDigits: 0,
    67→  }).format(amount);
    68→
    69→const formatTimeAgo = (dateStr: string): string => {
    70→  const now = Date.now();
    71→  const then = new Date(dateStr).getTime();
    72→  const diffMs = now - then;
    73→  const seconds = Math.floor(diffMs / 1000);
    74→  const minutes = Math.floor(seconds / 60);
    75→  const hours = Math.floor(minutes / 60);
    76→  const days = Math.floor(hours / 24);
    77→  if (days > 0) return `${days} hari lalu`;
    78→  if (hours > 0) return `${hours} jam lalu`;
    79→  if (minutes > 0) return `${minutes} menit lalu`;
    80→  return 'Baru saja';
    81→};
    82→
    83→const formatCompact = (amount: number) => {
    84→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    85→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    86→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    87→  return String(amount);
    88→};

   324→const statusDotColors: Record<string, string> = {
   325→  success: 'bg-emerald-500',
   326→  pending: 'bg-amber-500',
   327→  failed: 'bg-red-500',
   328→  expired: 'bg-gray-400',

   331→
   332→function ActivityFeed() {
   333→  const { setActivePage } = useAppStore();
   334→  const [liveItems, setLiveItems] = useState<SocketTransaction[]>([]);
   335→
   336→  const { data: activityData, isLoading: activityLoading } = useQuery<{
   337→    data: ActivityTransaction[];
   338→  }>({
   339→    queryKey: ['activity-feed'],
   340→    queryFn: () =>
   341→      fetch('/api/transactions?limit=8&status=success').then((r) => r.json()),
   342→    staleTime: 15_000,
   343→    refetchInterval: 30_000,
   344→  });
   345→
   346→  const baseItems = activityData?.data?.slice(0, 6) ?? [];
   347→
   348→  // Socket integration — receive live transactions
   349→  useTransactionSocket({
   350→    onTransaction: (tx) => {
   351→      setLiveItems((prev) => [tx, ...prev].slice(0, 10));
   352→      toast('Pembayaran Baru!', {
   353→        description: `${formatRp(tx.amount)} dari ${tx.customerName} — ${tx.merchantName}`,
   354→        icon: <CreditCard className="h-4 w-4 text-emerald-500" />,
   355→      });
   356→    },
   357→  });
   358→
   359→  // Merge: live items first, then DB items, deduplicated
   360→  const liveIds = new Set(liveItems.map((t) => t.id));
   361→  const merged: (ActivityTransaction | SocketTransaction)[] = [
   362→    ...liveItems,
   363→    ...baseItems.filter((t) => !liveIds.has(t.id)),
   364→  ].slice(0, 6);
   365→
   366→  return (
   367→    <motion.div
   368→      initial={{ opacity: 0, y: 20 }}
   369→      animate={{ opacity: 1, y: 0 }}
   370→      transition={{ duration: 0.5, delay: 0.8 }}
   371→    >
   372→      <Card className="card-hover shadow-sm">
   373→        <CardHeader>
   374→          <div className="flex items-center justify-between">
   375→            <div className="flex items-center gap-2">
   376→              <Activity className="h-4 w-4 text-emerald-500" />
   377→              <span className="rounded-full w-2 h-2 bg-emerald-500 inline-block mr-2 status-dot" />
   378→              <CardTitle className="text-base">Aktivitas Terkini</CardTitle>
   379→            </div>
   380→            <span className="relative flex h-2.5 w-2.5">
   381→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   382→              <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-emerald-500" />
   383→            </span>
   384→          </div>
   385→          <p className="text-muted-foreground text-sm">
   386→            Transaksi terbaru yang berhasil diproses
   387→          </p>
   388→        </CardHeader>
   389→        <CardContent className="scrollbar-thin">
   390→          {activityLoading ? (
   391→            <div className="space-y-4">
   392→              {Array.from({ length: 4 }).map((_, i) => (
   393→                <div key={i} className="flex gap-3">
   394→                  <Skeleton className="h-8 w-8 shrink-0 rounded-full" />
   395→                  <div className="flex-1 space-y-2">
   396→                    <Skeleton className="h-4 w-3/4" />
   397→                    <Skeleton className="h-3 w-1/2" />
   398→                  </div>
   399→                </div>
   400→              ))}
   401→            </div>
   402→          ) : merged.length === 0 ? (
   403→            <p className="py-8 text-center text-sm text-muted-foreground">
   404→              Belum ada aktivitas terkini.
   405→            </p>
   406→          ) : (
   407→            <div className="relative space-y-0">
   408→              {/* Vertical connecting line */}
   409→              <div className="absolute bottom-0 left-[11px] top-2 w-px bg-border" />
   410→
   411→              {merged.map((tx, idx) => {
   412→                // Handle both ActivityTransaction (nested) and SocketTransaction (flat)
   413→                const merchantName =
   414→                  'merchant' in tx && tx.merchant
   415→                    ? (tx as ActivityTransaction).merchant.name
   416→                    : (tx as SocketTransaction).merchantName;
   417→                const customerName =
   418→                  'customer' in tx && tx.customer
   419→                    ? (tx as ActivityTransaction).customer!.name
   420→                    : (tx as SocketTransaction).customerName;
   421→
   422→                return (
   423→                <motion.div
   424→                  key={tx.id}
   425→                  initial={{ opacity: 0, x: -10 }}
   426→                  animate={{ opacity: 1, x: 0 }}
   427→                  transition={{ duration: 0.3, delay: idx * 0.07 }}
   428→                  className={cn(
   429→                    'group relative flex gap-4 rounded-lg p-2.5 transition-colors hover:bg-muted/50',
   430→                    idx < liveItems.length && 'bg-emerald-50/50 dark:bg-emerald-950/20'
   431→                  )}
   432→                >
   433→                  {/* Timeline dot */}
   434→                  <div className="relative z-10 mt-1 flex shrink-0 items-start justify-center">
   435→                    <div
   436→                      className={cn(
   437→                        'h-3 w-3 rounded-full ring-4 ring-background',
   438→                        statusDotColors[tx.status] ?? 'bg-gray-400'
   439→                      )}
   440→                    />
   441→                  </div>
   442→
   443→                  {/* Content */}
   444→                  <div className="min-w-0 flex-1 pb-3">
   445→                    <div className="flex flex-col gap-0.5 sm:flex-row sm:items-baseline sm:justify-between">
   446→                      <p className="text-sm">
   447→                        <span className="font-semibold">{merchantName}</span>
   448→                        <span className="text-muted-foreground"> — </span>
   449→                        <span className="text-muted-foreground">
   450→                          {customerName.split(' ').slice(0, 2).join(' ')}
   451→                        </span>
   452→                      </p>
   453→                      <span className="text-xs font-semibold text-emerald-600 dark:text-emerald-400">
   454→                        +{formatRp(tx.amount)}
   455→                      </span>
   456→                    </div>
   457→                    <p className="mt-0.5 text-xs text-muted-foreground">
   458→                      {formatTimeAgo(tx.createdAt)}
   459→                    </p>
   460→                  </div>
   461→                </motion.div>
   462→                );
   463→              })}
   464→            </div>
   465→          )}
   466→
   467→          {/* View all link */}
   468→          <div className="mt-2 border-t pt-3">
   469→            <button
   470→              onClick={() => setActivePage('transactions')}
   471→              className="text-sm font-medium text-emerald-600 transition-colors hover:text-emerald-700 dark:text-emerald-400 dark:hover:text-emerald-300"
   472→            >
   473→              Lihat Semua →
   474→            </button>
   475→          </div>
   476→        </CardContent>
   477→      </Card>
   478→    </motion.div>
   479→  );
   480→}
