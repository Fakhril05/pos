   219→// ── Timeline Item ────────────────────────────────────────────────────────────
   220→
   221→interface TimelineEvent {
   222→  label: string;
   223→  date: string;
   224→  icon: React.ElementType;
   225→  color: string;
   226→}
   227→
   228→function buildTimeline(tx: Transaction): TimelineEvent[] {
   229→  const events: TimelineEvent[] = [
   230→    {
   231→      label: 'Transaksi dibuat',
   232→      date: tx.createdAt,
   233→      icon: Receipt,
   234→      color: 'text-emerald-500',
   235→    },
   236→  ];
   237→  if (tx.paidAt) {
   238→    events.push({
   239→      label: 'Pembayaran diterima',
   240→      date: tx.paidAt,
   241→      icon: CheckCircle2,
   242→      color: 'text-emerald-500',
   243→    });
   244→  }
   245→  if (tx.status === 'failed') {
   246→    events.push({
   247→      label: 'Transaksi gagal',
   248→      date: tx.updatedAt,
   249→      icon: XCircle,
   250→      color: 'text-red-500',
   251→    });
   252→  }
   253→  if (tx.status === 'refunded') {
   254→    events.push({
   255→      label: 'Refund diproses',
   256→      date: tx.updatedAt,
   257→      icon: RotateCcw,
   258→      color: 'text-blue-500',
   259→    });
   260→  }
   261→  return events;
   262→}
   263→
   264→function TimelineItem({ event, isLast }: { event: TimelineEvent; isLast: boolean }) {
   265→  const Icon = event.icon;
   266→  return (
   267→    <div className="flex gap-3">
   268→      <div className="flex flex-col items-center">
   269→        <div
   270→          className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-muted ${event.color}`}
   271→        >
   272→          <Icon className="h-4 w-4" />
   273→        </div>
   274→        {!isLast && <div className="w-px flex-1 bg-border" />}
   275→      </div>
   276→      <div className={`pb-6 ${isLast ? 'pb-0' : ''}`}>
   277→        <p className="text-sm font-medium">{event.label}</p>
   278→        <p className="text-muted-foreground text-xs">{formatDate(event.date)}</p>
   279→      </div>
   280→    </div>
   281→  );
   282→}

   300→
   301→  const timeline = useMemo(() => (tx ? buildTimeline(tx) : []), [tx]);
   302→
   303→  const handleRefund = async () => {
   304→    if (!tx) return;
   305→    if (!refundReason.trim()) {
   306→      setRefundReasonError('Alasan refund wajib diisi');
   307→      return;
   308→    }
   309→    setRefunding(true);
   310→    try {
   311→      const res = await fetch(`/api/transactions/${tx.id}`, {
   312→        method: 'POST',
   313→        headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({ amount: tx.totalAmount, reason: refundReason.trim() }),
   315→      });
   316→      if (!res.ok) {
   317→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   318→        throw new Error(err.error || 'Gagal memproses refund');
   319→      }
   320→      toast.success('Refund berhasil diajukan');
   321→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   322→      setRefundDialogOpen(false);
   323→      setRefundReason('');
   324→      onOpenChange(false);
   325→    } catch (e) {
   326→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');
   327→    } finally {
   328→      setRefunding(false);
   329→    }
   330→  };

   332→  const handleCopyInvoice = () => {
   333→    if (tx) {
   334→      navigator.clipboard.writeText(tx.invoiceNo);
   335→      toast.success('Nomor invoice disalin');
   336→    }
   337→  };
   338→
   339→  if (!tx) return null;
   340→
   341→  const statusInfo = STATUS_MAP[tx.status] ?? STATUS_MAP.pending;
   342→  const StatusIcon = statusInfo.icon;
   343→
   344→  return (
   345→    <Dialog open={open} onOpenChange={onOpenChange}>
   346→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin">
   347→        <DialogHeader>

   349→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950/50">
   350→              <Receipt className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   351→            </div>
   352→            <div>
   353→              <DialogTitle className="text-base">Detail Transaksi</DialogTitle>
   354→              <DialogDescription>
   355→                <span className="flex items-center gap-1.5 font-mono text-xs">
   356→                  {tx.invoiceNo}
   357→                  <button
   358→                    onClick={handleCopyInvoice}
   359→                    className="inline-flex hover:text-foreground transition-colors"
   360→                  >
   361→                    <Copy className="h-3 w-3" />
   362→                  </button>
   363→                </span>
   364→              </DialogDescription>
   365→            </div>
   366→          </div>
   367→        </DialogHeader>
   368→
   369→        <div className="space-y-5">
   370→          {/* Status Banner */}
   371→          <div
   372→            className={`flex items-center gap-3 rounded-lg border p-4 ${statusInfo.bgClass}`}
   373→          >
   374→            <StatusIcon className="h-5 w-5" />
   375→            <div>
   376→              <p className="text-sm font-semibold">Status: {statusInfo.label}</p>
   377→              <p className="text-xs opacity-80">
   378→                {tx.status === 'paid' || tx.status === 'success'
   379→                  ? 'Pembayaran telah berhasil diterima'
   380→                  : tx.status === 'pending'
   381→                    ? 'Menunggu pembayaran dari pelanggan'
   382→                    : tx.status === 'failed'
   383→                      ? 'Pembayaran gagal diproses'
   384→                      : tx.status === 'cancelled' || tx.status === 'void'
   385→                        ? 'Transaksi dibatalkan'
   386→                        : 'Dana telah dikembalikan ke pelanggan'}
   387→              </p>
   388→            </div>
   389→          </div>
   390→
   391→          {/* Amount Section */}

   687→// ── Pagination Helper ────────────────────────────────────────────────────────
   688→
   689→function getPaginationRange(current: number, total: number) {
   690→  const delta = 2;
   691→  const range: (number | 'dots')[] = [];
   692→  const left = Math.max(2, current - delta);
   693→  const right = Math.min(total - 1, current + delta);
   694→
   695→  range.push(1);
   696→  if (left > 2) range.push('dots');
   697→  for (let i = left; i <= right; i++) range.push(i);
   698→  if (right < total - 1) range.push('dots');
   699→  if (total > 1) range.push(total);
   700→
   701→  return range;
   702→}
