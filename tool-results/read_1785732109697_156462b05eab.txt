   189→   189→// ── Pagination Helper ────────────────────────────────────────────────────────
   190→   190→
   191→   191→function getPaginationRange(current: number, total: number) {
   192→   192→  const delta = 2;
   193→   193→  const range: (number | 'dots')[] = [];
   194→   194→  const left = Math.max(2, current - delta);
   195→   195→  const right = Math.min(total - 1, current + delta);
   196→   196→
   197→   197→  range.push(1);
   198→   198→  if (left > 2) range.push('dots');
   199→   199→  for (let i = left; i <= right; i++) range.push(i);
   200→   200→  if (right < total - 1) range.push('dots');
   201→   201→  if (total > 1) range.push(total);
   202→   202→
   203→   203→  return range;
   204→   204→}
