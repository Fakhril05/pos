   196→   196→
   197→   197→function buildTimeline(tx: Transaction): TimelineEvent[] {
   198→   198→  const events: TimelineEvent[] = [
   199→   199→    {
   200→   200→      label: 'Transaksi dibuat',
   201→   201→      date: tx.createdAt,
   202→   202→      icon: Receipt,
   203→   203→      color: 'text-emerald-500',
   204→   204→    },
   205→   205→  ];
   206→   206→  if (tx.paidAt) {
   207→   207→    events.push({
   208→   208→      label: 'Pembayaran diterima',
   209→   209→      date: tx.paidAt,
   210→   210→      icon: CheckCircle2,
   211→   211→      color: 'text-emerald-500',
   212→   212→    });
   213→   213→  }
   214→   214→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   215→   215→    events.push({
   216→   216→      label: 'Transaksi kedaluwarsa',
   217→   217→      date: tx.expiredAt,
   218→   218→      icon: AlertTriangle,
   219→   219→      color: 'text-gray-500',
   220→   220→    });
   221→   221→  }
   222→   222→  if (tx.status === 'failed') {
   223→   223→    events.push({
   224→   224→      label: 'Transaksi gagal',
   225→   225→      date: tx.updatedAt,
   226→   226→      icon: XCircle,
   227→   227→      color: 'text-red-500',
   228→   228→    });
   229→   229→  }
   230→   230→  if (tx.status === 'refunded') {
   231→   231→    events.push({
   232→   232→      label: 'Refund diproses',
   233→   233→      date: tx.updatedAt,
   234→   234→      icon: RotateCcw,
   235→   235→      color: 'text-blue-500',
   236→   236→    });
   237→   237→  }
   238→   238→  return events;
   239→   239→}

   263→   263→function TransactionDetailDialog({
   264→   264→  tx,
   265→   265→  open,
   266→   266→  onOpenChange,
   267→   267→}: {
   268→   268→  tx: Transaction | null;
   269→   269→  open: boolean;
   270→   270→  onOpenChange: (v: boolean) => void;
   271→   271→}) {
   272→   272→  const queryClient = useQueryClient();
   273→   273→  const [refunding, setRefunding] = useState(false);
   274→   274→
   275→   275→  const timeline = useMemo(() => (tx ? buildTimeline(tx) : []), [tx]);
   276→   276→
   277→   277→  const handleRefund = async () => {
   278→   278→    if (!tx) return;
   279→   279→    setRefunding(true);
   280→   280→    try {
   281→   281→      const res = await fetch(`/api/transactions/${tx.id}`, {
   282→   282→        method: 'POST',
   283→   283→        headers: { 'Content-Type': 'application/json' },
   284→   284→        body: JSON.stringify({ amount: tx.amount, reason: 'Permintaan refund dari dashboard' }),
   285→   285→      });
   286→   286→      if (!res.ok) {
   287→   287→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   288→   288→        throw new Error(err.error || 'Gagal memproses refund');
   289→   289→      }
   290→   290→      toast.success('Refund berhasil diajukan');
   291→   291→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   292→   292→      onOpenChange(false);
   293→   293→    } catch (e) {
   294→   294→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');
   295→   295→    } finally {
   296→   296→      setRefunding(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  const handleCopyInvoice = () => {
   301→   301→    if (tx) {
   302→   302→      navigator.clipboard.writeText(tx.invoiceNo);
   303→   303→      toast.success('Nomor invoice disalin');
   304→   304→    }
   305→   305→  };
   306→   306→
   307→   307→  if (!tx) return null;
   308→   308→
   309→   309→  const statusInfo = STATUS_MAP[tx.status] ?? STATUS_MAP.pending;
   310→   310→  const StatusIcon = statusInfo.icon;
   311→   311→
   312→   312→  return (
   313→   313→    <Dialog open={open} onOpenChange={onOpenChange}>
   314→   314→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   315→   315→        <DialogHeader>
   316→   316→          <div className="flex items-center gap-3">
   317→   317→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950/50">
   318→   318→              <Receipt className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   319→   319→            </div>
   320→   320→            <div>
   321→   321→              <DialogTitle className="text-base">Detail Transaksi</DialogTitle>
   322→   322→              <DialogDescription>
   323→   323→                <span className="flex items-center gap-1.5 font-mono text-xs">
   324→   324→                  {tx.invoiceNo}
   325→   325→                  <button
   326→   326→                    onClick={handleCopyInvoice}
   327→   327→                    className="inline-flex hover:text-foreground transition-colors"
   328→   328→                  >
   329→   329→                    <Copy className="h-3 w-3" />
   330→   330→                  </button>
   331→   331→                </span>
   332→   332→              </DialogDescription>
   333→   333→            </div>
   334→   334→          </div>
   335→   335→        </DialogHeader>
   336→   336→
   337→   337→        <div className="space-y-5">
   338→   338→          {/* Status Banner */}
   339→   339→          <div
   340→   340→            className={`flex items-center gap-3 rounded-lg border p-4 ${statusInfo.bgClass}`}
   341→   341→          >
   342→   342→            <StatusIcon className="h-5 w-5" />
   343→   343→            <div>
   344→   344→              <p className="text-sm font-semibold">Status: {statusInfo.label}</p>
   345→   345→              <p className="text-xs opacity-80">
   346→   346→                {tx.status === 'success'
   347→   347→                  ? 'Pembayaran telah berhasil diterima'
   348→   348→                  : tx.status === 'pending'
   349→   349→                    ? 'Menunggu pembayaran dari pelanggan'
   350→   350→                    : tx.status === 'failed'
   351→   351→                      ? 'Pembayaran gagal diproses'
   352→   352→                      : tx.status === 'expired'
   353→   353→                        ? 'Batas waktu pembayaran telah habis'
   354→   354→                        : 'Dana telah dikembalikan ke pelanggan'}
   355→   355→              </p>
   356→   356→            </div>
   357→   357→          </div>
   358→   358→
   359→   359→          {/* Amount Section */}
   360→   360→          <div className="rounded-lg border bg-muted/30 p-4">

   378→   378→              icon={Store}

   401→   401→            />
   402→   402→            {tx.paidAt && (
   403→   403→              <DetailItem

   538→   538→// ── Pagination Helper ────────────────────────────────────────────────────────
   539→   539→
   540→   540→function getPaginationRange(current: number, total: number) {
   541→   541→  const delta = 2;
   542→   542→  const range: (number | 'dots')[] = [];
   543→   543→  const left = Math.max(2, current - delta);
   544→   544→  const right = Math.min(total - 1, current + delta);
   545→   545→
   546→   546→  range.push(1);
   547→   547→  if (left > 2) range.push('dots');
   548→   548→  for (let i = left; i <= right; i++) range.push(i);
   549→   549→  if (right < total - 1) range.push('dots');
   550→   550→  if (total > 1) range.push(total);
   551→   551→
   552→   552→  return range;
   553→   553→}
