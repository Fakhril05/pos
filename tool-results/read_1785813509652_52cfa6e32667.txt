   124→// ── Component ─────────────────────────────────────────────────────────────
   125→
   126→export function Kasir() {
   127→  const queryClient = useQueryClient();
   128→
   129→  // State
   130→  const [search, setSearch] = useState('');
   131→  const [category, setCategory] = useState<Category>('Semua');
   132→  const [cart, setCart] = useState<CartItem[]>([]);
   133→  const [payDialogOpen, setPayDialogOpen] = useState(false);
   134→  const [selectedPayment, setSelectedPayment] = useState<PaymentMethod | null>(null);
   135→  const [customerName, setCustomerName] = useState('');
   136→  const [customerPhone, setCustomerPhone] = useState('');
   137→  const [customerEmail, setCustomerEmail] = useState('');
   138→  const [notes, setNotes] = useState('');
   139→  const [selectedMerchant, setSelectedMerchant] = useState('');
   140→  const [cartExpanded, setCartExpanded] = useState(false);
   141→  const [numpadOpen, setNumpadOpen] = useState(false);
   142→  const [discount, setDiscount] = useState(0);
   143→  const [paymentStep, setPaymentStep] = useState<'select' | 'qris' | 'processing' | 'success'>('select');
   144→  const [qrisExpiry, setQrisExpiry] = useState(300); // 5 minutes
   145→  const [qrisInvoice, setQrisInvoice] = useState('');
   146→  const autoCompleteRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   147→  const [lastTx, setLastTx] = useState<{
   148→    invoiceNo: string;
   149→    amount: number;
   150→    fee: number;
   151→    netAmount: number;
   152→    paymentMethod: string;
   153→    paymentChannel: string;
   154→    createdAt: string;
   155→    paidAt: string;
   156→    customerName: string | null;
   157→    merchantName: string;
   158→    items: CartItem[];
   159→    subtotal: number;
   160→    tax: number;
   161→    discount: number;
   162→  } | null>(null);
   163→
   164→  // Fetch merchants for dropdown
   165→  const { data: merchantsData } = useQuery({
   166→    queryKey: ['merchants-list-pos'],
   167→    queryFn: async () => {
   168→      const res = await fetch('/api/merchants?limit=100');
   169→      const json = await res.json();
   170→      return json.data as { id: string; name: string; status: string }[];
   171→    },
   172→  });
   173→
   174→  const merchants = merchantsData ?? [];
   175→
   176→  // Filtered products
   177→  const filteredProducts = useMemo(() => {
   178→    return PRODUCTS.filter((p) => {
   179→      const matchCategory = category === 'Semua' || p.category === category;
   180→      const matchSearch = p.name.toLowerCase().includes(search.toLowerCase());
   181→      return matchCategory && matchSearch;
   182→    });
   183→  }, [category, search]);
   184→
   185→  // Cart calculations
   186→  const cartTotal = useMemo(() => {
   187→    return cart.reduce((sum, item) => sum + item.product.price * item.qty, 0);
   188→  }, [cart]);
   189→
   190→  const cartCount = useMemo(() => {
   191→    return cart.reduce((sum, item) => sum + item.qty, 0);
   192→  }, [cart]);
   193→
   194→  const discountAmount = Math.round(cartTotal * (discount / 100));
   195→  const afterDiscount = cartTotal - discountAmount;
   196→  const tax = Math.round(afterDiscount * 0.11);
   197→  const totalWithTax = afterDiscount + tax;
   198→
   199→  // Add to cart
   200→  const addToCart = useCallback(
   201→    (product: Product) => {
   202→      setCart((prev) => {
   203→        const existing = prev.find((item) => item.product.id === product.id);
   204→        if (existing) {
   205→          return prev.map((item) =>
   206→            item.product.id === product.id ? { ...item, qty: item.qty + 1 } : item,
   207→          );
   208→        }
   209→        return [...prev, { product, qty: 1 }];
   210→      });
   211→      toast.success(`${product.name} ditambahkan ke keranjang`);
   212→    },
   213→    [],
   214→  );
   215→
   216→  // Update quantity
   217→  const updateQty = useCallback((productId: string, delta: number) => {
   218→    setCart((prev) =>
   219→      prev
   220→        .map((item) =>
   221→          item.product.id === productId ? { ...item, qty: Math.max(0, item.qty + delta) } : item,
   222→        )
   223→        .filter((item) => item.qty > 0),
   224→    );
   225→  }, []);
   226→
   227→  // Remove from cart
   228→  const removeFromCart = useCallback((productId: string) => {
   229→    setCart((prev) => prev.filter((item) => item.product.id !== productId));
   230→  }, []);
   231→
   232→  // Clear cart
   233→  const clearCart = useCallback(() => {
   234→    setCart([]);
   235→    setLastTx(null);
   236→  }, []);
   237→
   238→  // Payment mutation
   239→  const payMutation = useMutation({
   240→    mutationFn: async () => {
   241→      if (!selectedMerchant) throw new Error('Pilih merchant terlebih dahulu');
   242→      if (!selectedPayment) throw new Error('Pilih metode pembayaran');
   243→
   244→      const res = await fetch('/api/quick-pay', {
   245→        method: 'POST',
   246→        headers: { 'Content-Type': 'application/json' },
   247→        body: JSON.stringify({
   248→          merchantId: selectedMerchant,
   249→          amount: totalWithTax,
   250→          paymentMethod: selectedPayment.method,
   251→          paymentChannel: selectedPayment.channel,
   252→          customerName: customerName || undefined,
   253→          customerEmail: customerEmail || undefined,
   254→          customerPhone: customerPhone || undefined,
   255→          description: notes || undefined,
   256→          items: cart.map((item) => ({
   257→            name: item.product.name,
   258→            qty: item.qty,
   259→            price: item.product.price,
   260→          })),
   261→        }),
   262→      });
   263→
   264→      const json = await res.json();
   265→      if (!res.ok) throw new Error(json.error || 'Gagal memproses pembayaran');
   266→      return json.data;
   267→    },
   268→    onSuccess: (data) => {
   269→      queryClient.invalidateQueries({ queryKey: ['quick-stats'] });
   270→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   271→      toast.success('Pembayaran berhasil!');
   272→      setLastTx({
   273→        invoiceNo: data.invoiceNo,
   274→        amount: data.amount,
   275→        fee: data.fee,
   276→        netAmount: data.netAmount,
   277→        paymentMethod: data.paymentMethod,
   278→        paymentChannel: data.paymentChannel,
   279→        createdAt: data.createdAt,
   280→        paidAt: data.paidAt,
   281→        customerName: data.customerName,
   282→        merchantName: merchants.find(m => m.id === selectedMerchant)?.name ?? 'Pos Ril Kasir',
   283→        items: [...cart],
   284→        subtotal: cartTotal,
   285→        tax,
   286→        discount: discountAmount,
   287→      });
   288→    },
   289→    onError: (err: Error) => {
   290→      toast.error(err.message);
   291→    },
   292→  });
   293→
   294→  // Print receipt
   295→  const printReceipt = useCallback(() => {
   296→    if (!lastTx) return;
   297→    const html = generateReceiptHtml({
   298→      invoiceNo: lastTx.invoiceNo,
   299→      amount: lastTx.amount,
   300→      fee: lastTx.fee,
   301→      netAmount: lastTx.netAmount,
   302→      merchantName: lastTx.merchantName,
   303→      customerName: lastTx.customerName ?? 'Umum',
   304→      paymentMethod: lastTx.paymentMethod,
   305→      paymentChannel: lastTx.paymentChannel,
   306→      status: 'success',
   307→      createdAt: lastTx.createdAt,
   308→      paidAt: lastTx.paidAt,
   309→      description: `Kasir POS - ${lastTx.items.length} item`,
   310→      formattedAmount: formatRp(lastTx.amount),
   311→      formattedFee: formatRp(lastTx.fee),
   312→      formattedNetAmount: formatRp(lastTx.netAmount),
   313→      formattedCreatedAt: new Date(lastTx.createdAt).toLocaleString('id-ID'),
   314→      formattedPaidAt: lastTx.paidAt
   315→        ? new Date(lastTx.paidAt).toLocaleString('id-ID')
   316→        : null,
   317→      statusLabel: 'Berhasil',
   318→    });
   319→    const win = window.open('', '_blank');
   320→    if (win) {
   321→      win.document.write(html);
   322→      win.document.close();
   323→      win.focus();
   324→      win.print();
   325→    }
   326→  }, [lastTx]);
   327→
   328→  // QRIS countdown timer
   329→  useEffect(() => {
   330→    if (paymentStep !== 'qris') return;
   331→    const timer = setInterval(() => {
   332→      setQrisExpiry((prev) => {
   333→        if (prev <= 1) {
   334→          return 0;
   335→        }
   336→        return prev - 1;
   337→      });
   338→    }, 1000);
   339→    return () => clearInterval(timer);
   340→  }, [paymentStep]);
   341→
   342→  // QRIS auto-complete simulation (5-8 seconds)
   343→  useEffect(() => {
   344→    if (paymentStep !== 'qris') return;
   345→    const delay = 5000 + Math.random() * 3000;
   346→    autoCompleteRef.current = setTimeout(() => {
   347→      setPaymentStep('processing');
   348→      // After 2s processing, actually submit
   349→      setTimeout(() => {
   350→        payMutation.mutate();
   351→      }, 2000);
   352→    }, delay);
   353→    return () => {
   354→      if (autoCompleteRef.current) {
   355→        clearTimeout(autoCompleteRef.current);
   356→      }
   357→    };
   358→  }, [paymentStep]);
   359→
   360→  // Open pay dialog
   361→  const handleOpenPay = () => {
   362→    if (cart.length === 0) return;
   363→    setPayDialogOpen(true);
   364→  };
   365→
   366→  // Process payment
   367→  const handlePay = () => {
   368→    if (selectedPayment?.method === 'qris') {
   369→      // Show QRIS screen instead of immediately submitting
   370→      setQrisInvoice(`PR-QR-${Date.now().toString(36).toUpperCase()}`);
   371→      setQrisExpiry(300);
   372→      setPaymentStep('qris');
   373→    } else {
   374→      payMutation.mutate();
   375→    }
   376→  };
   377→
   378→  // New transaction
   379→  const handleNewTx = () => {
   380→    setCart([]);
   381→    setLastTx(null);
   382→    setPayDialogOpen(false);
   383→    setSelectedPayment(null);
   384→    setCustomerName('');
   385→    setCustomerPhone('');
   386→    setCustomerEmail('');
   387→    setNotes('');
   388→    setPaymentStep('select');
   389→    setQrisExpiry(300);
   390→    setQrisInvoice('');
   391→    setDiscount(0);
   392→    setNumpadOpen(false);
   393→  };
   394→
   395→  // Close pay dialog resets payment state
   396→  const handleClosePayDialog = () => {
   397→    setPayDialogOpen(false);
   398→    if (!lastTx) {
   399→      setSelectedPayment(null);
   400→      setPaymentStep('select');
   401→      setQrisExpiry(300);
   402→      setQrisInvoice('');
   403→      if (autoCompleteRef.current) {
   404→        clearTimeout(autoCompleteRef.current);
   405→      }
   406→    }
   407→  };
   408→
   409→  return (
   410→    <div className="bg-mesh animate-fade-in-up rounded-2xl p-4 md:p-6">
   411→      <PageHeader
   412→        title="Kasir"
   413→        description="Terminal pembayaran point-of-sale"
   414→        icon={Calculator}
   415→        iconGradient="emerald"
   416→      />
   417→
   418→      <div className="flex flex-col gap-4 lg:flex-row lg:gap-6">
   419→        {/* ── Left Panel: Product Catalog ── */}
   420→        <div className="min-w-0 flex-1">
   421→          {/* Search & Categories */}
   422→          <div className="card-elevated mb-4 rounded-xl p-4">
   423→            <div className="relative mb-3">
   424→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   425→              <Input
   426→                placeholder="Cari produk..."
   427→                value={search}
   428→                onChange={(e) => setSearch(e.target.value)}
   429→                className="pl-9 shadow-inner-soft"
   430→              />
   431→            </div>
   432→            <div className="flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin">
   433→              {CATEGORIES.map((cat) => (
   434→                <button
   435→                  key={cat}
   436→                  onClick={() => setCategory(cat)}
   437→                  className={cn(
   438→                    'shrink-0 rounded-lg px-3 py-1.5 text-sm font-medium transition-all duration-200',
   439→                    category === cat
   440→                      ? 'bg-emerald-500 text-white shadow-sm'
   441→                      : 'bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground',
   442→                  )}
   443→                >
   444→                  {cat}
   445→                </button>
   446→              ))}
   447→            </div>
   448→          </div>
   449→
   450→          {/* Product Grid */}
   451→          <div className="card-elevated rounded-xl p-4">
   452→            <motion.div
   453→              className="grid grid-cols-2 gap-3 md:grid-cols-3"
   454→              initial="hidden"
   455→              animate="show"
   456→              variants={{
   457→                hidden: {},
   458→                show: { transition: { staggerChildren: 0.03 } },
   459→              }}
   460→            >
   461→              <AnimatePresence mode="popLayout">
   462→                {filteredProducts.map((product) => (
   463→                  <motion.button
   464→                    key={product.id}
   465→                    layout
   466→                    variants={{
   467→                      hidden: { opacity: 0, y: 12, scale: 0.95 },
   468→                      show: { opacity: 1, y: 0, scale: 1 },
   469→                      exit: { opacity: 0, scale: 0.9, y: -8 },
   470→                    }}
   471→                    transition={{ duration: 0.25, ease: [0.4, 0, 0.2, 1] }}
   472→                    whileTap={{ scale: 0.95 }}
   473→                    onClick={() => addToCart(product)}
   474→                    className="card-hover flex flex-col items-start gap-2 rounded-xl border border-border bg-card p-3 text-left"
   475→                  >
   476→                    <span className="text-2xl leading-none">{product.emoji}</span>
   477→                    <div className="min-w-0 flex-1">
   478→                      <p className="truncate text-sm font-medium leading-tight">{product.name}</p>
   479→                      <p className="mt-1 text-sm font-bold text-emerald-600 dark:text-emerald-400">
   480→                        {formatRp(product.price)}
   481→                      </p>
   482→                    </div>
   483→                    <div className="flex w-full items-center justify-end">
   484→                      <span className="flex h-6 w-6 items-center justify-center rounded-full bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400">
   485→                        <Plus className="h-3.5 w-3.5" />
   486→                      </span>
   487→                    </div>
   488→                  </motion.button>
   489→                ))}
   490→              </AnimatePresence>
   491→            </motion.div>
   492→
   493→            {filteredProducts.length === 0 && (
   494→              <div className="empty-state flex flex-col items-center justify-center py-16 text-center">
   495→                <Search className="mb-3 h-10 w-10 text-muted-foreground/40" />
   496→                <p className="text-sm font-medium text-muted-foreground">Produk tidak ditemukan</p>
   497→                <p className="mt-1 text-xs text-muted-foreground/60">
   498→                  Coba kata kunci lain atau ubah kategori
   499→                </p>
   500→              </div>
   501→            )}
   502→          </div>
   503→        </div>
   504→
   505→        {/* ── Right Panel: Cart + Numpad (Desktop) ── */}
   506→        <div className="hidden w-full lg:flex lg:flex-col lg:w-[380px] gap-3">
   507→          <div className="card-elevated sticky top-24 flex h-[calc(100vh-8rem)] flex-col rounded-xl">
   508→            <CartPanel
   509→              cart={cart}
   510→              cartCount={cartCount}
   511→              cartTotal={cartTotal}
   512→              tax={tax}
   513→              totalWithTax={totalWithTax}
   514→              discount={discount}
   515→              discountAmount={discountAmount}
   516→              onUpdateQty={updateQty}
   517→              onRemove={removeFromCart}
   518→              onClear={clearCart}
   519→              onPay={handleOpenPay}
   520→              onDiscountChange={setDiscount}
   521→            />
   522→          </div>
   523→
   524→          {/* Numpad Toggle */}
   525→          <button
   526→            onClick={() => setNumpadOpen(!numpadOpen)}
   527→            className={cn(
   528→              'btn-press flex items-center justify-center gap-2 rounded-xl border px-4 py-2.5 text-sm font-medium transition-all',
   529→              numpadOpen
   530→                ? 'border-emerald-500 bg-emerald-500/10 text-emerald-700 dark:border-emerald-400 dark:bg-emerald-500/15 dark:text-emerald-300'
   531→                : 'border-border bg-card hover:border-emerald-500/30 hover:bg-muted/50'
   532→            )}
   533→          >

  1246→    (key: string) => {
