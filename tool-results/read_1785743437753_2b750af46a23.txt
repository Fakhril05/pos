   118→// ── Component ─────────────────────────────────────────────────────────────
   119→
   120→export function Kasir() {
   121→  const queryClient = useQueryClient();
   122→
   123→  // State
   124→  const [search, setSearch] = useState('');
   125→  const [category, setCategory] = useState<Category>('Semua');
   126→  const [cart, setCart] = useState<CartItem[]>([]);
   127→  const [payDialogOpen, setPayDialogOpen] = useState(false);
   128→  const [selectedPayment, setSelectedPayment] = useState<PaymentMethod | null>(null);
   129→  const [customerName, setCustomerName] = useState('');
   130→  const [customerPhone, setCustomerPhone] = useState('');
   131→  const [customerEmail, setCustomerEmail] = useState('');
   132→  const [notes, setNotes] = useState('');
   133→  const [selectedMerchant, setSelectedMerchant] = useState('');
   134→  const [cartExpanded, setCartExpanded] = useState(false);
   135→  const [lastTx, setLastTx] = useState<{
   136→    invoiceNo: string;
   137→    amount: number;
   138→    fee: number;
   139→    netAmount: number;
   140→    paymentMethod: string;
   141→    paymentChannel: string;
   142→    createdAt: string;
   143→    paidAt: string;
   144→    customerName: string | null;
   145→    merchantName: string;
   146→    items: CartItem[];
   147→    subtotal: number;
   148→    tax: number;
   149→  } | null>(null);
   150→
   151→  // Fetch merchants for dropdown
   152→  const { data: merchantsData } = useQuery({
   153→    queryKey: ['merchants-list-pos'],
   154→    queryFn: async () => {
   155→      const res = await fetch('/api/merchants?limit=100');
   156→      const json = await res.json();
   157→      return json.data as { id: string; name: string; status: string }[];
   158→    },
   159→  });
   160→
   161→  const merchants = merchantsData ?? [];
   162→
   163→  // Filtered products
   164→  const filteredProducts = useMemo(() => {
   165→    return PRODUCTS.filter((p) => {
   166→      const matchCategory = category === 'Semua' || p.category === category;
   167→      const matchSearch = p.name.toLowerCase().includes(search.toLowerCase());
   168→      return matchCategory && matchSearch;
   169→    });
   170→  }, [category, search]);
   171→
   172→  // Cart calculations
   173→  const cartTotal = useMemo(() => {
   174→    return cart.reduce((sum, item) => sum + item.product.price * item.qty, 0);
   175→  }, [cart]);
   176→
   177→  const cartCount = useMemo(() => {
   178→    return cart.reduce((sum, item) => sum + item.qty, 0);
   179→  }, [cart]);
   180→
   181→  const tax = Math.round(cartTotal * 0.11);
   182→  const totalWithTax = cartTotal + tax;
   183→
   184→  // Add to cart
   185→  const addToCart = useCallback(
   186→    (product: Product) => {
   187→      setCart((prev) => {
   188→        const existing = prev.find((item) => item.product.id === product.id);
   189→        if (existing) {
   190→          return prev.map((item) =>
   191→            item.product.id === product.id ? { ...item, qty: item.qty + 1 } : item,
   192→          );
   193→        }
   194→        return [...prev, { product, qty: 1 }];
   195→      });
   196→      toast.success(`${product.name} ditambahkan ke keranjang`);
   197→    },
   198→    [],
   199→  );
   200→
   201→  // Update quantity
   202→  const updateQty = useCallback((productId: string, delta: number) => {
   203→    setCart((prev) =>
   204→      prev
   205→        .map((item) =>
   206→          item.product.id === productId ? { ...item, qty: Math.max(0, item.qty + delta) } : item,
   207→        )
   208→        .filter((item) => item.qty > 0),
   209→    );
   210→  }, []);
   211→
   212→  // Remove from cart
   213→  const removeFromCart = useCallback((productId: string) => {
   214→    setCart((prev) => prev.filter((item) => item.product.id !== productId));
   215→  }, []);
   216→
   217→  // Clear cart
   218→  const clearCart = useCallback(() => {
   219→    setCart([]);
   220→    setLastTx(null);
   221→  }, []);
   222→
   223→  // Payment mutation
   224→  const payMutation = useMutation({
   225→    mutationFn: async () => {
   226→      if (!selectedMerchant) throw new Error('Pilih merchant terlebih dahulu');
   227→      if (!selectedPayment) throw new Error('Pilih metode pembayaran');
   228→
   229→      const res = await fetch('/api/quick-pay', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({
   233→          merchantId: selectedMerchant,
   234→          amount: totalWithTax,
   235→          paymentMethod: selectedPayment.method,
   236→          paymentChannel: selectedPayment.channel,
   237→          customerName: customerName || undefined,
   238→          customerEmail: customerEmail || undefined,
   239→          customerPhone: customerPhone || undefined,
   240→          description: notes || undefined,
   241→          items: cart.map((item) => ({
   242→            name: item.product.name,
   243→            qty: item.qty,
   244→            price: item.product.price,
   245→          })),
   246→        }),
   247→      });
   248→
   249→      const json = await res.json();
   250→      if (!res.ok) throw new Error(json.error || 'Gagal memproses pembayaran');
   251→      return json.data;
   252→    },
   253→    onSuccess: (data) => {
   254→      const merchant = merchants.find((m) => m.id === selectedMerchant);
   255→      setLastTx({
   256→        invoiceNo: data.invoiceNo,
   257→        amount: data.amount,
   258→        fee: data.fee,
   259→        netAmount: data.netAmount,
   260→        paymentMethod: data.paymentMethod,
   261→        paymentChannel: data.paymentChannel,
   262→        createdAt: data.createdAt,
   263→        paidAt: data.paidAt,
   264→        customerName: data.customerName,
   265→        merchantName: merchant?.name ?? 'Unknown',
   266→        items: [...cart],
   267→        subtotal: cartTotal,
   268→        tax,
   269→      });
   270→      queryClient.invalidateQueries({ queryKey: ['quick-stats'] });
   271→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   272→      toast.success('Pembayaran berhasil!');
   273→    },
   274→    onError: (err: Error) => {
   275→      toast.error(err.message);
   276→    },
   277→  });
   278→
   279→  // Print receipt
   280→  const printReceipt = useCallback(() => {
   281→    if (!lastTx) return;
   282→    const html = generateReceiptHtml({
   283→      invoiceNo: lastTx.invoiceNo,
   284→      amount: lastTx.amount,
   285→      fee: lastTx.fee,
   286→      netAmount: lastTx.netAmount,
   287→      merchantName: lastTx.merchantName,
   288→      customerName: lastTx.customerName ?? 'Umum',
   289→      paymentMethod: lastTx.paymentMethod,
   290→      paymentChannel: lastTx.paymentChannel,
   291→      status: 'success',
   292→      createdAt: lastTx.createdAt,
   293→      paidAt: lastTx.paidAt,
   294→      description: `Kasir POS - ${lastTx.items.length} item`,
   295→      formattedAmount: formatRp(lastTx.amount),
   296→      formattedFee: formatRp(lastTx.fee),
   297→      formattedNetAmount: formatRp(lastTx.netAmount),
   298→      formattedCreatedAt: new Date(lastTx.createdAt).toLocaleString('id-ID'),
   299→      formattedPaidAt: lastTx.paidAt
   300→        ? new Date(lastTx.paidAt).toLocaleString('id-ID')
   301→        : null,
   302→      statusLabel: 'Berhasil',
   303→    });
   304→    const win = window.open('', '_blank');
   305→    if (win) {
   306→      win.document.write(html);
   307→      win.document.close();
   308→      win.focus();
   309→      win.print();
   310→    }
   311→  }, [lastTx]);
   312→
   313→  // Open pay dialog
   314→  const handleOpenPay = () => {
   315→    if (cart.length === 0) return;
   316→    setPayDialogOpen(true);
   317→  };
   318→
   319→  // Process payment
   320→  const handlePay = () => {
   321→    payMutation.mutate();
   322→  };
   323→
   324→  // New transaction
   325→  const handleNewTx = () => {
   326→    setCart([]);
   327→    setLastTx(null);
   328→    setPayDialogOpen(false);
   329→    setSelectedPayment(null);
   330→    setCustomerName('');
   331→    setCustomerPhone('');
   332→    setCustomerEmail('');
   333→    setNotes('');
   334→  };
   335→
   336→  // Close pay dialog resets payment state
   337→  const handleClosePayDialog = () => {
   338→    setPayDialogOpen(false);
   339→    if (!lastTx) {
   340→      setSelectedPayment(null);
   341→    }
   342→  };
   343→
   344→  return (
   345→    <div className="bg-mesh rounded-2xl p-4 md:p-6">
   346→      <PageHeader
   347→        title="Kasir"
   348→        description="Terminal pembayaran point-of-sale"
   349→        icon={Calculator}
   350→        iconGradient="emerald"
   351→      />
   352→
   353→      <div className="flex flex-col gap-4 lg:flex-row lg:gap-6">
   354→        {/* ── Left Panel: Product Catalog ── */}
   355→        <div className="min-w-0 flex-1">
   356→          {/* Search & Categories */}
   357→          <div className="card-elevated mb-4 rounded-xl p-4">
   358→            <div className="relative mb-3">
   359→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   360→              <Input
   361→                placeholder="Cari produk..."
   362→                value={search}
   363→                onChange={(e) => setSearch(e.target.value)}
   364→                className="pl-9"
   365→              />
   366→            </div>
   367→            <div className="flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin">
   368→              {CATEGORIES.map((cat) => (
   369→                <button
   370→                  key={cat}
   371→                  onClick={() => setCategory(cat)}
   372→                  className={cn(
   373→                    'shrink-0 rounded-lg px-3 py-1.5 text-sm font-medium transition-all duration-200',
   374→                    category === cat
   375→                      ? 'bg-emerald-500 text-white shadow-sm'
   376→                      : 'bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground',
   377→                  )}
   378→                >
   379→                  {cat}
   380→                </button>
   381→              ))}
   382→            </div>
   383→          </div>
   384→
   385→          {/* Product Grid */}
   386→          <div className="card-elevated rounded-xl p-4">
   387→            <motion.div
   388→              className="grid grid-cols-2 gap-3 md:grid-cols-3"
   389→              initial="hidden"
   390→              animate="show"
   391→              variants={{
   392→                hidden: {},
   393→                show: { transition: { staggerChildren: 0.03 } },
   394→              }}
   395→            >
   396→              <AnimatePresence mode="popLayout">
   397→                {filteredProducts.map((product) => (
   398→                  <motion.button
   399→                    key={product.id}
   400→                    layout
   401→                    variants={{
   402→                      hidden: { opacity: 0, y: 12, scale: 0.95 },
   403→                      show: { opacity: 1, y: 0, scale: 1 },
   404→                      exit: { opacity: 0, scale: 0.9, y: -8 },
   405→                    }}
   406→                    transition={{ duration: 0.25, ease: [0.4, 0, 0.2, 1] }}
   407→                    whileTap={{ scale: 0.95 }}
   408→                    onClick={() => addToCart(product)}
   409→                    className="card-hover flex flex-col items-start gap-2 rounded-xl border border-border bg-card p-3 text-left"
   410→                  >
   411→                    <span className="text-2xl leading-none">{product.emoji}</span>
   412→                    <div className="min-w-0 flex-1">
   413→                      <p className="truncate text-sm font-medium leading-tight">{product.name}</p>
   414→                      <p className="mt-1 text-sm font-bold text-emerald-600 dark:text-emerald-400">
   415→                        {formatRp(product.price)}
   416→                      </p>
   417→                    </div>
   418→                    <div className="flex w-full items-center justify-end">
   419→                      <span className="flex h-6 w-6 items-center justify-center rounded-full bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400">
   420→                        <Plus className="h-3.5 w-3.5" />
   421→                      </span>
   422→                    </div>
   423→                  </motion.button>
   424→                ))}
   425→              </AnimatePresence>
   426→            </motion.div>
   427→
   428→            {filteredProducts.length === 0 && (
   429→              <div className="empty-state flex flex-col items-center justify-center py-16 text-center">
   430→                <Search className="mb-3 h-10 w-10 text-muted-foreground/40" />
   431→                <p className="text-sm font-medium text-muted-foreground">Produk tidak ditemukan</p>
   432→                <p className="mt-1 text-xs text-muted-foreground/60">
   433→                  Coba kata kunci lain atau ubah kategori
   434→                </p>
   435→              </div>
   436→            )}
   437→          </div>
   438→        </div>
   439→
   440→        {/* ── Right Panel: Cart (Desktop) ── */}
   441→        <div className="hidden w-full lg:block lg:w-[380px]">
   442→          <div className="card-elevated sticky top-24 flex h-[calc(100vh-8rem)] flex-col rounded-xl">
   443→            <CartPanel
   444→              cart={cart}
   445→              cartCount={cartCount}
   446→              cartTotal={cartTotal}
   447→              tax={tax}
   448→              totalWithTax={totalWithTax}
   449→              onUpdateQty={updateQty}
   450→              onRemove={removeFromCart}
   451→              onClear={clearCart}
   452→              onPay={handleOpenPay}
   453→            />
   454→          </div>
   455→        </div>
   456→      </div>
   457→
   458→      {/* ── Mobile Cart (Bottom Sheet) ── */}
   459→      <div className="fixed right-0 bottom-16 left-0 z-40 lg:hidden">
   460→        <AnimatePresence mode="wait">
   461→          {cart.length > 0 && (
   462→            <motion.div
   463→              key="cart-bar"
   464→              initial={{ y: 100 }}
   465→              animate={{ y: 0 }}
   466→              exit={{ y: 100 }}
   467→              transition={{ type: 'spring', damping: 25, stiffness: 300 }}
   468→              className="mx-3 overflow-hidden rounded-t-2xl border border-border bg-card shadow-2xl"
   469→            >
   470→              {/* Collapsed bar */}
   471→              {!cartExpanded && (
   472→                <button
   473→                  onClick={() => setCartExpanded(true)}
   474→                  className="flex w-full items-center justify-between px-4 py-3"
   475→                >
   476→                  <div className="flex items-center gap-2">
   477→                    <div className="relative">
   478→                      <ShoppingBag className="h-5 w-5 text-emerald-500" />
   479→                      <span className="absolute -right-2 -top-2 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-500 text-[10px] font-bold text-white">
   480→                        {cartCount}
   481→                      </span>
   482→                    </div>
   483→                    <span className="text-sm font-medium">Keranjang</span>
   484→                  </div>
   485→                  <div className="flex items-center gap-2">
   486→                    <span className="text-sm font-bold text-emerald-600 dark:text-emerald-400">
   487→                      {formatRp(totalWithTax)}
   488→                    </span>
   489→                    <ChevronUp className="h-4 w-4 text-muted-foreground" />
   490→                  </div>
   491→                </button>
   492→              )}
   493→
   494→              {/* Expanded cart */}
   495→              <AnimatePresence>
   496→                {cartExpanded && (
   497→                  <motion.div
   498→                    initial={{ height: 0 }}
   499→                    animate={{ height: 'auto' }}
   500→                    exit={{ height: 0 }}
   501→                    transition={{ duration: 0.3, ease: [0.4, 0, 0.2, 1] }}
   502→                    className="overflow-hidden"
   503→                  >
   504→                    <div className="max-h-[60vh] overflow-y-auto scrollbar-thin">
   505→                      <CartPanel
   506→                        cart={cart}
   507→                        cartCount={cartCount}
   508→                        cartTotal={cartTotal}
   509→                        tax={tax}
   510→                        totalWithTax={totalWithTax}
   511→                        onUpdateQty={updateQty}
   512→                        onRemove={removeFromCart}
   513→                        onClear={clearCart}
   514→                        onPay={handleOpenPay}
   515→                        onCollapse={() => setCartExpanded(false)}
   516→                      />
   517→                    </div>
   518→                  </motion.div>
   519→                )}
   520→              </AnimatePresence>
   521→            </motion.div>
   522→          )}
   523→        </AnimatePresence>
   524→      </div>
   525→
   526→      {/* ── Payment Dialog ── */}
   527→      <Dialog open={payDialogOpen} onOpenChange={handleClosePayDialog}>
   528→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   529→          <AnimatePresence mode="wait">
   530→            {lastTx ? (
   531→              /* ── Success State ── */
   532→              <motion.div
   533→                key="success"
   534→                initial={{ opacity: 0, scale: 0.9 }}
   535→                animate={{ opacity: 1, scale: 1 }}
   536→                exit={{ opacity: 0, scale: 0.9 }}
   537→                transition={{ duration: 0.4, ease: [0.4, 0, 0.2, 1] }}
