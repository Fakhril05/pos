    63→// ── Helpers ──────────────────────────────────────────────────────────────────
    64→
    65→const formatRp = (amount: number) =>
    66→  new Intl.NumberFormat('id-ID', {
    67→    style: 'currency',
    68→    currency: 'IDR',
    69→    minimumFractionDigits: 0,
    70→  }).format(amount);
    71→
    72→const formatTimeAgo = (dateStr: string): string => {
    73→  const now = Date.now();
    74→  const then = new Date(dateStr).getTime();
    75→  const diffMs = now - then;
    76→  const seconds = Math.floor(diffMs / 1000);
    77→  const minutes = Math.floor(seconds / 60);
    78→  const hours = Math.floor(minutes / 60);
    79→  const days = Math.floor(hours / 24);
    80→  if (days > 0) return `${days} hari lalu`;
    81→  if (hours > 0) return `${hours} jam lalu`;
    82→  if (minutes > 0) return `${minutes} menit lalu`;
    83→  return 'Baru saja';
    84→};
    85→
    86→const formatCompact = (amount: number) => {
    87→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    88→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    89→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    90→  return String(amount);
    91→};

   342→    queryKey: ['activity-feed'],

   344→      fetch('/api/transactions?limit=8&status=success').then((r) => r.json()),

   485→// ── Live Clock Widget ───────────────────────────────────────────────────────

   487→const HARI = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];
   488→const BULAN = [
   489→  'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
   490→  'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember',
   491→];
   492→
   493→function LiveClock() {
   494→  const [now, setNow] = useState(() => new Date());
   495→
   496→  useEffect(() => {
   497→    const id = setInterval(() => setNow(new Date()), 1000);
   498→    return () => clearInterval(id);
   499→  }, []);
   500→
   501→  const time = now.toLocaleTimeString('id-ID', {
   502→    hour: '2-digit',
   503→    minute: '2-digit',
   504→    second: '2-digit',
   505→    hour12: false,
   506→  });
   507→  const date = `${HARI[now.getDay()]}, ${now.getDate()} ${BULAN[now.getMonth()]} ${now.getFullYear()}`;
   508→
   509→  return (
   510→    <div className="glass-card flex flex-col items-end rounded-xl px-4 py-2.5 text-sm">
   511→      <div className="flex items-center gap-2">
   512→        <span className="relative flex h-2 w-2">
   513→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-300 opacity-75" />
   514→          <span className="relative inline-flex h-2 w-2 rounded-full bg-emerald-400" />

   516→        <span className="text-[10px] font-medium uppercase tracking-wider text-white/70">
   517→          Live
   518→        </span>
   519→      </div>
   520→      <span className="mt-0.5 text-xl font-bold tabular-nums tracking-wide text-white/90">
   521→        {time}
   522→      </span>
   523→      <span className="mt-0.5 text-xs text-white/70">{date}</span>
   524→    </div>
   525→  );
