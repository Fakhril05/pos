    54→const formatRp = (amount: number) =>
    55→  new Intl.NumberFormat('id-ID', {
    56→    style: 'currency',
    57→    currency: 'IDR',
    58→    minimumFractionDigits: 0,
    59→  }).format(amount);
    60→
    61→const formatCompact = (amount: number) => {
    62→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    63→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    64→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    65→  return String(amount);
    66→};

    68→const formatDateShort = (dateStr: string) => {
    69→  const d = new Date(dateStr);
    70→  return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
    71→};
    72→
    73→const formatDateFull = (dateStr: string) => {
    74→  const d = new Date(dateStr);
    75→  return d.toLocaleDateString('id-ID', {
    76→    day: 'numeric',
    77→    month: 'long',
    78→    year: 'numeric',

   261→export default function Reports() {
   262→  const [preset, setPreset] = useState<DatePreset>('30');
   263→  const [customFrom, setCustomFrom] = useState('');
   264→  const [customTo, setCustomTo] = useState('');
   265→
   266→  const dateFrom = useMemo(() => {
   267→    if (preset === 'custom') return customFrom;
   268→    const d = new Date();
   269→    d.setDate(d.getDate() - parseInt(preset, 10));
   270→    return d.toISOString().split('T')[0];
   271→  }, [preset, customFrom]);
   272→
   273→  const dateTo = useMemo(() => {
   274→    if (preset === 'custom') return customTo;
   275→    return new Date().toISOString().split('T')[0];
   276→  }, [preset, customTo]);
   277→
   278→  const isCustomValid = preset !== 'custom' || (customFrom && customTo);
   279→
   280→  const { data, isLoading, error } = useQuery<ReportsData>({
   281→    queryKey: ['reports', dateFrom, dateTo],
   282→    queryFn: () =>
   283→      fetch(`/api/dashboard?reportFrom=${dateFrom}&reportTo=${dateTo}`).then(
   284→        (r) => r.json(),
   285→      ),
   286→    staleTime: 30_000,
   287→    enabled: !!isCustomValid,
   288→  });
   289→
   290→  // Fallback to empty data if API returns dashboard format
   291→  const reportData = useMemo(() => {
   292→    if (!data) return null;
   293→
   294→    // If API returns the standard dashboard format, transform it
   295→    if ('stats' in data && 'monthlyChartData' in data) {
   296→      const dash = data as Record<string, unknown>;
   297→      const stats = dash.stats as { totalRevenue: number; totalTransactions: number };
   298→      const monthly = (dash.monthlyChartData as { month: string; revenue: number; count: number }[]) || [];
   299→
   300→      // Create daily data from monthly (since we don't have daily)
   301→      const dailyRevenue: DailyRevenue[] = monthly.map((m) => ({
   302→        date: m.month,
   303→        revenue: m.revenue,
   304→        count: m.count,
   305→      }));
   306→
   307→      const paymentDist = (dash.paymentMethodDistribution as { method: string; count: number; amount: number }[]) || [];
   308→      const paymentBreakdown: PaymentMethodBreakdown[] = paymentDist.map((p) => ({
   309→        method: p.method,
   310→        count: p.count,
   311→        revenue: p.amount,
   312→      }));
   313→
   314→      const topMerchantsData = (dash.topMerchants as { id: string; name: string; totalSales: number; totalTx: number }[]) || [];
   315→      const topMerchants: TopMerchant[] = topMerchantsData.map((m) => ({
   316→        id: m.id,
   317→        name: m.name,
   318→        totalTx: m.totalTx,
   319→        totalRevenue: m.totalSales,
   320→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   321→      }));
   322→
   323→      const totalDays = Math.max(dailyRevenue.length, 1);
   324→      const bestDayEntry = [...dailyRevenue].sort((a, b) => b.revenue - a.revenue)[0];
   325→
   326→      return {
   327→        summary: {
   328→          totalRevenue: stats?.totalRevenue ?? 0,
   329→          totalTransactions: stats?.totalTransactions ?? 0,
   330→          avgDailyRevenue:
   331→            stats?.totalRevenue
   332→              ? Math.round(stats.totalRevenue / totalDays)
   333→              : 0,
   334→          bestDay: bestDayEntry?.date ?? '-',
   335→          bestDayRevenue: bestDayEntry?.revenue ?? 0,
   336→        },
   337→        dailyRevenue,
   338→        paymentMethodBreakdown: paymentBreakdown,
   339→        topMerchants,
   340→        statusDistribution: [
   341→          { status: 'success', count: Math.round((stats?.totalTransactions ?? 0) * 0.85), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.85) },
   342→          { status: 'pending', count: Math.round((stats?.totalTransactions ?? 0) * 0.08), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.08) },
   343→          { status: 'failed', count: Math.round((stats?.totalTransactions ?? 0) * 0.04), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.04) },
   344→          { status: 'expired', count: Math.round((stats?.totalTransactions ?? 0) * 0.02), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.02) },
   345→          { status: 'refunded', count: Math.round((stats?.totalTransactions ?? 0) * 0.01), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.01) },
   346→        ],
   347→      } as ReportsData;
   348→    }
   349→
   350→    return data as ReportsData;
   351→  }, [data]);
   352→
   353→  if (isLoading) return <ReportsSkeleton />;
   354→  if (error || !reportData) {
   355→    return (
   356→      <div className="flex flex-col items-center justify-center gap-4 py-20">
   357→        <p className="text-muted-foreground">
   358→          Gagal memuat data laporan. Silakan coba lagi.
   359→        </p>
   360→        <Button variant="outline" onClick={() => window.location.reload()}>
   361→          Muat Ulang
   362→        </Button>
   363→      </div>
   364→    );
   365→  }
   366→
   367→  const { summary, dailyRevenue, paymentMethodBreakdown, topMerchants, statusDistribution } =
   368→    reportData;
   369→
   370→  // ── Pie data for status distribution
   371→  const pieData = statusDistribution
   372→    .filter((s) => s.count > 0)
   373→    .map((s) => ({
   374→      name: STATUS_LABELS[s.status] || s.status,
   375→      value: s.count,
   376→      revenue: s.revenue,
   377→      fill:
   378→        statusChartConfig[s.status as keyof typeof statusChartConfig]?.color ??
   379→        PIE_COLORS[0],
   380→    }));
   381→
   382→  // ── Summary cards data
   383→  const summaryCards = [
   384→    {

   387→      icon: DollarSign,
   388→      description: `${formatDateShort(dateFrom)} – ${formatDateShort(dateTo)}`,
   389→      gradient: 'from-emerald-500 to-teal-400',
   390→    },
   391→    {
   392→      label: 'Total Transaksi',
   393→      value: summary.totalTransactions.toLocaleString('id-ID'),
   394→      icon: ArrowLeftRight,
   395→      description: 'Dalam periode ini',
   396→      gradient: 'from-green-500 to-emerald-400',
   397→    },
   398→    {
   399→      label: 'Rata-rata Harian',
   400→      value: formatRp(summary.avgDailyRevenue),
   401→      icon: TrendingUp,
   402→      description: 'Pendapatan per hari',
   403→      gradient: 'from-teal-500 to-cyan-400',
   404→    },
   405→    {
   406→      label: 'Hari Terbaik',
   407→      value: formatRp(summary.bestDayRevenue),
   408→      icon: Trophy,
   409→      description: summary.bestDay !== '-' ? formatDateFull(summary.bestDay) : '-',
   410→      gradient: 'from-lime-500 to-green-400',
   411→    },
   412→  ];
   413→
   414→  // ── Bar chart data
   415→  const barData = paymentMethodBreakdown.map((p) => ({
   416→    method: p.method,
   417→    count: p.count,
   418→    revenue: p.revenue,
   419→  }));
   420→
   421→  return (
   422→    <div className="space-y-6 bg-mesh">
   423→      <PageHeader
   424→        title="Laporan"
   425→        description="Analisis performa transaksi dan pendapatan"

   428→      />
   429→
   430→      {/* ─── Date Range Picker ─────────────────────────────────────── */}
   431→      <motion.div
   432→        initial={{ opacity: 0, y: -10 }}
   433→        animate={{ opacity: 1, y: 0 }}
   434→        transition={{ duration: 0.3 }}
   435→      >
   436→        <Card className="card-elevated">
   437→          <CardContent className="p-4">
   438→            <div className="flex flex-wrap items-center gap-3">
   439→              <div className="flex items-center gap-1 rounded-lg bg-muted p-1">
   440→                {([
   441→                  { value: '7', label: '7 Hari' },
   442→                  { value: '30', label: '30 Hari' },

   444→                  { value: 'custom', label: 'Kustom' },
   445→                ] as { value: DatePreset; label: string }[]).map((item) => (

   459→
   460→              {preset === 'custom' && (
   461→                <motion.div
   462→                  initial={{ opacity: 0, width: 0 }}
   463→                  animate={{ opacity: 1, width: 'auto' }}
   464→                  exit={{ opacity: 0, width: 0 }}

   466→                >
   467→                  <div className="relative">
   468→                    <Calendar className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   469→                    <Input

   502→
   503→      {/* ─── Summary Cards ─────────────────────────────────────────── */}
