   152→  // ── Fetch paginated payouts ──
   153→  const { data, isLoading, isError } = useQuery({
   154→    queryKey: ['payouts', page, statusFilter],
   155→    queryFn: async () => {
   156→      const params = new URLSearchParams({

   165→        data: Payout[];
   166→        pagination: { page: number; limit: number; total: number; totalPages: number };
   167→      };
   168→    },
   169→    staleTime: 15_000,
   170→  });

   183→        successRes.json(),
   184→        pendingRes.json(),
   185→      ]);

   187→        total: all.pagination?.total ?? 0,
   188→        completed: success.pagination?.total ?? 0,
   189→        pending: pending.pagination?.total ?? 0,

   205→  });

   211→        method: 'PUT',
   212→        headers: { 'Content-Type': 'application/json' },

   236→        headers: { 'Content-Type': 'application/json' },

   238→          merchantId: formMerchant,
   239→          amount: Number(formAmount),
   240→          fee: Number(formFee) || 0,
   241→          bankName: formBankName,
   242→          accountNo: formAccountNo,
   243→          accountName: formAccountName,
   244→          notes: formNotes || undefined,
   245→        }),
   246→      });

   248→        const err = await res.json();
   249→        throw new Error(err.error || 'Gagal membuat pencairan');
   250→      }
   251→      return res.json();
   252→    },
   253→    onSuccess: () => {
   254→      toast.success('Pencairan berhasil dibuat');

   256→      queryClient.invalidateQueries({ queryKey: ['payouts-stats'] });
   257→      setCreateOpen(false);

   261→      toast.error(err.message);

   266→  const pagination = data?.pagination;
   267→

   269→    setFormMerchant('');
   270→    setFormAmount('');

   272→    setFormBankName('');
   273→    setFormAccountNo('');

   310→      const end = Math.min(tp - 1, p + 1);

   312→      if (p < tp - 2) pages.push('ellipsis');
   313→      pages.push(tp);

   317→
   318→  // ─── Render ───────────────────────────────────────────────────────────

   338→          { label: 'Payout Berhasil', value: (statsData?.completed ?? 0).toLocaleString('id-ID'), icon: CheckCircle2, gradient: 'from-emerald-600 to-green-500' },
   339→          { label: 'Payout Menunggu', value: (statsData?.pending ?? 0).toLocaleString('id-ID'), icon: Clock, gradient: 'from-amber-500 to-orange-400' },
   340→        ].map((stat, i) => (

   342→            key={stat.label}

   350→                  <stat.icon className="h-5 w-5 text-white" />

   352→                <div className="min-w-0">
   353→                  <p className="text-sm text-muted-foreground truncate">{stat.label}</p>

   356→              </CardContent>

   378→                  {f.label}
   379→                </SelectItem>
   380→              ))}
   381→            </SelectContent>
   382→          </Select>
   383→        </div>
   384→        <Button
   385→          className="bg-emerald-600 hover:bg-emerald-700 text-white w-full sm:w-auto"
   386→          onClick={() => setCreateOpen(true)}
   387→        >
   388→          <Plus className="h-4 w-4 mr-2" />
   389→          Buat Pencairan
   390→        </Button>
   391→      </motion.div>
   392→
   393→      {/* ── Active filter indicator ── */}
   394→      <AnimatePresence>

   396→          <motion.div
   397→            initial={{ opacity: 0, height: 0 }}

   400→            className="flex items-center gap-2"
   401→          >
   402→            <span className="text-sm text-muted-foreground">Filter:</span>
   403→            <Badge variant="secondary" className="gap-1 pr-1">
   404→              {statusFilters.find((f) => f.value === statusFilter)?.label || statusFilter}
   405→              <button
   406→                onClick={() => handleFilterChange('')}
   407→                className="ml-1 rounded-full p-0.5 hover:bg-muted"
   408→              >
   409→                <X className="h-3 w-3" />
   410→              </button>
   411→            </Badge>
   412→          </motion.div>
   413→        )}
   414→      </AnimatePresence>
   415→
   416→      {/* ── Payouts Table ── */}
   417→      <motion.div
   418→        initial={{ opacity: 0, y: 16 }}

   420→        transition={{ duration: 0.35, delay: 0.28 }}

   422→        <Card>
   423→          <CardHeader className="pb-3">
   424→            <div className="flex items-center justify-between">
   425→              <CardTitle className="flex items-center gap-2 text-base">
   426→                <Banknote className="h-5 w-5 text-emerald-600" />
   427→                Daftar Pencairan
   428→              </CardTitle>
   429→              {pagination && (
   430→                <span className="text-xs text-muted-foreground">

   434→            </div>
   435→          </CardHeader>
   436→          <CardContent className="p-0">
   437→            {isLoading ? (

   439→                {Array.from({ length: 5 }).map((_, i) => (
   440→                  <Skeleton key={i} className="h-12 w-full" />

   442→              </div>
   443→            ) : payouts.length === 0 ? (
   444→              <div className="flex flex-col items-center justify-center py-16">
   445→                <div className="h-14 w-14 rounded-full bg-muted flex items-center justify-center mb-4">
   446→                  <Banknote className="h-7 w-7 text-muted-foreground" />

   459→                      <TableHead>Bank</TableHead>

   475→                            initial={{ opacity: 0, y: 6 }}
   476→                            animate={{ opacity: 1, y: 0 }}

   478→                            transition={{ duration: 0.2, delay: i * 0.03 }}

   482→                              <div className="flex items-center gap-2">
   483→                                <div className="h-8 w-8 rounded-lg bg-gradient-to-br from-emerald-500 to-teal-500 flex items-center justify-center text-white text-xs font-bold flex-shrink-0">
   484→                                  {po.merchant.name.charAt(0).toUpperCase()}
   485→                                </div>
   486→                                <span className="font-medium text-sm truncate max-w-[140px]">
   487→                                  {po.merchant.name}
   488→                                </span>
   489→                              </div>
   490→                            </TableCell>
   491→                            <TableCell>
   492→                              <div className="text-sm">
   493→                                <p className="font-medium">{po.bankName}</p>
   494→                                <p className="text-xs text-muted-foreground">
   495→                                  {po.accountNo} • {po.accountName}
   496→                                </p>
   497→                              </div>
   498→                            </TableCell>

   556→                  onClick={() => setPage((p) => Math.max(1, p - 1))}

   574→                  </PaginationItem>

   607→          <form onSubmit={handleSubmitCreate} className="space-y-4">

   649→                <Label htmlFor="payout-fee" className="flex items-center gap-1.5">
   650→                  Biaya
   651→                </Label>
   652→                <div className="relative">
   653→                  <span className="absolute left-3 top-1/2 -translate-y-1/2 text-sm text-muted-foreground">Rp</span>
   654→                  <Input
   655→                    id="payout-fee"
   656→                    type="number"
   657→                    placeholder="0"
   658→                    className="pl-10"
   659→                    value={formFee}
   660→                    onChange={(e) => setFormFee(e.target.value)}

   662→                  />
   663→                </div>

   667→            {/* Live net amount preview */}
   668→            {formAmount && Number(formAmount) > 0 && (

   674→                <span className="text-sm text-emerald-700 dark:text-emerald-400">Jumlah Bersih</span>

   676→                  {formatMoney(Number(formAmount) - (Number(formFee) || 0))}

   685→              <Building2 className="h-3.5 w-3.5" />
   686→              Detail Rekening Bank

   688→
   689→            <div className="space-y-2">
   690→              <Label htmlFor="bank-name" className="flex items-center gap-1.5">

   706→            <div className="grid grid-cols-2 gap-3">

   758→                disabled={createMutation.isPending}
   759→              >
