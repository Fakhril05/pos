   122→// ── Component ─────────────────────────────────────────────────────────────
   123→
   124→export function Kasir() {
   125→  const queryClient = useQueryClient();
   126→
   127→  // State
   128→  const [search, setSearch] = useState('');
   129→  const [category, setCategory] = useState<Category>('Semua');
   130→  const [cart, setCart] = useState<CartItem[]>([]);
   131→  const [payDialogOpen, setPayDialogOpen] = useState(false);
   132→  const [selectedPayment, setSelectedPayment] = useState<PaymentMethod | null>(null);
   133→  const [customerName, setCustomerName] = useState('');
   134→  const [customerPhone, setCustomerPhone] = useState('');
   135→  const [customerEmail, setCustomerEmail] = useState('');
   136→  const [notes, setNotes] = useState('');
   137→  const [selectedMerchant, setSelectedMerchant] = useState('');
   138→  const [cartExpanded, setCartExpanded] = useState(false);
   139→  const [paymentStep, setPaymentStep] = useState<'select' | 'qris' | 'processing' | 'success'>('select');
   140→  const [qrisExpiry, setQrisExpiry] = useState(300); // 5 minutes
   141→  const [qrisInvoice, setQrisInvoice] = useState('');
   142→  const autoCompleteRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   143→  const [lastTx, setLastTx] = useState<{
   144→    invoiceNo: string;
   145→    amount: number;
   146→    fee: number;
   147→    netAmount: number;
   148→    paymentMethod: string;
   149→    paymentChannel: string;
   150→    createdAt: string;
   151→    paidAt: string;
   152→    customerName: string | null;
   153→    merchantName: string;
   154→    items: CartItem[];
   155→    subtotal: number;
   156→    tax: number;
   157→  } | null>(null);
   158→
   159→  // Fetch merchants for dropdown
   160→  const { data: merchantsData } = useQuery({
   161→    queryKey: ['merchants-list-pos'],
   162→    queryFn: async () => {
   163→      const res = await fetch('/api/merchants?limit=100');
   164→      const json = await res.json();
   165→      return json.data as { id: string; name: string; status: string }[];
   166→    },
   167→  });
   168→
   169→  const merchants = merchantsData ?? [];
   170→
   171→  // Filtered products
   172→  const filteredProducts = useMemo(() => {
   173→    return PRODUCTS.filter((p) => {
   174→      const matchCategory = category === 'Semua' || p.category === category;
   175→      const matchSearch = p.name.toLowerCase().includes(search.toLowerCase());
   176→      return matchCategory && matchSearch;
   177→    });
   178→  }, [category, search]);
   179→
   180→  // Cart calculations
   181→  const cartTotal = useMemo(() => {
   182→    return cart.reduce((sum, item) => sum + item.product.price * item.qty, 0);
   183→  }, [cart]);
   184→
   185→  const cartCount = useMemo(() => {
   186→    return cart.reduce((sum, item) => sum + item.qty, 0);
   187→  }, [cart]);
   188→
   189→  const tax = Math.round(cartTotal * 0.11);
   190→  const totalWithTax = cartTotal + tax;
   191→
   192→  // Add to cart
   193→  const addToCart = useCallback(
   194→    (product: Product) => {
   195→      setCart((prev) => {
   196→        const existing = prev.find((item) => item.product.id === product.id);
   197→        if (existing) {
   198→          return prev.map((item) =>
   199→            item.product.id === product.id ? { ...item, qty: item.qty + 1 } : item,
   200→          );
   201→        }
   202→        return [...prev, { product, qty: 1 }];
   203→      });
   204→      toast.success(`${product.name} ditambahkan ke keranjang`);
   205→    },
   206→    [],
   207→  );
   208→
   209→  // Update quantity
   210→  const updateQty = useCallback((productId: string, delta: number) => {
   211→    setCart((prev) =>
   212→      prev
   213→        .map((item) =>
   214→          item.product.id === productId ? { ...item, qty: Math.max(0, item.qty + delta) } : item,
   215→        )
   216→        .filter((item) => item.qty > 0),
   217→    );
   218→  }, []);
   219→
   220→  // Remove from cart
   221→  const removeFromCart = useCallback((productId: string) => {
   222→    setCart((prev) => prev.filter((item) => item.product.id !== productId));
   223→  }, []);
   224→
   225→  // Clear cart
   226→  const clearCart = useCallback(() => {
   227→    setCart([]);
   228→    setLastTx(null);
   229→  }, []);
   230→
   231→  // Payment mutation
   232→  const payMutation = useMutation({
   233→    mutationFn: async () => {
   234→      if (!selectedMerchant) throw new Error('Pilih merchant terlebih dahulu');
   235→      if (!selectedPayment) throw new Error('Pilih metode pembayaran');
   236→
   237→      const res = await fetch('/api/quick-pay', {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          merchantId: selectedMerchant,
   242→          amount: totalWithTax,
   243→          paymentMethod: selectedPayment.method,
   244→          paymentChannel: selectedPayment.channel,
   245→          customerName: customerName || undefined,
   246→          customerEmail: customerEmail || undefined,
   247→          customerPhone: customerPhone || undefined,
   248→          description: notes || undefined,
   249→          items: cart.map((item) => ({
   250→            name: item.product.name,
   251→            qty: item.qty,
   252→            price: item.product.price,
   253→          })),
   254→        }),
   255→      });
   256→
   257→      const json = await res.json();
   258→      if (!res.ok) throw new Error(json.error || 'Gagal memproses pembayaran');
   259→      return json.data;
   260→    },
   261→    onSuccess: (data) => {
   262→      const merchant = merchants.find((m) => m.id === selectedMerchant);
   263→      setLastTx({
   264→        invoiceNo: data.invoiceNo,
   265→        amount: data.amount,
   266→        fee: data.fee,
   267→        netAmount: data.netAmount,
   268→        paymentMethod: data.paymentMethod,
   269→        paymentChannel: data.paymentChannel,
   270→        createdAt: data.createdAt,
   271→        paidAt: data.paidAt,
   272→        customerName: data.customerName,
   273→        merchantName: merchant?.name ?? 'Unknown',
   274→        items: [...cart],
   275→        subtotal: cartTotal,
   276→        tax,
   277→      });
   278→      queryClient.invalidateQueries({ queryKey: ['quick-stats'] });
   279→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   280→      toast.success('Pembayaran berhasil!');
   281→    },
   282→    onError: (err: Error) => {
   283→      toast.error(err.message);
   284→    },
   285→  });
   286→
   287→  // Print receipt
   288→  const printReceipt = useCallback(() => {
   289→    if (!lastTx) return;
   290→    const html = generateReceiptHtml({
   291→      invoiceNo: lastTx.invoiceNo,
   292→      amount: lastTx.amount,
   293→      fee: lastTx.fee,
   294→      netAmount: lastTx.netAmount,
   295→      merchantName: lastTx.merchantName,
   296→      customerName: lastTx.customerName ?? 'Umum',
   297→      paymentMethod: lastTx.paymentMethod,
   298→      paymentChannel: lastTx.paymentChannel,
   299→      status: 'success',
   300→      createdAt: lastTx.createdAt,
   301→      paidAt: lastTx.paidAt,
   302→      description: `Kasir POS - ${lastTx.items.length} item`,
   303→      formattedAmount: formatRp(lastTx.amount),
   304→      formattedFee: formatRp(lastTx.fee),
   305→      formattedNetAmount: formatRp(lastTx.netAmount),
   306→      formattedCreatedAt: new Date(lastTx.createdAt).toLocaleString('id-ID'),
   307→      formattedPaidAt: lastTx.paidAt
   308→        ? new Date(lastTx.paidAt).toLocaleString('id-ID')
   309→        : null,
   310→      statusLabel: 'Berhasil',
   311→    });
   312→    const win = window.open('', '_blank');
   313→    if (win) {
   314→      win.document.write(html);
   315→      win.document.close();
   316→      win.focus();
   317→      win.print();
   318→    }
   319→  }, [lastTx]);
   320→
   321→  // QRIS countdown timer
   322→  useEffect(() => {
   323→    if (paymentStep !== 'qris') return;
   324→    const timer = setInterval(() => {
   325→      setQrisExpiry((prev) => {
   326→        if (prev <= 1) {
   327→          return 0;
   328→        }
   329→        return prev - 1;
   330→      });
   331→    }, 1000);
   332→    return () => clearInterval(timer);
   333→  }, [paymentStep]);
   334→
   335→  // QRIS auto-complete simulation (5-8 seconds)
   336→  useEffect(() => {
   337→    if (paymentStep !== 'qris') return;
   338→    const delay = 5000 + Math.random() * 3000;
   339→    autoCompleteRef.current = setTimeout(() => {
   340→      setPaymentStep('processing');
   341→      // After 2s processing, actually submit
   342→      setTimeout(() => {
   343→        payMutation.mutate();
   344→      }, 2000);
   345→    }, delay);
   346→    return () => {
   347→      if (autoCompleteRef.current) {
   348→        clearTimeout(autoCompleteRef.current);
   349→      }
   350→    };
   351→  }, [paymentStep]);
   352→
   353→  // Open pay dialog
   354→  const handleOpenPay = () => {
   355→    if (cart.length === 0) return;
   356→    setPayDialogOpen(true);
   357→  };
   358→
   359→  // Process payment
   360→  const handlePay = () => {
   361→    if (selectedPayment?.method === 'qris') {
   362→      // Show QRIS screen instead of immediately submitting
   363→      setQrisInvoice(`PR-QR-${Date.now().toString(36).toUpperCase()}`);
   364→      setQrisExpiry(300);
   365→      setPaymentStep('qris');
   366→    } else {
   367→      payMutation.mutate();
   368→    }
   369→  };
   370→
   371→  // New transaction
   372→  const handleNewTx = () => {
   373→    setCart([]);
   374→    setLastTx(null);
   375→    setPayDialogOpen(false);
   376→    setSelectedPayment(null);
   377→    setCustomerName('');
   378→    setCustomerPhone('');
   379→    setCustomerEmail('');
   380→    setNotes('');
   381→    setPaymentStep('select');
   382→    setQrisExpiry(300);
   383→    setQrisInvoice('');
   384→  };
   385→
   386→  // Close pay dialog resets payment state
   387→  const handleClosePayDialog = () => {
   388→    setPayDialogOpen(false);
   389→    if (!lastTx) {
   390→      setSelectedPayment(null);
   391→      setPaymentStep('select');
   392→      setQrisExpiry(300);
   393→      setQrisInvoice('');
   394→      if (autoCompleteRef.current) {
   395→        clearTimeout(autoCompleteRef.current);
   396→      }
   397→    }
   398→  };
   399→
   400→  return (
   401→    <div className="bg-mesh rounded-2xl p-4 md:p-6">
   402→      <PageHeader
   403→        title="Kasir"
   404→        description="Terminal pembayaran point-of-sale"
   405→        icon={Calculator}
   406→        iconGradient="emerald"
   407→      />
   408→
   409→      <div className="flex flex-col gap-4 lg:flex-row lg:gap-6">
   410→        {/* ── Left Panel: Product Catalog ── */}
   411→        <div className="min-w-0 flex-1">
   412→          {/* Search & Categories */}
   413→          <div className="card-elevated mb-4 rounded-xl p-4">
   414→            <div className="relative mb-3">
   415→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   416→              <Input
   417→                placeholder="Cari produk..."
   418→                value={search}
   419→                onChange={(e) => setSearch(e.target.value)}
   420→                className="pl-9"
   421→              />
   422→            </div>
   423→            <div className="flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin">
   424→              {CATEGORIES.map((cat) => (
   425→                <button
   426→                  key={cat}
   427→                  onClick={() => setCategory(cat)}
   428→                  className={cn(
   429→                    'shrink-0 rounded-lg px-3 py-1.5 text-sm font-medium transition-all duration-200',
   430→                    category === cat
   431→                      ? 'bg-emerald-500 text-white shadow-sm'
   432→                      : 'bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground',
   433→                  )}
   434→                >
   435→                  {cat}
   436→                </button>
   437→              ))}
   438→            </div>
   439→          </div>
   440→
   441→          {/* Product Grid */}
   442→          <div className="card-elevated rounded-xl p-4">
   443→            <motion.div
   444→              className="grid grid-cols-2 gap-3 md:grid-cols-3"
   445→              initial="hidden"
   446→              animate="show"
   447→              variants={{
   448→                hidden: {},
   449→                show: { transition: { staggerChildren: 0.03 } },
   450→              }}
   451→            >
   452→              <AnimatePresence mode="popLayout">
   453→                {filteredProducts.map((product) => (
   454→                  <motion.button
   455→                    key={product.id}
   456→                    layout
   457→                    variants={{
   458→                      hidden: { opacity: 0, y: 12, scale: 0.95 },
   459→                      show: { opacity: 1, y: 0, scale: 1 },
   460→                      exit: { opacity: 0, scale: 0.9, y: -8 },
   461→                    }}
   462→                    transition={{ duration: 0.25, ease: [0.4, 0, 0.2, 1] }}
   463→                    whileTap={{ scale: 0.95 }}
   464→                    onClick={() => addToCart(product)}
   465→                    className="card-hover flex flex-col items-start gap-2 rounded-xl border border-border bg-card p-3 text-left"
   466→                  >
   467→                    <span className="text-2xl leading-none">{product.emoji}</span>
   468→                    <div className="min-w-0 flex-1">
   469→                      <p className="truncate text-sm font-medium leading-tight">{product.name}</p>
   470→                      <p className="mt-1 text-sm font-bold text-emerald-600 dark:text-emerald-400">
   471→                        {formatRp(product.price)}
   472→                      </p>
   473→                    </div>
   474→                    <div className="flex w-full items-center justify-end">
   475→                      <span className="flex h-6 w-6 items-center justify-center rounded-full bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400">
   476→                        <Plus className="h-3.5 w-3.5" />
   477→                      </span>
   478→                    </div>
   479→                  </motion.button>
   480→                ))}
   481→              </AnimatePresence>
   482→            </motion.div>
   483→
   484→            {filteredProducts.length === 0 && (
   485→              <div className="empty-state flex flex-col items-center justify-center py-16 text-center">
   486→                <Search className="mb-3 h-10 w-10 text-muted-foreground/40" />
   487→                <p className="text-sm font-medium text-muted-foreground">Produk tidak ditemukan</p>
   488→                <p className="mt-1 text-xs text-muted-foreground/60">
   489→                  Coba kata kunci lain atau ubah kategori
   490→                </p>
   491→              </div>
   492→            )}
   493→          </div>
   494→        </div>
   495→
   496→        {/* ── Right Panel: Cart (Desktop) ── */}
   497→        <div className="hidden w-full lg:block lg:w-[380px]">
   498→          <div className="card-elevated sticky top-24 flex h-[calc(100vh-8rem)] flex-col rounded-xl">
   499→            <CartPanel
   500→              cart={cart}
   501→              cartCount={cartCount}
   502→              cartTotal={cartTotal}
   503→              tax={tax}
   504→              totalWithTax={totalWithTax}
   505→              onUpdateQty={updateQty}
   506→              onRemove={removeFromCart}
   507→              onClear={clearCart}
   508→              onPay={handleOpenPay}
   509→            />
   510→          </div>
   511→        </div>
   512→      </div>
   513→
   514→      {/* ── Mobile Cart (Bottom Sheet) ── */}
   515→      <div className="fixed right-0 bottom-16 left-0 z-40 lg:hidden">
   516→        <AnimatePresence mode="wait">
   517→          {cart.length > 0 && (
   518→            <motion.div
   519→              key="cart-bar"
   520→              initial={{ y: 100 }}
   521→              animate={{ y: 0 }}
   522→              exit={{ y: 100 }}
   523→              transition={{ type: 'spring', damping: 25, stiffness: 300 }}
   524→              className="mx-3 overflow-hidden rounded-t-2xl border border-border bg-card shadow-2xl"
   525→            >
   526→              {/* Collapsed bar */}
   527→              {!cartExpanded && (
   528→                <button
   529→                  onClick={() => setCartExpanded(true)}
   530→                  className="flex w-full items-center justify-between px-4 py-3"
   531→                >
   532→                  <div className="flex items-center gap-2">
   533→                    <div className="relative">
   534→                      <ShoppingBag className="h-5 w-5 text-emerald-500" />
   535→                      <span className="absolute -right-2 -top-2 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-500 text-[10px] font-bold text-white">
   536→                        {cartCount}
   537→                      </span>
