   122→// ── Component ─────────────────────────────────────────────────────────────
   123→
   124→export function Kasir() {
   125→  const queryClient = useQueryClient();
   126→
   127→  // State
   128→  const [search, setSearch] = useState('');
   129→  const [category, setCategory] = useState<Category>('Semua');
   130→  const [cart, setCart] = useState<CartItem[]>([]);
   131→  const [payDialogOpen, setPayDialogOpen] = useState(false);
   132→  const [selectedPayment, setSelectedPayment] = useState<PaymentMethod | null>(null);
   133→  const [customerName, setCustomerName] = useState('');
   134→  const [customerPhone, setCustomerPhone] = useState('');
   135→  const [customerEmail, setCustomerEmail] = useState('');
   136→  const [notes, setNotes] = useState('');
   137→  const [selectedMerchant, setSelectedMerchant] = useState('');
   138→  const [cartExpanded, setCartExpanded] = useState(false);
   139→  const [paymentStep, setPaymentStep] = useState<'select' | 'qris' | 'processing' | 'success'>('select');
   140→  const [qrisExpiry, setQrisExpiry] = useState(300); // 5 minutes
   141→  const [qrisInvoice, setQrisInvoice] = useState('');
   142→  const autoCompleteRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   143→  const [lastTx, setLastTx] = useState<{
   144→    invoiceNo: string;
   145→    amount: number;
   146→    fee: number;
   147→    netAmount: number;
   148→    paymentMethod: string;
   149→    paymentChannel: string;
   150→    createdAt: string;
   151→    paidAt: string;
   152→    customerName: string | null;
   153→    merchantName: string;
   154→    items: CartItem[];
   155→    subtotal: number;
   156→    tax: number;
   157→  } | null>(null);
   158→
   159→  // Fetch merchants for dropdown
   160→  const { data: merchantsData } = useQuery({
   161→    queryKey: ['merchants-list-pos'],
   162→    queryFn: async () => {
   163→      const res = await fetch('/api/merchants?limit=100');
   164→      const json = await res.json();
   165→      return json.data as { id: string; name: string; status: string }[];
   166→    },
   167→  });
   168→
   169→  const merchants = merchantsData ?? [];
   170→
   171→  // Filtered products
   172→  const filteredProducts = useMemo(() => {
   173→    return PRODUCTS.filter((p) => {
   174→      const matchCategory = category === 'Semua' || p.category === category;
   175→      const matchSearch = p.name.toLowerCase().includes(search.toLowerCase());
   176→      return matchCategory && matchSearch;
   177→    });
   178→  }, [category, search]);
   179→
   180→  // Cart calculations
   181→  const cartTotal = useMemo(() => {
   182→    return cart.reduce((sum, item) => sum + item.product.price * item.qty, 0);
   183→  }, [cart]);
   184→
   185→  const cartCount = useMemo(() => {
   186→    return cart.reduce((sum, item) => sum + item.qty, 0);
   187→  }, [cart]);
   188→
   189→  const tax = Math.round(cartTotal * 0.11);
   190→  const totalWithTax = cartTotal + tax;
   191→
   192→  // Add to cart
   193→  const addToCart = useCallback(
   194→    (product: Product) => {
   195→      setCart((prev) => {
   196→        const existing = prev.find((item) => item.product.id === product.id);
   197→        if (existing) {
   198→          return prev.map((item) =>
   199→            item.product.id === product.id ? { ...item, qty: item.qty + 1 } : item,
   200→          );
   201→        }
   202→        return [...prev, { product, qty: 1 }];
   203→      });
   204→      toast.success(`${product.name} ditambahkan ke keranjang`);
   205→    },
   206→    [],
   207→  );
   208→
   209→  // Update quantity
   210→  const updateQty = useCallback((productId: string, delta: number) => {
   211→    setCart((prev) =>
   212→      prev
   213→        .map((item) =>
   214→          item.product.id === productId ? { ...item, qty: Math.max(0, item.qty + delta) } : item,
   215→        )
   216→        .filter((item) => item.qty > 0),
   217→    );
   218→  }, []);
   219→
   220→  // Remove from cart
   221→  const removeFromCart = useCallback((productId: string) => {
   222→    setCart((prev) => prev.filter((item) => item.product.id !== productId));
   223→  }, []);
   224→
   225→  // Clear cart
   226→  const clearCart = useCallback(() => {
   227→    setCart([]);
   228→    setLastTx(null);
   229→  }, []);
   230→
   231→  // Payment mutation
   232→  const payMutation = useMutation({
   233→    mutationFn: async () => {
   234→      if (!selectedMerchant) throw new Error('Pilih merchant terlebih dahulu');
   235→      if (!selectedPayment) throw new Error('Pilih metode pembayaran');
   236→
   237→      const res = await fetch('/api/quick-pay', {
   238→        method: 'POST',
   239→        headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({
   241→          merchantId: selectedMerchant,
   242→          amount: totalWithTax,
   243→          paymentMethod: selectedPayment.method,
   244→          paymentChannel: selectedPayment.channel,
   245→          customerName: customerName || undefined,
   246→          customerEmail: customerEmail || undefined,
   247→          customerPhone: customerPhone || undefined,
   248→          description: notes || undefined,
   249→          items: cart.map((item) => ({
   250→            name: item.product.name,
   251→            qty: item.qty,
   252→            price: item.product.price,
   253→          })),
   254→        }),
   255→      });
   256→
   257→      const json = await res.json();
   258→      if (!res.ok) throw new Error(json.error || 'Gagal memproses pembayaran');
   259→      return json.data;
   260→    },
   261→    onSuccess: (data) => {
   262→      queryClient.invalidateQueries({ queryKey: ['quick-stats'] });
   263→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   264→      toast.success('Pembayaran berhasil!');
   265→      setLastTx({
   266→        invoiceNo: data.invoiceNo,
   267→        amount: data.amount,
   268→        fee: data.fee,
   269→        netAmount: data.netAmount,
   270→        paymentMethod: data.paymentMethod,
   271→        paymentChannel: data.paymentChannel,
   272→        createdAt: data.createdAt,
   273→        paidAt: data.paidAt,
   274→        customerName: data.customerName,
   275→        merchantName: merchants.find(m => m.id === selectedMerchant)?.name ?? 'Pos Ril Kasir',
   276→        items: [...cart],
   277→        subtotal: cartTotal,
   278→        tax,
   279→      });
   280→    },
   281→    onError: (err: Error) => {
   282→      toast.error(err.message);
   283→    },
   284→  });
   285→
   286→  // Print receipt
   287→  const printReceipt = useCallback(() => {
   288→    if (!lastTx) return;
   289→    const html = generateReceiptHtml({
   290→      invoiceNo: lastTx.invoiceNo,
   291→      amount: lastTx.amount,
   292→      fee: lastTx.fee,
   293→      netAmount: lastTx.netAmount,
   294→      merchantName: lastTx.merchantName,
   295→      customerName: lastTx.customerName ?? 'Umum',
   296→      paymentMethod: lastTx.paymentMethod,
   297→      paymentChannel: lastTx.paymentChannel,
   298→      status: 'success',
   299→      createdAt: lastTx.createdAt,
   300→      paidAt: lastTx.paidAt,
   301→      description: `Kasir POS - ${lastTx.items.length} item`,
   302→      formattedAmount: formatRp(lastTx.amount),
   303→      formattedFee: formatRp(lastTx.fee),
   304→      formattedNetAmount: formatRp(lastTx.netAmount),
   305→      formattedCreatedAt: new Date(lastTx.createdAt).toLocaleString('id-ID'),
   306→      formattedPaidAt: lastTx.paidAt
   307→        ? new Date(lastTx.paidAt).toLocaleString('id-ID')
   308→        : null,
   309→      statusLabel: 'Berhasil',
   310→    });
   311→    const win = window.open('', '_blank');
   312→    if (win) {
   313→      win.document.write(html);
   314→      win.document.close();
   315→      win.focus();
   316→      win.print();
   317→    }
   318→  }, [lastTx]);
   319→
   320→  // QRIS countdown timer
   321→  useEffect(() => {
   322→    if (paymentStep !== 'qris') return;
   323→    const timer = setInterval(() => {
   324→      setQrisExpiry((prev) => {
   325→        if (prev <= 1) {
   326→          return 0;
   327→        }
   328→        return prev - 1;
   329→      });
   330→    }, 1000);
   331→    return () => clearInterval(timer);
   332→  }, [paymentStep]);
   333→
   334→  // QRIS auto-complete simulation (5-8 seconds)
   335→  useEffect(() => {
   336→    if (paymentStep !== 'qris') return;
   337→    const delay = 5000 + Math.random() * 3000;
   338→    autoCompleteRef.current = setTimeout(() => {
   339→      setPaymentStep('processing');
   340→      // After 2s processing, actually submit
   341→      setTimeout(() => {
   342→        payMutation.mutate();
   343→      }, 2000);
   344→    }, delay);
   345→    return () => {
   346→      if (autoCompleteRef.current) {
   347→        clearTimeout(autoCompleteRef.current);
   348→      }
   349→    };
   350→  }, [paymentStep]);
   351→
   352→  // Open pay dialog
   353→  const handleOpenPay = () => {
   354→    if (cart.length === 0) return;
   355→    setPayDialogOpen(true);
   356→  };
   357→
   358→  // Process payment
   359→  const handlePay = () => {
   360→    if (selectedPayment?.method === 'qris') {
   361→      // Show QRIS screen instead of immediately submitting
   362→      setQrisInvoice(`PR-QR-${Date.now().toString(36).toUpperCase()}`);
   363→      setQrisExpiry(300);
   364→      setPaymentStep('qris');
   365→    } else {
   366→      payMutation.mutate();
   367→    }
   368→  };
   369→
   370→  // New transaction
   371→  const handleNewTx = () => {
   372→    setCart([]);
   373→    setLastTx(null);
   374→    setPayDialogOpen(false);
   375→    setSelectedPayment(null);
   376→    setCustomerName('');
   377→    setCustomerPhone('');
   378→    setCustomerEmail('');
   379→    setNotes('');
   380→    setPaymentStep('select');
   381→    setQrisExpiry(300);
   382→    setQrisInvoice('');
   383→  };
   384→
   385→  // Close pay dialog resets payment state
   386→  const handleClosePayDialog = () => {
   387→    setPayDialogOpen(false);
   388→    if (!lastTx) {
   389→      setSelectedPayment(null);
   390→      setPaymentStep('select');
   391→      setQrisExpiry(300);
   392→      setQrisInvoice('');
   393→      if (autoCompleteRef.current) {
   394→        clearTimeout(autoCompleteRef.current);
   395→      }
   396→    }
   397→  };
   398→
   399→  return (
   400→    <div className="bg-mesh rounded-2xl p-4 md:p-6">
   401→      <PageHeader
   402→        title="Kasir"
   403→        description="Terminal pembayaran point-of-sale"
   404→        icon={Calculator}
   405→        iconGradient="emerald"
   406→      />
   407→
   408→      <div className="flex flex-col gap-4 lg:flex-row lg:gap-6">
   409→        {/* ── Left Panel: Product Catalog ── */}
   410→        <div className="min-w-0 flex-1">
   411→          {/* Search & Categories */}
   412→          <div className="card-elevated mb-4 rounded-xl p-4">
   413→            <div className="relative mb-3">
   414→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   415→              <Input
   416→                placeholder="Cari produk..."
   417→                value={search}
   418→                onChange={(e) => setSearch(e.target.value)}
   419→                className="pl-9"
   420→              />
   421→            </div>
   422→            <div className="flex gap-1.5 overflow-x-auto pb-1 scrollbar-thin">
   423→              {CATEGORIES.map((cat) => (
   424→                <button
   425→                  key={cat}
   426→                  onClick={() => setCategory(cat)}
   427→                  className={cn(
   428→                    'shrink-0 rounded-lg px-3 py-1.5 text-sm font-medium transition-all duration-200',
   429→                    category === cat
   430→                      ? 'bg-emerald-500 text-white shadow-sm'
   431→                      : 'bg-muted text-muted-foreground hover:bg-muted/80 hover:text-foreground',
   432→                  )}
   433→                >
   434→                  {cat}
   435→                </button>
   436→              ))}
   437→            </div>
   438→          </div>
   439→
   440→          {/* Product Grid */}
   441→          <div className="card-elevated rounded-xl p-4">
   442→            <motion.div
   443→              className="grid grid-cols-2 gap-3 md:grid-cols-3"
   444→              initial="hidden"
   445→              animate="show"
   446→              variants={{
   447→                hidden: {},
   448→                show: { transition: { staggerChildren: 0.03 } },
   449→              }}
   450→            >
   451→              <AnimatePresence mode="popLayout">
   452→                {filteredProducts.map((product) => (
   453→                  <motion.button
   454→                    key={product.id}
   455→                    layout
   456→                    variants={{
   457→                      hidden: { opacity: 0, y: 12, scale: 0.95 },
   458→                      show: { opacity: 1, y: 0, scale: 1 },
   459→                      exit: { opacity: 0, scale: 0.9, y: -8 },
   460→                    }}
   461→                    transition={{ duration: 0.25, ease: [0.4, 0, 0.2, 1] }}
   462→                    whileTap={{ scale: 0.95 }}
   463→                    onClick={() => addToCart(product)}
   464→                    className="card-hover flex flex-col items-start gap-2 rounded-xl border border-border bg-card p-3 text-left"
   465→                  >
   466→                    <span className="text-2xl leading-none">{product.emoji}</span>
   467→                    <div className="min-w-0 flex-1">
   468→                      <p className="truncate text-sm font-medium leading-tight">{product.name}</p>
   469→                      <p className="mt-1 text-sm font-bold text-emerald-600 dark:text-emerald-400">
   470→                        {formatRp(product.price)}
   471→                      </p>
   472→                    </div>
   473→                    <div className="flex w-full items-center justify-end">
   474→                      <span className="flex h-6 w-6 items-center justify-center rounded-full bg-emerald-500/10 text-emerald-600 dark:bg-emerald-500/20 dark:text-emerald-400">
   475→                        <Plus className="h-3.5 w-3.5" />
   476→                      </span>
   477→                    </div>
   478→                  </motion.button>
   479→                ))}
   480→              </AnimatePresence>
   481→            </motion.div>
   482→
   483→            {filteredProducts.length === 0 && (
   484→              <div className="empty-state flex flex-col items-center justify-center py-16 text-center">
   485→                <Search className="mb-3 h-10 w-10 text-muted-foreground/40" />
   486→                <p className="text-sm font-medium text-muted-foreground">Produk tidak ditemukan</p>
   487→                <p className="mt-1 text-xs text-muted-foreground/60">
   488→                  Coba kata kunci lain atau ubah kategori
   489→                </p>
   490→              </div>
   491→            )}
   492→          </div>
   493→        </div>
   494→
   495→        {/* ── Right Panel: Cart (Desktop) ── */}
   496→        <div className="hidden w-full lg:block lg:w-[380px]">
   497→          <div className="card-elevated sticky top-24 flex h-[calc(100vh-8rem)] flex-col rounded-xl">
   498→            <CartPanel
   499→              cart={cart}
   500→              cartCount={cartCount}
   501→              cartTotal={cartTotal}
   502→              tax={tax}
   503→              totalWithTax={totalWithTax}
   504→              onUpdateQty={updateQty}
   505→              onRemove={removeFromCart}
   506→              onClear={clearCart}
   507→              onPay={handleOpenPay}
   508→            />
   509→          </div>
   510→        </div>
   511→      </div>
   512→
   513→      {/* ── Mobile Cart (Bottom Sheet) ── */}
   514→      <div className="fixed right-0 bottom-16 left-0 z-40 lg:hidden">
   515→        <AnimatePresence mode="wait">
   516→          {cart.length > 0 && (
   517→            <motion.div
   518→              key="cart-bar"
   519→              initial={{ y: 100 }}
   520→              animate={{ y: 0 }}
   521→              exit={{ y: 100 }}
   522→              transition={{ type: 'spring', damping: 25, stiffness: 300 }}
   523→              className="mx-3 overflow-hidden rounded-t-2xl border border-border bg-card shadow-2xl"
   524→            >
   525→              {/* Collapsed bar */}
   526→              {!cartExpanded && (
   527→                <button
   528→                  onClick={() => setCartExpanded(true)}
   529→                  className="flex w-full items-center justify-between px-4 py-3"
   530→                >
   531→                  <div className="flex items-center gap-2">
   532→                    <div className="relative">
   533→                      <ShoppingBag className="h-5 w-5 text-emerald-500" />
   534→                      <span className="absolute -right-2 -top-2 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-500 text-[10px] font-bold text-white">
   535→                        {cartCount}
   536→                      </span>
   537→                    </div>
