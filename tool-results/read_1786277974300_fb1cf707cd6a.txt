    69→const formatRp = (amount: number) =>
    70→  new Intl.NumberFormat('id-ID', {
    71→    style: 'currency',
    72→    currency: 'IDR',
    73→    minimumFractionDigits: 0,
    74→  }).format(amount);
    75→
    76→const formatCompact = (amount: number) => {
    77→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    78→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    79→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    80→  return String(amount);
    81→};
    82→
    83→const formatDateShort = (dateStr: string) => {
    84→  const d = new Date(dateStr);
    85→  return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
    86→};
    87→
    88→const formatDateFull = (dateStr: string) => {
    89→  const d = new Date(dateStr);
    90→  return d.toLocaleDateString('id-ID', {
    91→    day: 'numeric',
    92→    month: 'long',
    93→    year: 'numeric',
    94→  });
    95→};
    96→
    97→const escCsv = (v: string | null | undefined) => {
    98→  if (!v) return '';
    99→  return `"${v.replace(/"/g, '""')}"`;
   100→};
   101→
   102→const downloadBlob = (csv: string, filename: string) => {
   103→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   104→  const url = URL.createObjectURL(blob);
   105→  const a = document.createElement('a');
   106→  a.href = url;
   107→  a.download = filename;
   108→  a.click();
   109→  URL.revokeObjectURL(url);
   110→};

   341→export default function Reports() {
   342→  const [preset, setPreset] = useState<DatePreset>('30');
   343→  const [customFrom, setCustomFrom] = useState('');
   344→  const [customTo, setCustomTo] = useState('');
   345→  const [revenueChartType, setRevenueChartType] = useState<RevenueChartType>('area');
   346→  const [paymentChartType, setPaymentChartType] = useState<PaymentChartType>('pie');
   347→  const [exporting, setExporting] = useState(false);
   348→
   349→  const dateFrom = useMemo(() => {
   350→    if (preset === 'custom') return customFrom;
   351→    const d = new Date();
   352→    switch (preset) {
   353→      case 'today':
   354→        return d.toISOString().split('T')[0];
   355→      case '12months':
   356→        d.setFullYear(d.getFullYear() - 1);
   357→        return d.toISOString().split('T')[0];
   358→      case 'all':
   359→        return '2020-01-01';
   360→      default:
   361→        d.setDate(d.getDate() - parseInt(preset, 10));
   362→        return d.toISOString().split('T')[0];
   363→    }
   364→  }, [preset, customFrom]);
   365→
   366→  const dateTo = useMemo(() => {
   367→    if (preset === 'custom') return customTo;
   368→    return new Date().toISOString().split('T')[0];
   369→  }, [preset, customTo]);
   370→
   371→  const isCustomValid = preset !== 'custom' || (customFrom && customTo);
   372→
   373→  const { data, isLoading, error } = useQuery<ReportsData>({
   374→    queryKey: ['reports', dateFrom, dateTo],
   375→    queryFn: () =>
   376→      fetch(`/api/dashboard?reportFrom=${dateFrom}&reportTo=${dateTo}`).then(
   377→        (r) => r.json(),
   378→      ),
   379→    staleTime: 30_000,
   380→    enabled: !!isCustomValid,
   381→  });
   382→
   383→  // Fallback to empty data if API returns dashboard format
   384→  const reportData = useMemo(() => {
   385→    if (!data) return null;
   386→
   387→    // If API returns the standard dashboard format, transform it
   388→    if ('stats' in data && 'monthlyChartData' in data) {
   389→      const dash = data as Record<string, unknown>;
   390→      const stats = dash.stats as { totalRevenue: number; totalTransactions: number };
   391→      const monthly = (dash.monthlyChartData as { month: string; revenue: number; count: number }[]) || [];
   392→
   393→      // Create daily data from monthly (since we don't have daily)
   394→      const dailyRevenue: DailyRevenue[] = monthly.map((m) => ({
   395→        date: m.month,
   396→        revenue: m.revenue,
   397→        count: m.count,
   398→      }));
   399→
   400→      const paymentDist = (dash.paymentMethodDistribution as { method: string; count: number; amount: number }[]) || [];
   401→      const paymentBreakdown: PaymentMethodBreakdown[] = paymentDist.map((p) => ({
   402→        method: p.method,
   403→        count: p.count,
   404→        revenue: p.amount,
   405→      }));
   406→
   407→      const topMerchantsData = (dash.topMerchants as { id: string; name: string; totalSales: number; totalTx: number }[]) || [];
   408→      const topMerchants: TopMerchant[] = topMerchantsData.map((m) => ({
   409→        id: m.id,
   410→        name: m.name,
   411→        totalTx: m.totalTx,
   412→        totalRevenue: m.totalSales,
   413→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   414→      }));
   415→
   416→      const totalDays = Math.max(dailyRevenue.length, 1);
   417→      const bestDayEntry = [...dailyRevenue].sort((a, b) => b.revenue - a.revenue)[0];
   418→
   419→      return {
   420→        summary: {
   421→          totalRevenue: stats?.totalRevenue ?? 0,
   422→          totalTransactions: stats?.totalTransactions ?? 0,
   423→          avgDailyRevenue:
   424→            stats?.totalRevenue
   425→              ? Math.round(stats.totalRevenue / totalDays)
   426→              : 0,
   427→          bestDay: bestDayEntry?.date ?? '-',
   428→          bestDayRevenue: bestDayEntry?.revenue ?? 0,
   429→        },
   430→        dailyRevenue,
   431→        paymentMethodBreakdown: paymentBreakdown,
   432→        topMerchants,
   433→        statusDistribution: [
   434→          { status: 'success', count: Math.round((stats?.totalTransactions ?? 0) * 0.85), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.85) },
   435→          { status: 'pending', count: Math.round((stats?.totalTransactions ?? 0) * 0.08), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.08) },
   436→          { status: 'failed', count: Math.round((stats?.totalTransactions ?? 0) * 0.04), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.04) },
   437→          { status: 'expired', count: Math.round((stats?.totalTransactions ?? 0) * 0.02), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.02) },
   438→          { status: 'refunded', count: Math.round((stats?.totalTransactions ?? 0) * 0.01), revenue: Math.round((stats?.totalRevenue ?? 0) * 0.01) },
   439→        ],
   440→      } as ReportsData;
   441→    }
   442→
   443→    // Handle dashboard report format (may have topTenants instead of topMerchants)
   444→    const raw = data as Record<string, unknown>;
   445→    const topTenants = raw.topTenants as { id: string; name: string; totalSales: number; totalTx: number }[] | undefined;
   446→    const mapped: ReportsData = {
   447→      summary: raw.summary as ReportsData['summary'],
   448→      dailyRevenue: (raw.dailyRevenue as DailyRevenue[]) ?? [],
   449→      paymentMethodBreakdown: (raw.paymentMethodBreakdown as PaymentMethodBreakdown[]) ?? [],
   450→      topMerchants: (topTenants ?? []).map((m) => ({
   451→        id: m.id,
   452→        name: m.name,
   453→        totalTx: m.totalTx,
   454→        totalRevenue: m.totalSales,
   455→        avgTxSize: m.totalTx > 0 ? m.totalSales / m.totalTx : 0,
   456→      })),
   457→      statusDistribution: (raw.statusDistribution as StatusDistribution[]) ?? [],
   458→    };
   459→    return mapped;
   460→  }, [data]);
   461→
   462→  // ── CSV Export handlers ───────────────────────────────────────────────────
   463→
   464→  const handleExportTransactions = useCallback(async () => {
   465→    setExporting(true);
   466→    const loadingToast = toast.loading('Mengunduh data transaksi...');
   467→    try {
   468→      const res = await fetch(
   469→        `/api/transactions?startDate=${dateFrom}&endDate=${dateTo}&limit=1000`,
   470→      );
   471→      if (!res.ok) throw new Error('Gagal mengambil data');
   472→      const json = await res.json();
   473→      const transactions = json.data || json;
   474→
   475→      const header = 'Invoice No,Merchant,Pelanggan,Jumlah,Fee,Net Amount,Metode,Channel,Status,Tanggal Bayar,Tanggal Buat';
   476→      const rows = transactions.map((tx: Record<string, unknown>) => {
   477→        const merchant = tx.merchant as Record<string, string> | null;
   478→        return [
   479→          escCsv(tx.invoiceNo as string),
   480→          escCsv(merchant?.name ?? (tx.merchantName as string)),
   481→          escCsv(tx.customerName as string),
   482→          tx.amount ?? 0,
   483→          tx.fee ?? 0,
   484→          tx.netAmount ?? 0,
   485→          escCsv(tx.paymentMethod as string),
   486→          escCsv(tx.paymentChannel as string),
   487→          escCsv(tx.status as string),
   488→          tx.paidAt ? new Date(tx.paidAt as string).toLocaleString('id-ID') : '',
   489→          tx.createdAt ? new Date(tx.createdAt as string).toLocaleString('id-ID') : '',
   490→        ].join(',');
   491→      });
   492→
   493→      const csv = [header, ...rows].join('\n');
   494→      downloadBlob(csv, `transaksi-${dateFrom}-sd-${dateTo}.csv`);
   495→      toast.success('Ekspor CSV transaksi berhasil!', { id: loadingToast });
   496→    } catch {
   497→      toast.error('Gagal mengekspor transaksi. Silakan coba lagi.', { id: loadingToast });
   498→    } finally {
   499→      setExporting(false);
   500→    }
   501→  }, [dateFrom, dateTo]);
   502→
   503→  const handleExportSummary = useCallback(() => {
   504→    if (!reportData) return;
   505→    setExporting(true);
   506→    const loadingToast = toast.loading('Membuat ringkasan laporan...');
   507→    try {
   508→      const header = 'Bulan,Pendapatan,Jumlah Transaksi';
   509→      const rows = reportData.dailyRevenue.map((d) => {
   510→        return [
   511→          escCsv(d.date),
   512→          d.revenue,
   513→          d.count,
   514→        ].join(',');
   515→      });
   516→
   517→      const csv = [header, ...rows].join('\n');
   518→      downloadBlob(csv, `ringkasan-laporan-${dateFrom}-sd-${dateTo}.csv`);
   519→      toast.success('Ekspor CSV ringkasan berhasil!', { id: loadingToast });
   520→    } catch {
   521→      toast.error('Gagal mengekspor ringkasan. Silakan coba lagi.', { id: loadingToast });
   522→    } finally {
   523→      setExporting(false);
   524→    }
   525→  }, [reportData, dateFrom, dateTo]);
   526→
   527→  if (isLoading) return <ReportsSkeleton />;
   528→  if (error || !reportData) {
   529→    return (
   530→      <div className="flex flex-col items-center justify-center gap-4 py-20">
   531→        <p className="text-muted-foreground">
   532→          Gagal memuat data laporan. Silakan coba lagi.
   533→        </p>
   534→        <Button variant="outline" onClick={() => window.location.reload()}>
   535→          Muat Ulang
   536→        </Button>
   537→      </div>
   538→    );
   539→  }
   540→
   541→  const { summary, dailyRevenue, paymentMethodBreakdown, topMerchants, statusDistribution } =
   542→    reportData;
   543→
   544→  // ── Pie data for status distribution
   545→  const pieData = statusDistribution
   546→    .filter((s) => s.count > 0)
   547→    .map((s) => ({
   548→      name: STATUS_LABELS[s.status] || s.status,
   549→      value: s.count,
   550→      revenue: s.revenue,
   551→      fill:
   552→        statusChartConfig[s.status as keyof typeof statusChartConfig]?.color ??
   553→        PIE_COLORS[0],
   554→    }));
   555→
   556→  // ── Summary cards data (existing)
   557→  const summaryCards = [
   558→    {
   559→      label: 'Total Pendapatan',
   560→      value: formatRp(summary.totalRevenue),
   561→      icon: DollarSign,
   562→      description: `${formatDateShort(dateFrom)} – ${formatDateShort(dateTo)}`,
   563→      gradient: 'from-emerald-500 to-teal-400',
   564→    },
   565→    {
