   145→
   146→export function Kasir() {
   147→  const user = useAppStore((s) => s.user)
   148→  const queryClient = useQueryClient()
   149→
   150→  const [search, setSearch] = useState('')
   151→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   152→  const [cart, setCart] = useState<CartItem[]>([])
   153→  const [selectedPayment, setSelectedPayment] = useState<(typeof PAYMENT_METHODS)[number] | null>(null)
   154→  const [tendered, setTendered] = useState('')
   155→  const [showSuccess, setShowSuccess] = useState(false)
   156→  const [lastTx, setLastTx] = useState<TxResult | null>(null)
   157→  const [showShiftDialog, setShowShiftDialog] = useState<'open' | 'close' | null>(null)
   158→  const [shiftOpeningCash, setShiftOpeningCash] = useState('')
   159→  const [shiftClosingCash, setShiftClosingCash] = useState('')
   160→  const [shiftNote, setShiftNote] = useState('')
   161→  const [activeTab, setActiveTab] = useState<'pos' | 'history'>('pos')
   162→  const [historyFilter, setHistoryFilter] = useState<'shift' | 'today' | 'all'>('shift')
   163→  const [showTxDetail, setShowTxDetail] = useState<TxHistoryItem | null>(null)
   164→  const [showMobileCart, setShowMobileCart] = useState(false)
   165→  const searchRef = useRef<HTMLInputElement>(null)
   166→
   167→  // ── Queries ──────────────────────────────────────────────────────
   168→
   169→  const { data: productsData, isLoading: productsLoading } = useQuery({
   170→    queryKey: ['cashier-products', user?.tenantId, search, activeCategory],
   171→    queryFn: async () => {
   172→      const params = new URLSearchParams({ tenantId: user!.tenantId! })
   173→      if (search) params.set('search', search)
   174→      if (activeCategory) params.set('categoryId', activeCategory)
   175→      const res = await fetch(`/api/cashier/products?${params}`)
   176→      if (!res.ok) throw new Error('Gagal memuat produk')
   177→      return res.json() as Promise<{ products: Product[]; categories: Category[] }>
   178→    },
   179→    enabled: !!user?.tenantId,
   180→    refetchInterval: 30000,
   181→  })
   182→
   183→  const { data: shiftData, isLoading: shiftLoading } = useQuery({
   184→    queryKey: ['cashier-shift', user?.outletId, user?.id],
   185→    queryFn: async () => {
   186→      const params = new URLSearchParams({
   187→        outletId: user!.outletId!,
   188→        cashierId: user!.id,
   189→      })
   190→      const res = await fetch(`/api/cashier/shifts?${params}`)
   191→      if (!res.ok) throw new Error('Gagal memuat shift')
   192→      return res.json() as Promise<{ shift: ShiftData | null }>
   193→    },
   194→    enabled: !!user?.outletId && !!user?.id,
   195→  })
   196→
   197→  const { data: historyData, isLoading: historyLoading } = useQuery({
   198→    queryKey: ['cashier-history', user?.id, historyFilter],
   199→    queryFn: async () => {
   200→      const params = new URLSearchParams({
   201→        cashierId: user!.id,
   202→        filter: historyFilter,
   203→      })
   204→      const res = await fetch(`/api/cashier/history?${params}`)
   205→      if (!res.ok) throw new Error('Gagal memuat riwayat')
   206→      return res.json() as Promise<{ transactions: TxHistoryItem[] }>
   207→    },
   208→    enabled: activeTab === 'history' && !!user?.id,
   209→  })
   210→
   211→  // ── Mutations ──────────────────────────────────────────────────
   212→
   213→  const createTransaction = useMutation({
   214→    mutationFn: async () => {
   215→      if (!user || !shiftData?.shift || !selectedPayment) return null
   216→
   217→      const subtotal = cart.reduce((sum, c) => sum + c.price * c.quantity, 0)
   218→      const discountAmount = 0
   219→      const taxAmount = Math.round(subtotal * TAX_RATE)
   220→      const serviceCharge = Math.round(subtotal * SERVICE_CHARGE_RATE)
   221→      const totalAmount = subtotal - discountAmount + taxAmount + serviceCharge
   222→
   223→      const res = await fetch('/api/cashier/transaction', {
   224→        method: 'POST',
   225→        headers: { 'Content-Type': 'application/json' },
   226→        body: JSON.stringify({
   227→          tenantId: user.tenantId,
   228→          outletId: user.outletId,
   229→          cashierId: user.id,
   230→          items: cart.map((c) => ({
   231→            productId: c.productId,
   232→            name: c.name,
   233→            price: c.price,
   234→            buyPrice: c.buyPrice,
   235→            quantity: c.quantity,
   236→            unit: c.unit,
   237→            sku: c.sku,
   238→          })),
   239→          subtotalAmount: subtotal,
   240→          discountAmount,
   241→          taxAmount,
   242→          serviceCharge,
   243→          totalAmount,
   244→          paymentMethod: selectedPayment.method,
   245→          paymentChannel: selectedPayment.channel,
   246→          tendered: selectedPayment.method === 'cash' ? parseFloat(tendered) || 0 : undefined,
   247→        }),
   248→      })
   249→      if (!res.ok) {
   250→        const err = await res.json()
   251→        throw new Error(err.error || 'Gagal membuat transaksi')
   252→      }
   253→      return res.json() as Promise<TxResult>
   254→    },
   255→    onSuccess: (data) => {
   256→      if (data) {
   257→        setLastTx(data)
   258→        setShowSuccess(true)
   259→        setCart([])
   260→        setSelectedPayment(null)
   261→        setTendered('')
   262→        queryClient.invalidateQueries({ queryKey: ['cashier-products'] })
   263→        queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   264→        queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   265→      }
   266→    },
   267→    onError: (err) => {
   268→      toast.error(err.message)
   269→    },
   270→  })
   271→
   272→  const shiftMutation = useMutation({
   273→    mutationFn: async (action: 'open' | 'close') => {
   274→      if (!user) return null
   275→      const body: Record<string, unknown> = {
   276→        action,
   277→        tenantId: user.tenantId,
   278→        outletId: user.outletId,
   279→        cashierId: user.id,
   280→      }
   281→      if (action === 'open') body.openingCash = parseFloat(shiftOpeningCash) || 0
   282→      if (action === 'close') {
   283→        body.closingCash = parseFloat(shiftClosingCash) || 0
   284→        body.note = shiftNote
   285→      }
   286→      const res = await fetch('/api/cashier/shifts', {
   287→        method: 'POST',
   288→        headers: { 'Content-Type': 'application/json' },
   289→        body: JSON.stringify(body),
   290→      })
   291→      if (!res.ok) {
   292→        const err = await res.json()
   293→        throw new Error(err.error || 'Gagal')
   294→      }
   295→      return res.json()
   296→    },
   297→    onSuccess: () => {
   298→      queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   299→      queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   300→      setShowShiftDialog(null)
   301→      setShiftOpeningCash('')
   302→      setShiftClosingCash('')
   303→      setShiftNote('')
   304→      toast.success(
   305→        shiftMutation.variables === 'open' ? 'Shift berhasil dibuka' : 'Shift berhasil ditutup',
   306→      )
   307→    },
   308→    onError: (err) => {
   309→      toast.error(err.message)
   310→    },
   311→  })
   312→
   313→  // ── Cart Logic ──────────────────────────────────────────────────
   314→
   315→  const addToCart = useCallback(
   316→    (product: Product) => {
   317→      setCart((prev) => {
   318→        const existing = prev.find((c) => c.productId === product.id)
   319→        if (existing) {
   320→          if (existing.quantity >= product.stock) {
   321→            toast.warning(`Stok ${product.name} tidak cukup`)
   322→            return prev
   323→          }
   324→          return prev.map((c) =>
   325→            c.productId === product.id
   326→              ? { ...c, quantity: c.quantity + 1 }
   327→              : c,
   328→          )
   329→        }
   330→        if (product.stock <= 0) {
   331→          toast.warning(`${product.name} habis`)
   332→          return prev
   333→        }
   334→        return [
   335→          ...prev,
   336→          {
   337→            productId: product.id,
   338→            name: product.name,
   339→            price: product.sellPrice,
   340→            buyPrice: product.buyPrice,
   341→            quantity: 1,
   342→            unit: product.unit,
   343→            sku: product.sku,
   344→          },
   345→        ]
   346→      })
   347→    },
   348→    [],
   349→  )
   350→
   351→  const updateQty = useCallback((productId: string, delta: number) => {
   352→    setCart((prev) => {
   353→      return prev
   354→        .map((c) => {
   355→          if (c.productId !== productId) return c
   356→          const newQty = c.quantity + delta
   357→          if (newQty <= 0) return null
   358→          return { ...c, quantity: newQty }
   359→        })
   360→        .filter(Boolean) as CartItem[]
   361→    })
   362→  }, [])
   363→
   364→  const removeFromCart = useCallback((productId: string) => {
   365→    setCart((prev) => prev.filter((c) => c.productId !== productId))
   366→  }, [])
   367→
   368→  const clearCart = useCallback(() => {
   369→    setCart([])
   370→    setSelectedPayment(null)
   371→    setTendered('')
   372→  }, [])
   373→
   374→  // ── Computed ──────────────────────────────────────────────────
   375→
   376→  const cartSubtotal = useMemo(() => cart.reduce((s, c) => s + c.price * c.quantity, 0), [cart])
   377→  const cartTax = Math.round(cartSubtotal * TAX_RATE)
   378→  const cartServiceCharge = Math.round(cartSubtotal * SERVICE_CHARGE_RATE)
   379→  const cartTotal = cartSubtotal + cartTax + cartServiceCharge
   380→  const cartItemCount = cart.reduce((s, c) => s + c.quantity, 0)
   381→  const tenderedNum = parseFloat(tendered) || 0
   382→  const change = tenderedNum - cartTotal
   383→
   384→  const filteredProducts = useMemo(() => {
   385→    const prods = productsData?.products || []
   386→    if (!activeCategory) return prods
   387→    return prods.filter((p) => p.category?.id === activeCategory)
   388→  }, [productsData, activeCategory])
   389→
   390→  const categories = productsData?.categories || []
   391→  const activeShift = shiftData?.shift || null
   392→  const isProcessing = createTransaction.isPending || shiftMutation.isPending
   393→
   394→  // ── Keyboard shortcuts ──────────────────────────────────────────
   395→
   396→  useEffect(() => {
   397→    const handleKey = (e: KeyboardEvent) => {
   398→      const tag = (e.target as HTMLElement)?.tagName
   399→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   400→      if (e.key === 'F2') {
   401→        e.preventDefault()
   402→        searchRef.current?.focus()
   403→      }
   404→      if (e.key === 'Escape') {
   405→        setShowSuccess(false)
   406→        setShowShiftDialog(null)
   407→        setShowTxDetail(null)
   408→      }
   409→    }
   410→    document.addEventListener('keydown', handleKey)
   411→    return () => document.removeEventListener('keydown', handleKey)
   412→  }, [])
   413→
   414→  // ── Print Receipt ──────────────────────────────────────────
   415→
   416→  const printReceipt = useCallback(() => {
   417→    if (!lastTx) return
   418→    const txSubtotal = lastTx.items.reduce((s, i) => s + i.price * i.quantity, 0)
   419→    const txTax = Math.round(txSubtotal * TAX_RATE)
   420→    const txServiceCharge = Math.round(txSubtotal * SERVICE_CHARGE_RATE)
   421→
   422→    const html = generateReceiptHtml({
   423→      invoiceNo: lastTx.invoiceNo,
   424→      tenantName: lastTx.tenantName,
   425→      outletName: lastTx.outletName,
   426→      outletAddress: lastTx.outletAddress,
   427→      outletCity: lastTx.outletCity,
   428→      tenantPhone: lastTx.tenantPhone,
   429→      cashierName: lastTx.cashierName,
   430→      items: lastTx.items.map((i) => ({
   431→        name: i.name,
   432→        quantity: i.quantity,
   433→        price: i.price,
   434→        subtotal: i.price * i.quantity,
   435→      })),
   436→      subtotalAmount: txSubtotal,
   437→      discountAmount: 0,
   438→      taxAmount: txTax,
   439→      serviceCharge: txServiceCharge,
   440→      totalAmount: lastTx.transaction.totalAmount,
   441→      paymentMethod: lastTx.payment.paymentMethod,
   442→      paymentChannel: lastTx.payment.paymentChannel,
   443→      tendered: lastTx.payment.paymentMethod === 'cash' ? parseFloat(tendered) || 0 : undefined,
   444→      change: lastTx.payment.paymentMethod === 'cash' ? change : undefined,
   445→      createdAt: lastTx.transaction.createdAt,
   446→    })
   447→
   448→    const w = window.open('', '_blank', 'width=400,height=700')
   449→    if (w) {
   450→      w.document.write(html)
   451→      w.document.close()
   452→      setTimeout(() => w.print(), 300)
   453→    }
   454→  }, [lastTx, tendered, change, cart])
   455→
   456→
   457→  // ── Render ──────────────────────────────────────────────────
   458→
   459→  return (
   460→    <div className="flex h-[calc(100vh-8.5rem)] flex-col gap-4 md:gap-6">
   461→      {/* Tab Switcher */}
   462→      <div className="flex items-center gap-1">
   463→        <button
   464→          onClick={() => setActiveTab('pos')}
   465→          className={cn(
   466→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   467→            activeTab === 'pos'
   468→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   469→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   470→          )}
   471→        >
   472→          <span className="flex items-center gap-2">
   473→            <ShoppingCart className="h-4 w-4" />
   474→            Kasir POS
   475→          </span>
   476→        </button>
   477→        <button
   478→          onClick={() => setActiveTab('history')}
   479→          className={cn(
   480→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   481→            activeTab === 'history'
   482→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   483→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   484→          )}
   485→        >
   486→          <span className="flex items-center gap-2">
   487→            <History className="h-4 w-4" />
   488→            Riwayat
   489→          </span>
   490→        </button>
   491→      </div>
   492→
   493→      {activeTab === 'pos' ? (
   494→        <div className="flex min-h-0 flex-1 flex-col gap-4 lg:flex-row">
   495→          {/* ─── LEFT PANEL: Products ─── */}
   496→          <div className="flex min-h-0 flex-1 flex-col gap-3 lg:flex-[3]">
   497→            {/* Search */}
   498→            <div className="relative">
   499→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   500→              <Input
   501→                ref={searchRef}
   502→                value={search}
   503→                onChange={(e) => setSearch(e.target.value)}
   504→                placeholder="Cari produk... (F2)"
   505→                className="pl-10"
   506→              />
   507→            </div>
   508→
   509→            {/* Category Tabs */}
   510→            <div className="relative">
   511→              <ScrollArea className="w-full">
   512→                <div className="flex gap-2 pb-1">
   513→                  <button
   514→                    onClick={() => setActiveCategory(null)}
   515→                    className={cn(
   516→                      'shrink-0 rounded-full px-4 py-1.5 text-xs font-semibold transition-all btn-press',
   517→                      !activeCategory
   518→                        ? 'bg-emerald-500 text-white shadow-md shadow-emerald-500/20'
   519→                        : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground',
   520→                    )}
   521→                  >
   522→                    Semua
   523→                  </button>
   524→                  {categories.map((cat) => (
   525→                    <button
   526→                      key={cat.id}
   527→                      onClick={() => setActiveCategory(cat.id)}
   528→                      className={cn(
   529→                        'shrink-0 rounded-full px-4 py-1.5 text-xs font-semibold transition-all btn-press',
   530→                        activeCategory === cat.id
   531→                          ? 'bg-emerald-500 text-white shadow-md shadow-emerald-500/20'
   532→                          : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground',
   533→                      )}
   534→                    >
   535→                      {cat.icon || ''} {cat.name}
   536→                    </button>
   537→                  ))}
   538→                </div>
   539→              </ScrollArea>
   540→            </div>
   541→
   542→            {/* Product Grid */}
   543→            <div className="min-h-0 flex-1">
   544→              {productsLoading ? (
   545→                <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   546→                  {Array.from({ length: 8 }).map((_, i) => (
   547→                    <div key={i} className="space-y-2">
   548→                      <Skeleton className="h-28 w-full rounded-xl" />
   549→                      <Skeleton className="h-4 w-3/4" />
   550→                      <Skeleton className="h-4 w-1/2" />
   551→                    </div>
   552→                  ))}
   553→                </div>
   554→              ) : (
   555→                <ScrollArea className="h-full">
   556→                  <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4 pb-4">
