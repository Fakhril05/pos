    67→// ── Helpers ──────────────────────────────────────────────────────────────────
    68→
    69→const formatRp = (amount: number) =>
    70→  new Intl.NumberFormat('id-ID', {
    71→    style: 'currency',
    72→    currency: 'IDR',
    73→    minimumFractionDigits: 0,
    74→  }).format(amount);
    75→
    76→const formatTimeAgo = (dateStr: string): string => {
    77→  const now = Date.now();
    78→  const then = new Date(dateStr).getTime();
    79→  const diffMs = now - then;
    80→  const seconds = Math.floor(diffMs / 1000);
    81→  const minutes = Math.floor(seconds / 60);
    82→  const hours = Math.floor(minutes / 60);
    83→  const days = Math.floor(hours / 24);
    84→  if (days > 0) return `${days} hari lalu`;
    85→  if (hours > 0) return `${hours} jam lalu`;
    86→  if (minutes > 0) return `${minutes} menit lalu`;
    87→  return 'Baru saja';
    88→};
    89→
    90→const formatCompact = (amount: number) => {
    91→  if (amount >= 1_000_000_000) return `${(amount / 1_000_000_000).toFixed(1)}M`;
    92→  if (amount >= 1_000_000) return `${(amount / 1_000_000).toFixed(1)}Jt`;
    93→  if (amount >= 1_000) return `${(amount / 1_000).toFixed(0)}Rb`;
    94→  return String(amount);
    95→};

   489→// ── Live Clock Widget ───────────────────────────────────────────────────────

   491→const HARI = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];
   492→const BULAN = [
   493→  'Januari', 'Februari', 'Maret', 'April', 'Mei', 'Juni',
   494→  'Juli', 'Agustus', 'September', 'Oktober', 'November', 'Desember',
   495→];
   496→
   497→function LiveClock() {
   498→  const [now, setNow] = useState(() => new Date());
   499→
   500→  useEffect(() => {
   501→    const id = setInterval(() => setNow(new Date()), 1000);
   502→    return () => clearInterval(id);
   503→  }, []);
   504→
   505→  const time = now.toLocaleTimeString('id-ID', {
   506→    hour: '2-digit',
   507→    minute: '2-digit',
   508→    second: '2-digit',
   509→    hour12: false,
   510→  });
   511→  const date = `${HARI[now.getDay()]}, ${now.getDate()} ${BULAN[now.getMonth()]} ${now.getFullYear()}`;
   512→
   513→  return (
   514→    <div className="glass-card flex flex-col items-end rounded-xl px-4 py-2.5 text-sm">
   515→      <div className="flex items-center gap-2">
   516→        <span className="relative flex h-2 w-2">
   517→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-300 opacity-75" />

   521→          Live
   522→        </span>

   525→        {time}
   526→      </span>
   527→      <span className="mt-0.5 text-xs text-white/70">{date}</span>
