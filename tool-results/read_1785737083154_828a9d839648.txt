   193→// ── Timeline Item ────────────────────────────────────────────────────────────
   194→
   195→interface TimelineEvent {
   196→  label: string;
   197→  date: string;
   198→  icon: React.ElementType;
   199→  color: string;
   200→}
   201→
   202→function buildTimeline(tx: Transaction): TimelineEvent[] {
   203→  const events: TimelineEvent[] = [
   204→    {
   205→      label: 'Transaksi dibuat',
   206→      date: tx.createdAt,
   207→      icon: Receipt,
   208→      color: 'text-emerald-500',
   209→    },
   210→  ];
   211→  if (tx.paidAt) {
   212→    events.push({
   213→      label: 'Pembayaran diterima',
   214→      date: tx.paidAt,
   215→      icon: CheckCircle2,
   216→      color: 'text-emerald-500',
   217→    });
   218→  }
   219→  if (tx.expiredAt && (tx.status === 'expired' || tx.status === 'failed')) {
   220→    events.push({
   221→      label: 'Transaksi kedaluwarsa',
   222→      date: tx.expiredAt,
   223→      icon: AlertTriangle,
   224→      color: 'text-gray-500',
   225→    });
   226→  }
   227→  if (tx.status === 'failed') {
   228→    events.push({
   229→      label: 'Transaksi gagal',
   230→      date: tx.updatedAt,
   231→      icon: XCircle,
   232→      color: 'text-red-500',
   233→    });
   234→  }
   235→  if (tx.status === 'refunded') {
   236→    events.push({
   237→      label: 'Refund diproses',
   238→      date: tx.updatedAt,
   239→      icon: RotateCcw,
   240→      color: 'text-blue-500',
   241→    });
   242→  }
   243→  return events;
   244→}
   245→
   246→function TimelineItem({ event, isLast }: { event: TimelineEvent; isLast: boolean }) {
   247→  const Icon = event.icon;
   248→  return (
   249→    <div className="flex gap-3">
   250→      <div className="flex flex-col items-center">
   251→        <div
   252→          className={`flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-muted ${event.color}`}
   253→        >
   254→          <Icon className="h-4 w-4" />
   255→        </div>
   256→        {!isLast && <div className="w-px flex-1 bg-border" />}
   257→      </div>
   258→      <div className={`pb-6 ${isLast ? 'pb-0' : ''}`}>
   259→        <p className="text-sm font-medium">{event.label}</p>
   260→        <p className="text-muted-foreground text-xs">{formatDate(event.date)}</p>
   261→      </div>
   262→    </div>
   263→  );
   264→}

   285→    if (!tx) return;
   286→    if (!refundReason.trim()) {
   287→      setRefundReasonError('Alasan refund wajib diisi');
   288→      return;
   289→    }
   290→    setRefunding(true);
   291→    try {
   292→      const res = await fetch(`/api/transactions/${tx.id}`, {
   293→        method: 'POST',
   294→        headers: { 'Content-Type': 'application/json' },
   295→        body: JSON.stringify({ amount: tx.amount, reason: refundReason.trim() }),
   296→      });
   297→      if (!res.ok) {
   298→        const err = await res.json().catch(() => ({ error: 'Gagal memproses refund' }));
   299→        throw new Error(err.error || 'Gagal memproses refund');
   300→      }
   301→      toast.success('Refund berhasil diajukan');
   302→      queryClient.invalidateQueries({ queryKey: ['transactions'] });
   303→      setRefundDialogOpen(false);
   304→      setRefundReason('');
   305→      onOpenChange(false);
   306→    } catch (e) {
   307→      toast.error(e instanceof Error ? e.message : 'Gagal memproses refund');
   308→    } finally {
   309→      setRefunding(false);
   310→    }
   311→  };
   312→
   313→  const handleCopyInvoice = () => {
   314→    if (tx) {
   315→      navigator.clipboard.writeText(tx.invoiceNo);
   316→      toast.success('Nomor invoice disalin');
   317→    }
   318→  };
   319→
   320→  if (!tx) return null;
   321→
   322→  const statusInfo = STATUS_MAP[tx.status] ?? STATUS_MAP.pending;
   323→  const StatusIcon = statusInfo.icon;
   324→
   325→  return (
   326→    <Dialog open={open} onOpenChange={onOpenChange}>
   327→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   328→        <DialogHeader>
   329→          <div className="flex items-center gap-3">
   330→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-emerald-100 dark:bg-emerald-950/50">
   331→              <Receipt className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   332→            </div>
   333→            <div>
   334→              <DialogTitle className="text-base">Detail Transaksi</DialogTitle>
   335→              <DialogDescription>
   336→                <span className="flex items-center gap-1.5 font-mono text-xs">
   337→                  {tx.invoiceNo}
   338→                  <button
   339→                    onClick={handleCopyInvoice}
   340→                    className="inline-flex hover:text-foreground transition-colors"
   341→                  >
   342→                    <Copy className="h-3 w-3" />
   343→                  </button>
   344→                </span>
   345→              </DialogDescription>
   346→            </div>
   347→          </div>
   348→        </DialogHeader>
   349→
   350→        <div className="space-y-5">
   351→          {/* Status Banner */}
   352→          <div
   353→            className={`flex items-center gap-3 rounded-lg border p-4 ${statusInfo.bgClass}`}
   354→          >
   355→            <StatusIcon className="h-5 w-5" />
   356→            <div>
   357→              <p className="text-sm font-semibold">Status: {statusInfo.label}</p>
   358→              <p className="text-xs opacity-80">
   359→                {tx.status === 'success'
   360→                  ? 'Pembayaran telah berhasil diterima'
   361→                  : tx.status === 'pending'
   362→                    ? 'Menunggu pembayaran dari pelanggan'
   363→                    : tx.status === 'failed'
   364→                      ? 'Pembayaran gagal diproses'
   365→                      : tx.status === 'expired'
   366→                        ? 'Batas waktu pembayaran telah habis'
   367→                        : 'Dana telah dikembalikan ke pelanggan'}
   368→              </p>
   369→            </div>
   370→          </div>
   371→
   372→          {/* Amount Section */}
   373→          <div className="rounded-lg border bg-muted/30 p-4">
   374→            <p className="text-muted-foreground mb-1 text-xs font-medium uppercase tracking-wide">
   375→              Jumlah Pembayaran

   520→          <DialogHeader>

   535→            {/* Warning banner */}

   539→                <p className="font-medium text-red-800 dark:text-red-300">Refund sebesar {tx ? formatRp(tx.amount) : ''}</p>
   540→                <p className="text-red-600 dark:text-red-400 mt-0.5">
   541→                  Dana akan dikembalikan ke {tx?.customerName || tx?.customer?.name || 'pelanggan'}. Proses membutuhkan 1-3 hari kerja.

   545→
   546→            {/* Refund reason */}

   568→            {/* Quick reasons */}

   680→// ── Pagination Helper ────────────────────────────────────────────────────────
   681→
   682→function getPaginationRange(current: number, total: number) {
   683→  const delta = 2;
   684→  const range: (number | 'dots')[] = [];
   685→  const left = Math.max(2, current - delta);
   686→  const right = Math.min(total - 1, current + delta);
   687→
   688→  range.push(1);
   689→  if (left > 2) range.push('dots');
   690→  for (let i = left; i <= right; i++) range.push(i);
   691→  if (right < total - 1) range.push('dots');
   692→  if (total > 1) range.push(total);
   693→
   694→  return range;
   695→}
