   149→   149→// ── Page range helper ────────────────────────────────────────────────────────
   150→   150→
   151→   151→function getPageRange(current: number, total: number) {
   152→   152→  const pages: (number | 'ellipsis')[] = [];
   153→   153→  if (total <= 7) {
   154→   154→    for (let i = 1; i <= total; i++) pages.push(i);
   155→   155→  } else {
   156→   156→    pages.push(1);
   157→   157→    if (current > 3) pages.push('ellipsis');
   158→   158→    const start = Math.max(2, current - 1);
   159→   159→    const end = Math.min(total - 1, current + 1);
   160→   160→    for (let i = start; i <= end; i++) pages.push(i);
   161→   161→    if (current < total - 2) pages.push('ellipsis');
   162→   162→    pages.push(total);
   163→   163→  }
   164→   164→  return pages;
   165→   165→}
