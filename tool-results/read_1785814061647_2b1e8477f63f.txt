   145→
   146→export function Kasir() {
   147→  const user = useAppStore((s) => s.user)
   148→  const queryClient = useQueryClient()
   149→
   150→  const [search, setSearch] = useState('')
   151→  const [activeCategory, setActiveCategory] = useState<string | null>(null)
   152→  const [cart, setCart] = useState<CartItem[]>([])
   153→  const [selectedPayment, setSelectedPayment] = useState<(typeof PAYMENT_METHODS)[number] | null>(null)
   154→  const [tendered, setTendered] = useState('')
   155→  const [showSuccess, setShowSuccess] = useState(false)
   156→  const [lastTx, setLastTx] = useState<TxResult | null>(null)
   157→  const [showShiftDialog, setShowShiftDialog] = useState<'open' | 'close' | null>(null)
   158→  const [shiftOpeningCash, setShiftOpeningCash] = useState('')
   159→  const [shiftClosingCash, setShiftClosingCash] = useState('')
   160→  const [shiftNote, setShiftNote] = useState('')
   161→  const [activeTab, setActiveTab] = useState<'pos' | 'history'>('pos')
   162→  const [historyFilter, setHistoryFilter] = useState<'shift' | 'today' | 'all'>('shift')
   163→  const [showTxDetail, setShowTxDetail] = useState<TxHistoryItem | null>(null)
   164→  const searchRef = useRef<HTMLInputElement>(null)
   165→
   166→  // ── Queries ──────────────────────────────────────────────────────
   167→
   168→  const { data: productsData, isLoading: productsLoading } = useQuery({
   169→    queryKey: ['cashier-products', user?.tenantId, search, activeCategory],
   170→    queryFn: async () => {
   171→      const params = new URLSearchParams({ tenantId: user!.tenantId! })
   172→      if (search) params.set('search', search)
   173→      if (activeCategory) params.set('categoryId', activeCategory)
   174→      const res = await fetch(`/api/cashier/products?${params}`)
   175→      if (!res.ok) throw new Error('Gagal memuat produk')
   176→      return res.json() as Promise<{ products: Product[]; categories: Category[] }>
   177→    },
   178→    enabled: !!user?.tenantId,
   179→    refetchInterval: 30000,
   180→  })
   181→
   182→  const { data: shiftData, isLoading: shiftLoading } = useQuery({
   183→    queryKey: ['cashier-shift', user?.outletId, user?.id],
   184→    queryFn: async () => {
   185→      const params = new URLSearchParams({
   186→        outletId: user!.outletId!,
   187→        cashierId: user!.id,
   188→      })
   189→      const res = await fetch(`/api/cashier/shifts?${params}`)
   190→      if (!res.ok) throw new Error('Gagal memuat shift')
   191→      return res.json() as Promise<{ shift: ShiftData | null }>
   192→    },
   193→    enabled: !!user?.outletId && !!user?.id,
   194→  })
   195→
   196→  const { data: historyData, isLoading: historyLoading } = useQuery({
   197→    queryKey: ['cashier-history', user?.id, historyFilter],
   198→    queryFn: async () => {
   199→      const params = new URLSearchParams({
   200→        cashierId: user!.id,
   201→        filter: historyFilter,
   202→      })
   203→      const res = await fetch(`/api/cashier/history?${params}`)
   204→      if (!res.ok) throw new Error('Gagal memuat riwayat')
   205→      return res.json() as Promise<{ transactions: TxHistoryItem[] }>
   206→    },
   207→    enabled: activeTab === 'history' && !!user?.id,
   208→  })
   209→
   210→  // ── Mutations ──────────────────────────────────────────────────
   211→
   212→  const createTransaction = useMutation({
   213→    mutationFn: async () => {
   214→      if (!user || !shiftData?.shift || !selectedPayment) return null
   215→
   216→      const subtotal = cart.reduce((sum, c) => sum + c.price * c.quantity, 0)
   217→      const discountAmount = 0
   218→      const taxAmount = Math.round(subtotal * TAX_RATE)
   219→      const serviceCharge = Math.round(subtotal * SERVICE_CHARGE_RATE)
   220→      const totalAmount = subtotal - discountAmount + taxAmount + serviceCharge
   221→
   222→      const res = await fetch('/api/cashier/transaction', {
   223→        method: 'POST',
   224→        headers: { 'Content-Type': 'application/json' },
   225→        body: JSON.stringify({
   226→          tenantId: user.tenantId,
   227→          outletId: user.outletId,
   228→          cashierId: user.id,
   229→          items: cart.map((c) => ({
   230→            productId: c.productId,
   231→            name: c.name,
   232→            price: c.price,
   233→            buyPrice: c.buyPrice,
   234→            quantity: c.quantity,
   235→            unit: c.unit,
   236→            sku: c.sku,
   237→          })),
   238→          subtotalAmount: subtotal,
   239→          discountAmount,
   240→          taxAmount,
   241→          serviceCharge,
   242→          totalAmount,
   243→          paymentMethod: selectedPayment.method,
   244→          paymentChannel: selectedPayment.channel,
   245→          tendered: selectedPayment.method === 'cash' ? parseFloat(tendered) || 0 : undefined,
   246→        }),
   247→      })
   248→      if (!res.ok) {
   249→        const err = await res.json()
   250→        throw new Error(err.error || 'Gagal membuat transaksi')
   251→      }
   252→      return res.json() as Promise<TxResult>
   253→    },
   254→    onSuccess: (data) => {
   255→      if (data) {
   256→        setLastTx(data)
   257→        setShowSuccess(true)
   258→        setCart([])
   259→        setSelectedPayment(null)
   260→        setTendered('')
   261→        queryClient.invalidateQueries({ queryKey: ['cashier-products'] })
   262→        queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   263→        queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   264→      }
   265→    },
   266→    onError: (err) => {
   267→      toast.error(err.message)
   268→    },
   269→  })
   270→
   271→  const shiftMutation = useMutation({
   272→    mutationFn: async (action: 'open' | 'close') => {
   273→      if (!user) return null
   274→      const body: Record<string, unknown> = {
   275→        action,
   276→        tenantId: user.tenantId,
   277→        outletId: user.outletId,
   278→        cashierId: user.id,
   279→      }
   280→      if (action === 'open') body.openingCash = parseFloat(shiftOpeningCash) || 0
   281→      if (action === 'close') {
   282→        body.closingCash = parseFloat(shiftClosingCash) || 0
   283→        body.note = shiftNote
   284→      }
   285→      const res = await fetch('/api/cashier/shifts', {
   286→        method: 'POST',
   287→        headers: { 'Content-Type': 'application/json' },
   288→        body: JSON.stringify(body),
   289→      })
   290→      if (!res.ok) {
   291→        const err = await res.json()
   292→        throw new Error(err.error || 'Gagal')
   293→      }
   294→      return res.json()
   295→    },
   296→    onSuccess: () => {
   297→      queryClient.invalidateQueries({ queryKey: ['cashier-shift'] })
   298→      queryClient.invalidateQueries({ queryKey: ['cashier-history'] })
   299→      setShowShiftDialog(null)
   300→      setShiftOpeningCash('')
   301→      setShiftClosingCash('')
   302→      setShiftNote('')
   303→      toast.success(
   304→        shiftMutation.variables === 'open' ? 'Shift berhasil dibuka' : 'Shift berhasil ditutup',
   305→      )
   306→    },
   307→    onError: (err) => {
   308→      toast.error(err.message)
   309→    },
   310→  })
   311→
   312→  // ── Cart Logic ──────────────────────────────────────────────────
   313→
   314→  const addToCart = useCallback(
   315→    (product: Product) => {
   316→      setCart((prev) => {
   317→        const existing = prev.find((c) => c.productId === product.id)
   318→        if (existing) {
   319→          if (existing.quantity >= product.stock) {
   320→            toast.warning(`Stok ${product.name} tidak cukup`)
   321→            return prev
   322→          }
   323→          return prev.map((c) =>
   324→            c.productId === product.id
   325→              ? { ...c, quantity: c.quantity + 1 }
   326→              : c,
   327→          )
   328→        }
   329→        if (product.stock <= 0) {
   330→          toast.warning(`${product.name} habis`)
   331→          return prev
   332→        }
   333→        return [
   334→          ...prev,
   335→          {
   336→            productId: product.id,
   337→            name: product.name,
   338→            price: product.sellPrice,
   339→            buyPrice: product.buyPrice,
   340→            quantity: 1,
   341→            unit: product.unit,
   342→            sku: product.sku,
   343→          },
   344→        ]
   345→      })
   346→    },
   347→    [],
   348→  )
   349→
   350→  const updateQty = useCallback((productId: string, delta: number) => {
   351→    setCart((prev) => {
   352→      return prev
   353→        .map((c) => {
   354→          if (c.productId !== productId) return c
   355→          const newQty = c.quantity + delta
   356→          if (newQty <= 0) return null
   357→          return { ...c, quantity: newQty }
   358→        })
   359→        .filter(Boolean) as CartItem[]
   360→    })
   361→  }, [])
   362→
   363→  const removeFromCart = useCallback((productId: string) => {
   364→    setCart((prev) => prev.filter((c) => c.productId !== productId))
   365→  }, [])
   366→
   367→  const clearCart = useCallback(() => {
   368→    setCart([])
   369→    setSelectedPayment(null)
   370→    setTendered('')
   371→  }, [])
   372→
   373→  // ── Computed ──────────────────────────────────────────────────
   374→
   375→  const cartSubtotal = useMemo(() => cart.reduce((s, c) => s + c.price * c.quantity, 0), [cart])
   376→  const cartTax = Math.round(cartSubtotal * TAX_RATE)
   377→  const cartServiceCharge = Math.round(cartSubtotal * SERVICE_CHARGE_RATE)
   378→  const cartTotal = cartSubtotal + cartTax + cartServiceCharge
   379→  const cartItemCount = cart.reduce((s, c) => s + c.quantity, 0)
   380→  const tenderedNum = parseFloat(tendered) || 0
   381→  const change = tenderedNum - cartTotal
   382→
   383→  const filteredProducts = useMemo(() => {
   384→    const prods = productsData?.products || []
   385→    if (!activeCategory) return prods
   386→    return prods.filter((p) => p.category?.id === activeCategory)
   387→  }, [productsData, activeCategory])
   388→
   389→  const categories = productsData?.categories || []
   390→  const activeShift = shiftData?.shift || null
   391→  const isProcessing = createTransaction.isPending || shiftMutation.isPending
   392→
   393→  // ── Keyboard shortcuts ──────────────────────────────────────────
   394→
   395→  useEffect(() => {
   396→    const handleKey = (e: KeyboardEvent) => {
   397→      const tag = (e.target as HTMLElement)?.tagName
   398→      if (tag === 'INPUT' || tag === 'TEXTAREA') return
   399→      if (e.key === 'F2') {
   400→        e.preventDefault()
   401→        searchRef.current?.focus()
   402→      }
   403→      if (e.key === 'Escape') {
   404→        setShowSuccess(false)
   405→        setShowShiftDialog(null)
   406→        setShowTxDetail(null)
   407→      }
   408→    }
   409→    document.addEventListener('keydown', handleKey)
   410→    return () => document.removeEventListener('keydown', handleKey)
   411→  }, [])
   412→
   413→  // ── Print Receipt ──────────────────────────────────────────
   414→
   415→  const printReceipt = useCallback(() => {
   416→    if (!lastTx) return
   417→    const txSubtotal = lastTx.items.reduce((s, i) => s + i.price * i.quantity, 0)
   418→    const txTax = Math.round(txSubtotal * TAX_RATE)
   419→    const txServiceCharge = Math.round(txSubtotal * SERVICE_CHARGE_RATE)
   420→
   421→    const html = generateReceiptHtml({
   422→      invoiceNo: lastTx.invoiceNo,
   423→      tenantName: lastTx.tenantName,
   424→      outletName: lastTx.outletName,
   425→      outletAddress: lastTx.outletAddress,
   426→      outletCity: lastTx.outletCity,
   427→      tenantPhone: lastTx.tenantPhone,
   428→      cashierName: lastTx.cashierName,
   429→      items: lastTx.items.map((i) => ({
   430→        name: i.name,
   431→        quantity: i.quantity,
   432→        price: i.price,
   433→        subtotal: i.price * i.quantity,
   434→      })),
   435→      subtotalAmount: txSubtotal,
   436→      discountAmount: 0,
   437→      taxAmount: txTax,
   438→      serviceCharge: txServiceCharge,
   439→      totalAmount: lastTx.transaction.totalAmount,
   440→      paymentMethod: lastTx.payment.paymentMethod,
   441→      paymentChannel: lastTx.payment.paymentChannel,
   442→      tendered: lastTx.payment.paymentMethod === 'cash' ? parseFloat(tendered) || 0 : undefined,
   443→      change: lastTx.payment.paymentMethod === 'cash' ? change : undefined,
   444→      createdAt: lastTx.transaction.createdAt,
   445→    })
   446→
   447→    const w = window.open('', '_blank', 'width=400,height=700')
   448→    if (w) {
   449→      w.document.write(html)
   450→      w.document.close()
   451→      setTimeout(() => w.print(), 300)
   452→    }
   453→  }, [lastTx, tendered, change, cart])
   454→
   455→  // ── Mobile cart sheet trigger state ─────────────────────────────
   456→  const [showMobileCart, setShowMobileCart] = useState(false)
   457→
   458→  // ── Render ──────────────────────────────────────────────────
   459→
   460→  return (
   461→    <div className="flex h-[calc(100vh-8.5rem)] flex-col gap-4 md:gap-6">
   462→      {/* Tab Switcher */}
   463→      <div className="flex items-center gap-1">
   464→        <button
   465→          onClick={() => setActiveTab('pos')}
   466→          className={cn(
   467→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   468→            activeTab === 'pos'
   469→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   470→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   471→          )}
   472→        >
   473→          <span className="flex items-center gap-2">
   474→            <ShoppingCart className="h-4 w-4" />
   475→            Kasir POS
   476→          </span>
   477→        </button>
   478→        <button
   479→          onClick={() => setActiveTab('history')}
   480→          className={cn(
   481→            'rounded-lg px-4 py-2 text-sm font-semibold transition-all btn-press',
   482→            activeTab === 'history'
   483→              ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25'
   484→              : 'text-muted-foreground hover:bg-muted hover:text-foreground',
   485→          )}
   486→        >
   487→          <span className="flex items-center gap-2">
   488→            <History className="h-4 w-4" />
   489→            Riwayat
   490→          </span>
   491→        </button>
   492→      </div>
   493→
   494→      {activeTab === 'pos' ? (
   495→        <div className="flex min-h-0 flex-1 flex-col gap-4 lg:flex-row">
   496→          {/* ─── LEFT PANEL: Products ─── */}
   497→          <div className="flex min-h-0 flex-1 flex-col gap-3 lg:flex-[3]">
   498→            {/* Search */}
   499→            <div className="relative">
   500→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   501→              <Input
   502→                ref={searchRef}
   503→                value={search}
   504→                onChange={(e) => setSearch(e.target.value)}
   505→                placeholder="Cari produk... (F2)"
   506→                className="pl-10"
   507→              />
   508→            </div>
   509→
   510→            {/* Category Tabs */}
   511→            <div className="relative">
   512→              <ScrollArea className="w-full">
   513→                <div className="flex gap-2 pb-1">
   514→                  <button
   515→                    onClick={() => setActiveCategory(null)}
   516→                    className={cn(
   517→                      'shrink-0 rounded-full px-4 py-1.5 text-xs font-semibold transition-all btn-press',
   518→                      !activeCategory
   519→                        ? 'bg-emerald-500 text-white shadow-md shadow-emerald-500/20'
   520→                        : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground',
   521→                    )}
   522→                  >
   523→                    Semua
   524→                  </button>
   525→                  {categories.map((cat) => (
   526→                    <button
   527→                      key={cat.id}
   528→                      onClick={() => setActiveCategory(cat.id)}
   529→                      className={cn(
   530→                        'shrink-0 rounded-full px-4 py-1.5 text-xs font-semibold transition-all btn-press',
   531→                        activeCategory === cat.id
   532→                          ? 'bg-emerald-500 text-white shadow-md shadow-emerald-500/20'
   533→                          : 'bg-muted text-muted-foreground hover:bg-accent hover:text-foreground',
   534→                      )}
   535→                    >
   536→                      {cat.icon || ''} {cat.name}
   537→                    </button>
   538→                  ))}
   539→                </div>
   540→              </ScrollArea>
   541→            </div>
   542→
   543→            {/* Product Grid */}
   544→            <div className="min-h-0 flex-1">
   545→              {productsLoading ? (
   546→                <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
   547→                  {Array.from({ length: 8 }).map((_, i) => (
   548→                    <div key={i} className="space-y-2">
   549→                      <Skeleton className="h-28 w-full rounded-xl" />
   550→                      <Skeleton className="h-4 w-3/4" />
   551→                      <Skeleton className="h-4 w-1/2" />
   552→                    </div>
   553→                  ))}
   554→                </div>
   555→              ) : (
   556→                <ScrollArea className="h-full">
